- **Sprint**: S10 - Admin 운영 기능
- **Priority**: High
- **Estimated Time**: 3일
- **Dependencies**: S08 (감사 로그 기본 기능), 데이터베이스 접근 로깅 시스템

## 목표
기존 감사 로그 시스템을 고도화하여 상세한 활동 추적, SQL 쿼리 로깅, 데이터 변경 이력 추적, 이상 패턴 감지, 실시간 알림 기능을 구현합니다.

## 상세 요구사항

### 1. 상세 활동 추적 (SQL 쿼리 로깅)
- **SQL 쿼리 캡처**: 모든 데이터베이스 쿼리 로깅
- **실행 계획**: 성능 분석을 위한 실행 계획 저장
- **파라미터 로깅**: 쿼리 파라미터와 바인딩 값
- **성능 메트릭**: 실행 시간, 영향받은 행 수, 리소스 사용량

### 2. 데이터 변경 이력 (변경 전/후)
- **Change Tracking**: 테이블별 변경 사항 추적
- **Diff 생성**: 변경 전후 데이터 비교
- **필드별 추적**: 특정 민감 필드 변경 이력
- **변경 체인**: 연관된 변경 사항 추적

### 3. 이상 패턴 감지
- **비정상적 접근**: 비정상적인 시간대나 위치에서의 접근
- **대량 작업**: 대량 데이터 삭제/수정 감지
- **권한 남용**: 권한을 벗어난 접근 시도
- **실패 패턴**: 반복적인 로그인 실패나 권한 오류

### 4. 실시간 알림 연동
- **즉시 알림**: 중요 이벤트 즉시 알림
- **알림 채널**: 이메일, Slack, Webhook 지원
- **알림 규칙**: 사용자 정의 알림 조건
- **에스컬레이션**: 심각도에 따른 단계별 알림

```typescript
// Enhanced Audit Log System
export class EnhancedAuditLogService {
  private readonly sqlQueryLogger: SqlQueryLogger;
  private readonly changeTracker: DataChangeTracker;
  private readonly anomalyDetector: AnomalyDetector;
  private readonly alertService: AlertService;
  private readonly logProcessor: LogProcessor;

  async logActivity(context: AuditContext, activity: AuditActivity): Promise<void> {
    try {
      // 1. 기본 감사 로그 생성
      const auditLog = await this.createAuditLog(context, activity);
      
      // 2. SQL 쿼리 로깅 (해당하는 경우)
      if (activity.sqlQueries) {
        await this.sqlQueryLogger.logQueries(auditLog.id, activity.sqlQueries);
      }
      
      // 3. 데이터 변경 추적 (해당하는 경우)
      if (activity.dataChanges) {
        await this.changeTracker.trackChanges(auditLog.id, activity.dataChanges);
      }
      
      // 4. 이상 패턴 감지
      const anomalies = await this.anomalyDetector.detectAnomalies(auditLog);
      
      // 5. 실시간 알림 처리
      if (anomalies.length > 0 || activity.severity === 'high') {
        await this.alertService.processAlerts(auditLog, anomalies);
      }
      
      // 6. 로그 저장
      await this.saveAuditLog(auditLog);
      
    } catch (error) {
      this.logger.error(`Failed to log audit activity: ${error.message}`);
      // 감사 로그 실패는 시스템을 중단시키지 않음
    }
  }
}
```

```typescript
export class SqlQueryLogger {
  private readonly connectionService: DatabaseConnectionService;
  private readonly performanceAnalyzer: QueryPerformanceAnalyzer;
  
  async logQueries(auditLogId: string, queries: SqlQuery[]): Promise<void> {
    for (const query of queries) {
      try {
        const queryLog: SqlQueryLog = {
          id: this.generateId(),
          auditLogId,
          query: this.sanitizeQuery(query.sql),
          parameters: this.sanitizeParameters(query.parameters),
          executionTime: query.executionTime,
          affectedRows: query.affectedRows,
          executionPlan: await this.getExecutionPlan(query),
          resourceUsage: await this.getResourceUsage(query),
          timestamp: new Date()
        };
        
        await this.saveSqlQueryLog(queryLog);
        
        // 성능 임계값 확인
        if (query.executionTime > this.getPerformanceThreshold(query.type)) {
          await this.alertSlowQuery(queryLog);
        }
        
      } catch (error) {
        this.logger.error(`Failed to log SQL query: ${error.message}`);
      }
    }
  }
  
  private async getExecutionPlan(query: SqlQuery): Promise<ExecutionPlan | null> {
    if (!query.sql.toLowerCase().startsWith('select')) {
      return null; // EXPLAIN은 SELECT에만 적용
    }
    
    try {
      const explainQuery = `EXPLAIN FORMAT=JSON ${query.sql}`;
      const result = await this.connectionService.executeQuery(explainQuery, query.parameters);
      
      return {
        plan: result[0]?.EXPLAIN,
        estimatedCost: this.extractCost(result[0]?.EXPLAIN),
        indexUsage: this.extractIndexUsage(result[0]?.EXPLAIN)
      };
    } catch (error) {
      this.logger.warn(`Failed to get execution plan: ${error.message}`);
      return null;
    }
  }
  
  private sanitizeQuery(sql: string): string {
    // 민감한 정보 마스킹
    return sql
      .replace(/password\s*=\s*['"][^'"]*['"]/gi, "password='***'")
      .replace(/token\s*=\s*['"][^'"]*['"]/gi, "token='***'")
      .replace(/secret\s*=\s*['"][^'"]*['"]/gi, "secret='***'");
  }
  
  private sanitizeParameters(parameters: any[]): any[] {
    return parameters.map(param => {
      if (typeof param === 'string' && this.isSensitiveValue(param)) {
        return '***';
      }
      return param;
    });
  }
  
  private isSensitiveValue(value: string): boolean {
    const sensitivePatterns = [
      /^[A-Za-z0-9+/]{40,}={0,2}$/, // Base64 encoded (potential tokens)
      /^[0-9a-f]{32,}$/i, // Hex strings (potential hashes)
      /password|secret|token|key/i // Explicit sensitive keywords
    ];
    
    return sensitivePatterns.some(pattern => pattern.test(value));
  }
}
```

### 3. 데이터 변경 추적기

```typescript
export class DataChangeTracker {
  private readonly tableConfigs: Map<string, ChangeTrackingConfig>;
  
  constructor() {
    this.initializeTrackingConfigs();
  }
  
  async trackChanges(auditLogId: string, changes: DataChange[]): Promise<void> {
    for (const change of changes) {
      try {
        const config = this.tableConfigs.get(change.tableName);
        
        if (!config || !config.enabled) {
          continue; // 추적하지 않는 테이블
        }
        
        const changeLog: DataChangeLog = {
          id: this.generateId(),
          auditLogId,
          tableName: change.tableName,
          operation: change.operation, // INSERT, UPDATE, DELETE
          recordId: change.recordId,
          oldValues: await this.sanitizeValues(change.oldValues, config),
          newValues: await this.sanitizeValues(change.newValues, config),
          changedFields: this.getChangedFields(change.oldValues, change.newValues),
          diff: this.generateDiff(change.oldValues, change.newValues),
          timestamp: new Date()
        };
        
        await this.saveDataChangeLog(changeLog);
        
        // 민감한 필드 변경 감지
        if (this.hasSensitiveChanges(changeLog, config)) {
          await this.alertSensitiveChange(changeLog);
        }
        
      } catch (error) {
        this.logger.error(`Failed to track data change: ${error.message}`);
      }
    }
  }
  
  private initializeTrackingConfigs(): void {
    this.tableConfigs.set('users', {
      enabled: true,
      sensitiveFields: ['password', 'email', 'phone'],
      excludeFields: ['last_login', 'login_count'],
      retentionDays: 2555 // 7 years for user data
    });
    
    this.tableConfigs.set('dashboards', {
      enabled: true,
      sensitiveFields: ['config', 'data_source_config'],
      excludeFields: ['view_count', 'last_accessed'],
      retentionDays: 1095 // 3 years
    });
    
    this.tableConfigs.set('data_sources', {
      enabled: true,
      sensitiveFields: ['connection_string', 'username', 'password'],
      excludeFields: [],
      retentionDays: 2555 // 7 years for connection info
    });
    
    // 추가 테이블 설정...
  }
  
  private async sanitizeValues(values: any, config: ChangeTrackingConfig): Promise<any> {
    if (!values) return null;
    
    const sanitized = { ...values };
    
    for (const field of config.sensitiveFields) {
      if (sanitized[field] !== undefined) {
        sanitized[field] = this.maskSensitiveValue(sanitized[field]);
      }
    }
    
    // 제외 필드 제거
    for (const field of config.excludeFields) {
      delete sanitized[field];
    }
    
    return sanitized;
  }
  
  private generateDiff(oldValues: any, newValues: any): FieldDiff[] {
    const diffs: FieldDiff[] = [];
    
    if (!oldValues && newValues) {
      // 새 레코드 생성
      Object.keys(newValues).forEach(key => {
        diffs.push({
          field: key,
          oldValue: null,
          newValue: newValues[key],
          changeType: 'added'
        });
      });
    } else if (oldValues && !newValues) {
      // 레코드 삭제
      Object.keys(oldValues).forEach(key => {
        diffs.push({
          field: key,
          oldValue: oldValues[key],
          newValue: null,
          changeType: 'removed'
        });
      });
    } else if (oldValues && newValues) {
      // 레코드 업데이트
      const allKeys = new Set([...Object.keys(oldValues), ...Object.keys(newValues)]);
      
      allKeys.forEach(key => {
        const oldVal = oldValues[key];
        const newVal = newValues[key];
        
        if (oldVal !== newVal) {
          diffs.push({
            field: key,
            oldValue: oldVal,
            newValue: newVal,
            changeType: oldVal === undefined ? 'added' : 
                       newVal === undefined ? 'removed' : 'modified'
          });
        }
      });
    }
    
    return diffs;
  }
  
  private hasSensitiveChanges(changeLog: DataChangeLog, config: ChangeTrackingConfig): boolean {
    return changeLog.changedFields.some(field => 
      config.sensitiveFields.includes(field)
    );
  }
}
```

### 4. 이상 패턴 감지기

```typescript
export class AnomalyDetector {
  private readonly ruleEngine: AnomalyRuleEngine;
  private readonly mlDetector: MachineLearningDetector;
  private readonly behaviorAnalyzer: UserBehaviorAnalyzer;
  
  async detectAnomalies(auditLog: AuditLog): Promise<Anomaly[]> {
    const anomalies: Anomaly[] = [];
    
    try {
      // 1. 규칙 기반 이상 감지
      const ruleAnomalies = await this.ruleEngine.checkRules(auditLog);
      anomalies.push(...ruleAnomalies);
      
      // 2. 행동 패턴 분석
      const behaviorAnomalies = await this.behaviorAnalyzer.analyzeBehavior(auditLog);
      anomalies.push(...behaviorAnomalies);
      
      // 3. ML 기반 이상 감지
      const mlAnomalies = await this.mlDetector.detectAnomalies(auditLog);
      anomalies.push(...mlAnomalies);
      
      // 4. 이상 징후 저장
      for (const anomaly of anomalies) {
        await this.saveAnomaly(anomaly);
      }
      
      return anomalies;
      
    } catch (error) {
      this.logger.error(`Anomaly detection failed: ${error.message}`);
      return [];
    }
  }
}

export class AnomalyRuleEngine {
  private readonly rules: AnomalyRule[];
  
  constructor() {
    this.rules = [
      new OffHoursAccessRule(),
      new SuspiciousLocationRule(),
      new MassDataOperationRule(),
      new PrivilegeEscalationRule(),
      new RepeatedFailureRule(),
      new UnusualQueryPatternRule()
    ];
  }
  
  async checkRules(auditLog: AuditLog): Promise<Anomaly[]> {
    const anomalies: Anomaly[] = [];
    
    for (const rule of this.rules) {
      try {
        const result = await rule.evaluate(auditLog);
        
        if (result.isAnomaly) {
          anomalies.push({
            id: this.generateId(),
            auditLogId: auditLog.id,
            type: rule.getType(),
            severity: result.severity,
            description: result.description,
            confidence: result.confidence,
            metadata: result.metadata,
            detectedAt: new Date()
          });
        }
      } catch (error) {
        this.logger.error(`Rule evaluation failed for ${rule.getType()}: ${error.message}`);
      }
    }
    
    return anomalies;
  }
}

// 이상 감지 규칙 예시
export class OffHoursAccessRule implements AnomalyRule {
  getType(): string {
    return 'off_hours_access';
  }
  
  async evaluate(auditLog: AuditLog): Promise<RuleResult> {
    const accessTime = new Date(auditLog.timestamp);
    const hour = accessTime.getHours();
    const isWeekend = [0, 6].includes(accessTime.getDay());
    
    // 업무시간 외 접근 (오후 6시 ~ 오전 9시, 주말)
    const isOffHours = hour < 9 || hour > 18 || isWeekend;
    
    if (isOffHours && auditLog.action.includes('admin')) {
      return {
        isAnomaly: true,
        severity: 'medium',
        confidence: 0.8,
        description: `Administrative action performed during off-hours: ${auditLog.action}`,
        metadata: {
          accessTime: accessTime.toISOString(),
          hour,
          isWeekend
        }
      };
    }
    
    return { isAnomaly: false };
  }
}

export class MassDataOperationRule implements AnomalyRule {
  getType(): string {
    return 'mass_data_operation';
  }
  
  async evaluate(auditLog: AuditLog): Promise<RuleResult> {
    // SQL 쿼리에서 대량 데이터 작업 감지
    if (auditLog.metadata?.sqlQueries) {
      for (const query of auditLog.metadata.sqlQueries) {
        if (query.affectedRows > 1000) {
          return {
            isAnomaly: true,
            severity: 'high',
            confidence: 0.9,
            description: `Mass data operation detected: ${query.affectedRows} rows affected`,
            metadata: {
              query: query.query,
              affectedRows: query.affectedRows
            }
          };
        }
      }
    }
    
    return { isAnomaly: false };
  }
}
```

### 5. 실시간 알림 서비스

```typescript
export class AlertService {
  private readonly notificationChannels: Map<string, NotificationChannel>;
  private readonly alertRules: AlertRule[];
  
  constructor() {
    this.initializeChannels();
    this.initializeAlertRules();
  }
  
  async processAlerts(auditLog: AuditLog, anomalies: Anomaly[]): Promise<void> {
    try {
      // 1. 알림 규칙 평가
      const alerts = await this.evaluateAlertRules(auditLog, anomalies);
      
      // 2. 중복 알림 제거
      const deduplicatedAlerts = await this.deduplicateAlerts(alerts);
      
      // 3. 알림 발송
      for (const alert of deduplicatedAlerts) {
        await this.sendAlert(alert);
      }
      
    } catch (error) {
      this.logger.error(`Alert processing failed: ${error.message}`);
    }
  }
  
  private async evaluateAlertRules(auditLog: AuditLog, anomalies: Anomaly[]): Promise<Alert[]> {
    const alerts: Alert[] = [];
    
    for (const rule of this.alertRules) {
      try {
        const shouldAlert = await rule.shouldAlert(auditLog, anomalies);
        
        if (shouldAlert.trigger) {
          alerts.push({
            id: this.generateId(),
            auditLogId: auditLog.id,
            ruleId: rule.id,
            severity: shouldAlert.severity,
            title: shouldAlert.title,
            message: shouldAlert.message,
            channels: shouldAlert.channels,
            metadata: shouldAlert.metadata,
            createdAt: new Date()
          });
        }
      } catch (error) {
        this.logger.error(`Alert rule evaluation failed: ${error.message}`);
      }
    }
    
    return alerts;
  }
  
  private async sendAlert(alert: Alert): Promise<void> {
    for (const channelName of alert.channels) {
      try {
        const channel = this.notificationChannels.get(channelName);
        
        if (channel) {
          await channel.send({
            title: alert.title,
            message: alert.message,
            severity: alert.severity,
            metadata: alert.metadata
          });
          
          this.logger.info(`Alert sent via ${channelName}: ${alert.title}`);
        }
      } catch (error) {
        this.logger.error(`Failed to send alert via ${channelName}: ${error.message}`);
      }
    }
  }
  
  private initializeChannels(): void {
    this.notificationChannels.set('email', new EmailNotificationChannel());
    this.notificationChannels.set('slack', new SlackNotificationChannel());
    this.notificationChannels.set('webhook', new WebhookNotificationChannel());
    this.notificationChannels.set('sms', new SmsNotificationChannel());
  }
  
  private initializeAlertRules(): void {
    this.alertRules = [
      new HighSeverityAnomalyRule(),
      new AdminActionRule(),
      new DataBreachRule(),
      new SystemFailureRule(),
      new SecurityIncidentRule()
    ];
  }
}

// 알림 채널 구현 예시
export class SlackNotificationChannel implements NotificationChannel {
  private readonly webhookUrl: string;
  
  constructor() {
    this.webhookUrl = process.env.SLACK_WEBHOOK_URL;
  }
  
  async send(notification: Notification): Promise<void> {
    if (!this.webhookUrl) {
      throw new Error('Slack webhook URL not configured');
    }
    
    const payload = {
      username: 'VanillaMeta Security',
      icon_emoji: this.getSeverityEmoji(notification.severity),
      attachments: [{
        color: this.getSeverityColor(notification.severity),
        title: notification.title,
        text: notification.message,
        fields: this.formatMetadata(notification.metadata),
        footer: 'VanillaMeta Audit System',
        ts: Math.floor(Date.now() / 1000)
      }]
    };
    
    const response = await fetch(this.webhookUrl, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload)
    });
    
    if (!response.ok) {
      throw new Error(`Slack notification failed: ${response.statusText}`);
    }
  }
  
  private getSeverityEmoji(severity: string): string {
    const emojis = {
      low: ':information_source:',
      medium: ':warning:',
      high: ':rotating_light:',
      critical: ':fire:'
    };
    return emojis[severity] || ':question:';
  }
  
  private getSeverityColor(severity: string): string {
    const colors = {
      low: '#36a64f',    // green
      medium: '#ffeb3b',  // yellow
      high: '#ff9800',    // orange
      critical: '#f44336'  // red
    };
    return colors[severity] || '#9e9e9e';
  }
}
```

### 2. 비동기 로그 처리
```typescript
export class AsyncAuditLogger {
  private readonly logQueue: Queue<AuditLogEntry>;
  private readonly batchProcessor: BatchProcessor;
  
  constructor() {
    this.logQueue = new Queue('audit-logs');
    this.batchProcessor = new BatchProcessor({
      batchSize: 100,
      flushInterval: 5000 // 5초마다 플러시
    });
    
    this.startProcessing();
  }
  
  async logAsync(entry: AuditLogEntry): Promise<void> {
    await this.logQueue.add('process-log', entry);
  }
  
  private startProcessing(): void {
    this.logQueue.process('process-log', async (job) => {
      await this.batchProcessor.add(job.data);
    });
    
    this.batchProcessor.on('flush', async (batch) => {
      await this.processBatch(batch);
    });
  }
}
```
