```typescript
// System Monitoring Service
export class SystemMonitoringService {
  private readonly metricsCollector: MetricsCollector;
  private readonly healthChecker: HealthChecker;
  private readonly alertManager: AlertManager;
  private readonly dashboardService: MonitoringDashboardService;
  private readonly storageService: MetricsStorageService;

  constructor() {
    this.metricsCollector = new MetricsCollector();
    this.healthChecker = new HealthChecker();
    this.alertManager = new AlertManager();
    this.dashboardService = new MonitoringDashboardService();
    this.storageService = new MetricsStorageService();
    
    this.startMonitoring();
  }

  private startMonitoring(): void {
    // 메트릭 수집 시작 (매 30초)
    setInterval(async () => {
      try {
        const metrics = await this.collectSystemMetrics();
        await this.processMetrics(metrics);
      } catch (error) {
        this.logger.error(`Metrics collection failed: ${error.message}`);
      }
    }, 30000);

    // 헬스체크 시작 (매 1분)
    setInterval(async () => {
      try {
        const healthStatus = await this.performHealthCheck();
        await this.processHealthStatus(healthStatus);
      } catch (error) {
        this.logger.error(`Health check failed: ${error.message}`);
      }
    }, 60000);
  }

  private async collectSystemMetrics(): Promise<SystemMetrics> {
    const systemMetrics = await this.metricsCollector.collectAll();
    
    return {
      timestamp: new Date(),
      cpu: systemMetrics.cpu,
      memory: systemMetrics.memory,
      disk: systemMetrics.disk,
      network: systemMetrics.network,
      database: systemMetrics.database,
      application: systemMetrics.application
    };
  }

  private async processMetrics(metrics: SystemMetrics): Promise<void> {
    // 1. 메트릭 저장
    await this.storageService.store(metrics);

    // 2. 임계값 확인 및 알림
    const alerts = await this.alertManager.checkThresholds(metrics);
    
    for (const alert of alerts) {
      await this.alertManager.sendAlert(alert);
    }

    // 3. 대시보드 업데이트
    await this.dashboardService.updateMetrics(metrics);
  }
}
```

```typescript
export class MetricsCollector {
  private readonly osUtils: OSUtils;
  private readonly dbMonitor: DatabaseMonitor;
  private readonly appMonitor: ApplicationMonitor;

  async collectAll(): Promise<RawMetrics> {
    const [
      cpuMetrics,
      memoryMetrics,
      diskMetrics,
      networkMetrics,
      databaseMetrics,
      applicationMetrics
    ] = await Promise.all([
      this.collectCpuMetrics(),
      this.collectMemoryMetrics(),
      this.collectDiskMetrics(),
      this.collectNetworkMetrics(),
      this.collectDatabaseMetrics(),
      this.collectApplicationMetrics()
    ]);

    return {
      cpu: cpuMetrics,
      memory: memoryMetrics,
      disk: diskMetrics,
      network: networkMetrics,
      database: databaseMetrics,
      application: applicationMetrics
    };
  }

  private async collectCpuMetrics(): Promise<CpuMetrics> {
    const cpuUsage = await this.osUtils.getCpuUsage();
    const loadAverage = await this.osUtils.getLoadAverage();
    const processes = await this.osUtils.getTopProcesses();

    return {
      usage: cpuUsage,
      loadAverage: {
        '1min': loadAverage[0],
        '5min': loadAverage[1],
        '15min': loadAverage[2]
      },
      coreCount: os.cpus().length,
      topProcesses: processes.slice(0, 10)
    };
  }

  private async collectMemoryMetrics(): Promise<MemoryMetrics> {
    const totalMemory = os.totalmem();
    const freeMemory = os.freemem();
    const usedMemory = totalMemory - freeMemory;
    
    // 애플리케이션별 메모리 사용량
    const processMemory = process.memoryUsage();
    
    return {
      total: totalMemory,
      used: usedMemory,
      free: freeMemory,
      usagePercentage: (usedMemory / totalMemory) * 100,
      process: {
        rss: processMemory.rss,
        heapTotal: processMemory.heapTotal,
        heapUsed: processMemory.heapUsed,
        external: processMemory.external
      },
      swapUsage: await this.osUtils.getSwapUsage()
    };
  }

  private async collectDiskMetrics(): Promise<DiskMetrics[]> {
    const diskUsage = await this.osUtils.getDiskUsage();
    const diskIO = await this.osUtils.getDiskIO();

    return diskUsage.map(disk => ({
      device: disk.device,
      mountpoint: disk.mountpoint,
      total: disk.total,
      used: disk.used,
      available: disk.available,
      usagePercentage: (disk.used / disk.total) * 100,
      io: diskIO[disk.device] || { readBytes: 0, writeBytes: 0, readOps: 0, writeOps: 0 }
    }));
  }

  private async collectNetworkMetrics(): Promise<NetworkMetrics> {
    const networkStats = await this.osUtils.getNetworkStats();
    const activeConnections = await this.osUtils.getActiveConnections();

    return {
      interfaces: networkStats.map(iface => ({
        name: iface.name,
        bytesReceived: iface.rx_bytes,
        bytesSent: iface.tx_bytes,
        packetsReceived: iface.rx_packets,
        packetsSent: iface.tx_packets,
        errors: iface.rx_errors + iface.tx_errors
      })),
      activeConnections: activeConnections.length,
      connectionsByState: this.groupConnectionsByState(activeConnections)
    };
  }

  private async collectDatabaseMetrics(): Promise<DatabaseMetrics> {
    try {
      const connectionPool = await this.dbMonitor.getConnectionPoolStats();
      const slowQueries = await this.dbMonitor.getSlowQueries();
      const locks = await this.dbMonitor.getCurrentLocks();
      const replicationStatus = await this.dbMonitor.getReplicationStatus();

      return {
        connectionPool: {
          active: connectionPool.active,
          idle: connectionPool.idle,
          waiting: connectionPool.waiting,
          total: connectionPool.total
        },
        performance: {
          slowQueries: slowQueries.length,
          averageQueryTime: slowQueries.reduce((sum, q) => sum + q.duration, 0) / slowQueries.length || 0,
          qps: await this.dbMonitor.getQueriesPerSecond()
        },
        locks: {
          currentLocks: locks.length,
          deadlocks: await this.dbMonitor.getDeadlockCount(),
          lockWaits: locks.filter(l => l.waiting).length
        },
        replication: replicationStatus
      };
    } catch (error) {
      this.logger.error(`Database metrics collection failed: ${error.message}`);
      return this.getDefaultDatabaseMetrics();
    }
  }

  private async collectApplicationMetrics(): Promise<ApplicationMetrics> {
    const httpMetrics = await this.appMonitor.getHttpMetrics();
    const errorMetrics = await this.appMonitor.getErrorMetrics();
    const businessMetrics = await this.appMonitor.getBusinessMetrics();

    return {
      http: {
        requestsPerSecond: httpMetrics.rps,
        averageResponseTime: httpMetrics.avgResponseTime,
        responseTimeP95: httpMetrics.p95ResponseTime,
        responseTimeP99: httpMetrics.p99ResponseTime,
        statusCodes: httpMetrics.statusCodes
      },
      errors: {
        errorRate: errorMetrics.rate,
        errorCount: errorMetrics.count,
        topErrors: errorMetrics.topErrors
      },
      business: {
        activeUsers: businessMetrics.activeUsers,
        activeDashboards: businessMetrics.activeDashboards,
        queryExecutions: businessMetrics.queryExecutions,
        dataSourceConnections: businessMetrics.dataSourceConnections
      }
    };
  }
}
```

### 3. 헬스체크 시스템

```typescript
export class HealthChecker {
  private readonly healthChecks: Map<string, HealthCheck>;

  constructor() {
    this.healthChecks = new Map();
    this.registerHealthChecks();
  }

  private registerHealthChecks(): void {
    this.healthChecks.set('database', new DatabaseHealthCheck());
    this.healthChecks.set('redis', new RedisHealthCheck());
    this.healthChecks.set('elasticsearch', new ElasticsearchHealthCheck());
    this.healthChecks.set('s3', new S3HealthCheck());
    this.healthChecks.set('external_apis', new ExternalApiHealthCheck());
    this.healthChecks.set('disk_space', new DiskSpaceHealthCheck());
    this.healthChecks.set('memory', new MemoryHealthCheck());
  }

  async performHealthCheck(): Promise<HealthStatus> {
    const results: HealthCheckResult[] = [];
    let overallStatus: 'healthy' | 'degraded' | 'unhealthy' = 'healthy';

    for (const [name, checker] of this.healthChecks) {
      try {
        const result = await this.executeWithTimeout(
          () => checker.check(),
          10000 // 10초 타임아웃
        );

        results.push({
          name,
          status: result.status,
          message: result.message,
          details: result.details,
          responseTime: result.responseTime,
          timestamp: new Date()
        });

        // 전체 상태 결정
        if (result.status === 'unhealthy') {
          overallStatus = 'unhealthy';
        } else if (result.status === 'degraded' && overallStatus === 'healthy') {
          overallStatus = 'degraded';
        }

      } catch (error) {
        results.push({
          name,
          status: 'unhealthy',
          message: `Health check failed: ${error.message}`,
          details: { error: error.stack },
          responseTime: 10000,
          timestamp: new Date()
        });
        overallStatus = 'unhealthy';
      }
    }

    return {
      overall: overallStatus,
      checks: results,
      timestamp: new Date()
    };
  }

  private async executeWithTimeout<T>(
    operation: () => Promise<T>,
    timeoutMs: number
  ): Promise<T> {
    return Promise.race([
      operation(),
      new Promise<never>((_, reject) =>
        setTimeout(() => reject(new Error('Health check timeout')), timeoutMs)
      )
    ]);
  }
}

// 개별 헬스체크 구현 예시
export class DatabaseHealthCheck implements HealthCheck {
  private readonly connectionService: DatabaseConnectionService;

  constructor() {
    this.connectionService = new DatabaseConnectionService();
  }

  async check(): Promise<HealthCheckResult> {
    const startTime = Date.now();

    try {
      // 1. 기본 연결 테스트
      const connection = await this.connectionService.getConnection();
      
      // 2. 간단한 쿼리 실행
      const result = await connection.query('SELECT 1 as health_check');
      
      if (result.length === 0 || result[0].health_check !== 1) {
        throw new Error('Invalid query result');
      }

      // 3. 연결 풀 상태 확인
      const poolStats = await this.connectionService.getPoolStats();
      const connectionUtilization = poolStats.active / poolStats.total;

      let status: HealthStatus = 'healthy';
      let message = 'Database is healthy';

      if (connectionUtilization > 0.9) {
        status = 'degraded';
        message = 'High connection pool utilization';
      }

      if (connectionUtilization >= 1) {
        status = 'unhealthy';
        message = 'Connection pool exhausted';
      }

      return {
        status,
        message,
        details: {
          connectionPool: poolStats,
          responseTime: Date.now() - startTime
        },
        responseTime: Date.now() - startTime
      };

    } catch (error) {
      return {
        status: 'unhealthy',
        message: `Database connection failed: ${error.message}`,
        details: { error: error.message },
        responseTime: Date.now() - startTime
      };
    }
  }
}

export class DiskSpaceHealthCheck implements HealthCheck {
  async check(): Promise<HealthCheckResult> {
    const startTime = Date.now();

    try {
      const diskUsage = await this.getDiskUsage('/');
      const usagePercentage = (diskUsage.used / diskUsage.total) * 100;

      let status: HealthStatus = 'healthy';
      let message = `Disk usage: ${usagePercentage.toFixed(1)}%`;

      if (usagePercentage > 80) {
        status = 'degraded';
        message = `High disk usage: ${usagePercentage.toFixed(1)}%`;
      }

      if (usagePercentage > 95) {
        status = 'unhealthy';
        message = `Critical disk usage: ${usagePercentage.toFixed(1)}%`;
      }

      return {
        status,
        message,
        details: {
          usage: diskUsage,
          usagePercentage
        },
        responseTime: Date.now() - startTime
      };

    } catch (error) {
      return {
        status: 'unhealthy',
        message: `Disk space check failed: ${error.message}`,
        details: { error: error.message },
        responseTime: Date.now() - startTime
      };
    }
  }

  private async getDiskUsage(path: string): Promise<DiskUsage> {
    return new Promise((resolve, reject) => {
      require('fs').stat(path, (err, stats) => {
        if (err) {
          reject(err);
          return;
        }

        require('child_process').exec(`df -k ${path}`, (error, stdout) => {
          if (error) {
            reject(error);
            return;
          }

          const lines = stdout.trim().split('\n');
          const data = lines[1].split(/\s+/);
          
          resolve({
            total: parseInt(data[1]) * 1024,
            used: parseInt(data[2]) * 1024,
            available: parseInt(data[3]) * 1024
          });
        });
      });
    });
  }
}
```

### 4. 알림 관리자

```typescript
export class AlertManager {
  private readonly thresholds: Map<string, AlertThreshold>;
  private readonly notificationService: NotificationService;
  private readonly alertHistory: AlertHistory;

  constructor() {
    this.thresholds = new Map();
    this.notificationService = new NotificationService();
    this.alertHistory = new AlertHistory();
    this.initializeThresholds();
  }

  private initializeThresholds(): void {
    // CPU 임계값
    this.thresholds.set('cpu.usage', {
      warning: 70,
      critical: 90,
      duration: 300000 // 5분 지속
    });

    // 메모리 임계값
    this.thresholds.set('memory.usage', {
      warning: 80,
      critical: 95,
      duration: 180000 // 3분 지속
    });

    // 디스크 임계값
    this.thresholds.set('disk.usage', {
      warning: 80,
      critical: 95,
      duration: 600000 // 10분 지속
    });

    // 데이터베이스 연결 풀 임계값
    this.thresholds.set('database.connection_pool', {
      warning: 80,
      critical: 95,
      duration: 120000 // 2분 지속
    });

    // 응답 시간 임계값
    this.thresholds.set('http.response_time', {
      warning: 5000, // 5초
      critical: 10000, // 10초
      duration: 300000 // 5분 지속
    });
  }

  async checkThresholds(metrics: SystemMetrics): Promise<Alert[]> {
    const alerts: Alert[] = [];

    try {
      // CPU 사용률 확인
      if (metrics.cpu.usage > this.thresholds.get('cpu.usage').warning) {
        const alert = await this.createAlert('cpu_usage', metrics.cpu.usage, metrics.timestamp);
        if (alert) alerts.push(alert);
      }

      // 메모리 사용률 확인
      if (metrics.memory.usagePercentage > this.thresholds.get('memory.usage').warning) {
        const alert = await this.createAlert('memory_usage', metrics.memory.usagePercentage, metrics.timestamp);
        if (alert) alerts.push(alert);
      }

      // 디스크 사용률 확인
      for (const disk of metrics.disk) {
        if (disk.usagePercentage > this.thresholds.get('disk.usage').warning) {
          const alert = await this.createAlert('disk_usage', disk.usagePercentage, metrics.timestamp, {
            device: disk.device,
            mountpoint: disk.mountpoint
          });
          if (alert) alerts.push(alert);
        }
      }

      // 데이터베이스 연결 풀 확인
      const poolUtilization = (metrics.database.connectionPool.active / metrics.database.connectionPool.total) * 100;
      if (poolUtilization > this.thresholds.get('database.connection_pool').warning) {
        const alert = await this.createAlert('db_connection_pool', poolUtilization, metrics.timestamp);
        if (alert) alerts.push(alert);
      }

      // HTTP 응답 시간 확인
      if (metrics.application.http.averageResponseTime > this.thresholds.get('http.response_time').warning) {
        const alert = await this.createAlert('response_time', metrics.application.http.averageResponseTime, metrics.timestamp);
        if (alert) alerts.push(alert);
      }

      return alerts;

    } catch (error) {
      this.logger.error(`Threshold checking failed: ${error.message}`);
      return [];
    }
  }

  private async createAlert(
    metricName: string, 
    value: number, 
    timestamp: Date,
    metadata?: any
  ): Promise<Alert | null> {
    const threshold = this.thresholds.get(metricName);
    if (!threshold) return null;

    // 심각도 결정
    let severity: AlertSeverity = 'info';
    if (value >= threshold.critical) {
      severity = 'critical';
    } else if (value >= threshold.warning) {
      severity = 'warning';
    } else {
      return null; // 임계값 미만
    }

    // 중복 알림 방지 - 같은 메트릭에 대한 최근 알림 확인
    const recentAlert = await this.alertHistory.getRecentAlert(metricName, threshold.duration);
    if (recentAlert && recentAlert.severity === severity) {
      return null; // 중복 알림 방지
    }

    const alert: Alert = {
      id: this.generateAlertId(),
      metricName,
      severity,
      value,
      threshold: severity === 'critical' ? threshold.critical : threshold.warning,
      message: this.generateAlertMessage(metricName, value, severity),
      timestamp,
      metadata,
      resolved: false
    };

    // 알림 이력에 추가
    await this.alertHistory.saveAlert(alert);

    return alert;
  }

  async sendAlert(alert: Alert): Promise<void> {
    try {
      const notification: NotificationRequest = {
        title: `${alert.severity.toUpperCase()}: ${alert.metricName}`,
        message: alert.message,
        severity: alert.severity,
        metadata: {
          alertId: alert.id,
          metric: alert.metricName,
          value: alert.value,
          threshold: alert.threshold
        }
      };

      // 심각도에 따른 알림 채널 선택
      const channels = this.getNotificationChannels(alert.severity);
      
      for (const channel of channels) {
        await this.notificationService.send(channel, notification);
      }

      this.logger.info(`Alert sent: ${alert.id} - ${alert.message}`);

    } catch (error) {
      this.logger.error(`Failed to send alert ${alert.id}: ${error.message}`);
    }
  }

  private getNotificationChannels(severity: AlertSeverity): string[] {
    switch (severity) {
      case 'critical':
        return ['email', 'slack', 'sms', 'webhook'];
      case 'warning':
        return ['email', 'slack'];
      case 'info':
        return ['slack'];
      default:
        return [];
    }
  }

  private generateAlertMessage(metricName: string, value: number, severity: AlertSeverity): string {
    const messages = {
      cpu_usage: `CPU usage is ${severity}: ${value.toFixed(1)}%`,
      memory_usage: `Memory usage is ${severity}: ${value.toFixed(1)}%`,
      disk_usage: `Disk usage is ${severity}: ${value.toFixed(1)}%`,
      db_connection_pool: `Database connection pool utilization is ${severity}: ${value.toFixed(1)}%`,
      response_time: `Average response time is ${severity}: ${value.toFixed(0)}ms`
    };

    return messages[metricName] || `${metricName} threshold ${severity}: ${value}`;
  }
}
```

```tsx
                color: currentMetrics?.cpu.usage > 80 ? '#f5222d' : '#3f8600' 
              }}
            />
            <Progress
              percent={currentMetrics?.cpu.usage}
              size="small"
              status={currentMetrics?.cpu.usage > 80 ? 'exception' : 'normal'}
            />
          </Card>
        </Col>

        <Col span={6}>
          <Card>
            <Statistic
              title="메모리 사용률"
              value={currentMetrics?.memory.usagePercentage}
              suffix="%"
              precision={1}
              valueStyle={{
                color: currentMetrics?.memory.usagePercentage > 80 ? '#f5222d' : '#3f8600'
              }}
            />
            <Progress
              percent={currentMetrics?.memory.usagePercentage}
              size="small"
              status={currentMetrics?.memory.usagePercentage > 80 ? 'exception' : 'normal'}
            />
          </Card>
        </Col>

        <Col span={6}>
          <Card>
            <Statistic
              title="디스크 사용률"
              value={getMaxDiskUsage(currentMetrics?.disk)}
              suffix="%"
              precision={1}
              valueStyle={{
                color: getMaxDiskUsage(currentMetrics?.disk) > 80 ? '#f5222d' : '#3f8600'
              }}
            />
          </Card>
        </Col>

        <Col span={6}>
          <Card>
            <Statistic
              title="활성 연결"
              value={currentMetrics?.database.connectionPool.active}
              suffix={`/ ${currentMetrics?.database.connectionPool.total}`}
              valueStyle={{
                color: getConnectionPoolColor(currentMetrics?.database.connectionPool)
              }}
            />
          </Card>
        </Col>
      </Row>

      {/* 상세 차트 */}
      <Row gutter={[16, 16]} className="mb-6">
        <Col span={12}>
          <Card title="CPU & 메모리 사용률">
            <LineChart
              data={historicalMetrics}
              xField="timestamp"
              yField="value"
              seriesField="metric"
              smooth
              height={300}
            />
          </Card>
        </Col>

        <Col span={12}>
          <Card title="네트워크 트래픽">
            <AreaChart
              data={getNetworkTrafficData(historicalMetrics)}
              xField="timestamp"
              yField="bytes"
              seriesField="direction"
              height={300}
            />
          </Card>
        </Col>
      </Row>

      {/* 데이터베이스 모니터링 */}
      <Row gutter={[16, 16]} className="mb-6">
        <Col span={8}>
          <Card title="데이터베이스 성능">
            <div className="space-y-4">
              <div>
                <div className="flex justify-between mb-1">
                  <span>쿼리/초</span>
                  <span>{currentMetrics?.database.performance.qps}</span>
                </div>
                <Progress percent={Math.min(currentMetrics?.database.performance.qps * 2, 100)} size="small" />
              </div>
              
              <div>
                <div className="flex justify-between mb-1">
                  <span>평균 응답시간</span>
                  <span>{currentMetrics?.database.performance.averageQueryTime}ms</span>
                </div>
                <Progress
                  percent={Math.min(currentMetrics?.database.performance.averageQueryTime / 10, 100)}
                  size="small"
                  status={currentMetrics?.database.performance.averageQueryTime > 1000 ? 'exception' : 'normal'}
                />
              </div>

              <div>
                <div className="flex justify-between mb-1">
                  <span>느린 쿼리</span>
                  <span>{currentMetrics?.database.performance.slowQueries}</span>
                </div>
              </div>
            </div>
          </Card>
        </Col>

        <Col span={8}>
          <Card title="애플리케이션 메트릭">
            <div className="space-y-4">
              <Statistic
                title="요청/초"
                value={currentMetrics?.application.http.requestsPerSecond}
                precision={1}
              />
              
              <Statistic
                title="평균 응답시간"
                value={currentMetrics?.application.http.averageResponseTime}
                suffix="ms"
                precision={0}
              />
              
              <Statistic
                title="에러율"
                value={currentMetrics?.application.errors.errorRate}
                suffix="%"
                precision={2}
                valueStyle={{
                  color: currentMetrics?.application.errors.errorRate > 5 ? '#f5222d' : '#3f8600'
                }}
              />
            </div>
          </Card>
        </Col>

        <Col span={8}>
          <Card title="비즈니스 메트릭">
            <div className="space-y-4">
              <Statistic
                title="활성 사용자"
                value={currentMetrics?.application.business.activeUsers}
              />
              
              <Statistic
                title="활성 대시보드"
                value={currentMetrics?.application.business.activeDashboards}
              />
              
              <Statistic
                title="쿼리 실행"
                value={currentMetrics?.application.business.queryExecutions}
              />
            </div>
          </Card>
        </Col>
      </Row>

      {/* 헬스체크 상태 */}
      <Card title="서비스 헬스체크">
        <Row gutter={[16, 16]}>
          {healthStatus.checks.map(check => (
            <Col span={6} key={check.name}>
              <div className="p-4 border rounded-lg">
                <div className="flex items-center justify-between mb-2">
                  <span className="font-medium capitalize">{check.name.replace('_', ' ')}</span>
                  <Tag color={getHealthColor(check.status)}>
                    {getHealthLabel(check.status)}
                  </Tag>
                </div>
                <div className="text-sm text-gray-600">
                  {check.message}
                </div>
                <div className="text-xs text-gray-400 mt-1">
                  응답시간: {check.responseTime}ms
                </div>
              </div>
            </Col>
          ))}
        </Row>
      </Card>
    </div>
  );
};
```
