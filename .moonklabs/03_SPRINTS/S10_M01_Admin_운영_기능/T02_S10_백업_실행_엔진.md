- **Dependencies**: T01_S10 (백업 시스템 설계), mysqldump, AWS CLI

## 목표
실제 백업 작업을 수행하는 백업 실행 엔진을 구현합니다. 데이터베이스, 애플리케이션 설정, 파일 시스템의 백업 및 압축/암호화 기능을 포함합니다.

## 상세 요구사항

### 1. 데이터베이스 백업 (mysqldump)
- **전체 덤프**: 모든 테이블과 데이터
- **스키마 전용**: 구조만 백업
- **데이터 전용**: 데이터만 백업
- **특정 테이블**: 선택적 테이블 백업
- **트랜잭션 안전**: --single-transaction 옵션

### 2. 대시보드/위젯 설정 백업
- **대시보드 메타데이터**: JSON 형태로 백업
- **위젯 설정**: 차트 설정, 필터 조건
- **사용자 커스터마이제이션**: 레이아웃, 테마 설정
- **데이터소스 연결 정보**: 암호화하여 백업

### 3. 파일 시스템 백업
- **업로드 파일**: 사용자가 업로드한 파일
- **로그 파일**: 애플리케이션 로그 (최근 30일)
- **설정 파일**: 환경변수, 설정 파일
- **템플릿**: 차트 템플릿, 대시보드 템플릿

### 4. 백업 압축 및 암호화
- **압축**: gzip, bzip2, lz4 알고리즘 지원
- **암호화**: AES-256-GCM 방식
- **무결성 검증**: SHA-256 체크섬
- **메타데이터**: 백업 정보를 JSON으로 저장

```typescript
// BackupExecutorService - 메인 실행 엔진
export class BackupExecutorService {
  private readonly databaseBackup: DatabaseBackupHandler;
  private readonly fileSystemBackup: FileSystemBackupHandler;
  private readonly configBackup: ConfigBackupHandler;
  private readonly compressionService: CompressionService;
  private readonly encryptionService: EncryptionService;
  private readonly s3Service: S3Service;

  async executeBackup(job: BackupJob): Promise<BackupResult> {
    const backupId = this.generateBackupId();
    const workDir = `/tmp/backup_${backupId}`;
    
    try {
      // 1. 작업 디렉토리 생성
      await this.createWorkDirectory(workDir);
      
      // 2. 백업 실행
      const results = await this.performBackup(job, workDir);
      
      // 3. 압축 및 암호화
      const compressedFile = await this.compressAndEncrypt(workDir, backupId);
      
      // 4. S3 업로드
      const s3Location = await this.uploadToS3(compressedFile, backupId);
      
      // 5. 메타데이터 저장
      await this.saveMetadata(backupId, job, results, s3Location);
      
      return { success: true, backupId, s3Location };
      
    } finally {
      // 6. 임시 파일 정리
      await this.cleanup(workDir);
    }
  }
}
```

```typescript
export class DatabaseBackupHandler {
  private readonly connectionService: DatabaseConnectionService;
  
  async backupDatabase(job: BackupJob, outputDir: string): Promise<DatabaseBackupResult> {
    const connection = await this.connectionService.getConnection();
    const backupFile = path.join(outputDir, 'database.sql');
    
    const mysqldumpOptions = this.buildMysqldumpOptions(job);
    
    try {
      // mysqldump 실행
      const result = await this.executeMysqldump(connection, mysqldumpOptions, backupFile);
      
      // 백업 파일 검증
      await this.validateBackupFile(backupFile);
      
      return {
        success: true,
        fileName: 'database.sql',
        fileSize: await this.getFileSize(backupFile),
        tableCount: result.tableCount,
        rowCount: result.rowCount
      };
      
    } catch (error) {
      throw new BackupError(`Database backup failed: ${error.message}`);
    }
  }
  
  private buildMysqldumpOptions(job: BackupJob): MysqldumpOptions {
    const options: MysqldumpOptions = {
      singleTransaction: true,
      routines: true,
      triggers: true,
      lockTables: false,
      addDropTable: true,
      extendedInsert: true,
      completeInsert: true
    };
    
    if (job.type === 'incremental') {
      options.where = `updated_at >= '${job.lastBackupTime}'`;
    }
    
    if (job.options?.tables) {
      options.tables = job.options.tables;
    }
    
    return options;
  }
  
  private async executeMysqldump(
    connection: DatabaseConnection, 
    options: MysqldumpOptions, 
    outputFile: string
  ): Promise<MysqldumpResult> {
    return new Promise((resolve, reject) => {
      const command = this.buildMysqldumpCommand(connection, options, outputFile);
      
      const process = spawn('mysqldump', command.args, {
        stdio: ['pipe', 'pipe', 'pipe']
      });
      
      let stderr = '';
      let tableCount = 0;
      let rowCount = 0;
      
      process.stderr.on('data', (data) => {
        stderr += data.toString();
      });
      
      process.stdout.on('data', (data) => {
        // 진행률 추적을 위한 로직
        const output = data.toString();
        if (output.includes('Dumping table')) {
          tableCount++;
        }
      });
      
      process.on('close', (code) => {
        if (code === 0) {
          resolve({ tableCount, rowCount, size: 0 });
        } else {
          reject(new Error(`mysqldump failed: ${stderr}`));
        }
      });
    });
  }
}
```

### 3. 파일 시스템 백업 핸들러

```typescript
export class FileSystemBackupHandler {
  private readonly logger = new Logger(FileSystemBackupHandler.name);
  
  async backupFiles(job: BackupJob, outputDir: string): Promise<FileBackupResult> {
    const results: FileBackupResult = {
      uploadFiles: { count: 0, size: 0 },
      logFiles: { count: 0, size: 0 },
      configFiles: { count: 0, size: 0 },
      templates: { count: 0, size: 0 }
    };
    
    // 1. 업로드 파일 백업
    if (job.includeUploads) {
      results.uploadFiles = await this.backupUploadFiles(outputDir);
    }
    
    // 2. 로그 파일 백업
    if (job.includeLogs) {
      results.logFiles = await this.backupLogFiles(outputDir, job.logRetentionDays);
    }
    
    // 3. 설정 파일 백업
    if (job.includeConfig) {
      results.configFiles = await this.backupConfigFiles(outputDir);
    }
    
    // 4. 템플릿 백업
    if (job.includeTemplates) {
      results.templates = await this.backupTemplates(outputDir);
    }
    
    return results;
  }
  
  private async backupUploadFiles(outputDir: string): Promise<BackupStats> {
    const uploadsDir = process.env.UPLOADS_DIR || '/app/uploads';
    const targetDir = path.join(outputDir, 'uploads');
    
    await fs.ensureDir(targetDir);
    
    return this.copyDirectoryWithStats(uploadsDir, targetDir, {
      filter: (src) => {
        // 임시 파일 제외
        return !path.basename(src).startsWith('tmp_');
      }
    });
  }
  
  private async backupLogFiles(outputDir: string, retentionDays: number): Promise<BackupStats> {
    const logsDir = process.env.LOGS_DIR || '/app/logs';
    const targetDir = path.join(outputDir, 'logs');
    const cutoffDate = new Date();
    cutoffDate.setDate(cutoffDate.getDate() - retentionDays);
    
    await fs.ensureDir(targetDir);
    
    return this.copyDirectoryWithStats(logsDir, targetDir, {
      filter: async (src) => {
        const stats = await fs.stat(src);
        return stats.mtime >= cutoffDate;
      }
    });
  }
  
  private async copyDirectoryWithStats(
    src: string, 
    dest: string, 
    options: CopyOptions
  ): Promise<BackupStats> {
    let count = 0;
    let size = 0;
    
    const copyWithCallback = async (srcPath: string, destPath: string) => {
      const stats = await fs.stat(srcPath);
      
      if (stats.isFile()) {
        if (options.filter && !(await options.filter(srcPath))) {
          return;
        }
        
        await fs.copy(srcPath, destPath);
        count++;
        size += stats.size;
        
        this.logger.debug(`Copied file: ${srcPath} (${stats.size} bytes)`);
      }
    };
    
    await fs.copy(src, dest, { 
      ...options,
      filter: copyWithCallback 
    });
    
    return { count, size };
  }
}
```

```typescript
export class ConfigBackupHandler {
  private readonly dashboardService: DashboardService;
  private readonly widgetService: WidgetService;
  private readonly datasourceService: DatasourceService;
  
  async backupConfigurations(job: BackupJob, outputDir: string): Promise<ConfigBackupResult> {
    const configDir = path.join(outputDir, 'config');
    await fs.ensureDir(configDir);
    
    const results: ConfigBackupResult = {
      dashboards: { count: 0, size: 0 },
      widgets: { count: 0, size: 0 },
      datasources: { count: 0, size: 0 },
      userSettings: { count: 0, size: 0 }
    };
    
    // 1. 대시보드 설정 백업
    results.dashboards = await this.backupDashboards(configDir, job.options?.dashboardIds);
    
    // 2. 위젯 설정 백업
    results.widgets = await this.backupWidgets(configDir, job.options?.widgetIds);
    
    // 3. 데이터소스 설정 백업
    results.datasources = await this.backupDatasources(configDir);
    
    // 4. 사용자 설정 백업
    results.userSettings = await this.backupUserSettings(configDir, job.options?.userIds);
    
    return results;
  }
  
  private async backupDashboards(configDir: string, dashboardIds?: string[]): Promise<BackupStats> {
    const dashboards = dashboardIds 
      ? await this.dashboardService.findByIds(dashboardIds)
      : await this.dashboardService.findAll();
    
    const dashboardsFile = path.join(configDir, 'dashboards.json');
    const dashboardsData = {
      version: '1.0',
      exportedAt: new Date().toISOString(),
      dashboards: dashboards.map(dashboard => ({
        ...dashboard,
        // 민감한 정보 제거
        createdBy: dashboard.createdBy ? 'USER_ID_MASKED' : null
      }))
    };
    
    await fs.writeJson(dashboardsFile, dashboardsData, { spaces: 2 });
    
    const stats = await fs.stat(dashboardsFile);
    return { count: dashboards.length, size: stats.size };
  }
  
  private async backupDatasources(configDir: string): Promise<BackupStats> {
    const datasources = await this.datasourceService.findAll();
    
    const datasourcesFile = path.join(configDir, 'datasources.json');
    const datasourcesData = {
      version: '1.0',
      exportedAt: new Date().toISOString(),
      datasources: await Promise.all(datasources.map(async ds => ({
        ...ds,
        // 암호화된 연결 정보
        connectionString: await this.encryptionService.encrypt(ds.connectionString),
        password: await this.encryptionService.encrypt(ds.password || '')
      })))
    };
    
    await fs.writeJson(datasourcesFile, datasourcesData, { spaces: 2 });
    
    const stats = await fs.stat(datasourcesFile);
    return { count: datasources.length, size: stats.size };
  }
}
```

### 5. 압축 및 암호화 서비스

```typescript
export class CompressionService {
  async compress(inputDir: string, outputFile: string, algorithm: CompressionAlgorithm): Promise<CompressionResult> {
    switch (algorithm) {
      case 'gzip':
        return this.compressWithGzip(inputDir, outputFile);
      case 'bzip2':
        return this.compressWithBzip2(inputDir, outputFile);
      case 'lz4':
        return this.compressWithLz4(inputDir, outputFile);
      default:
        throw new Error(`Unsupported compression algorithm: ${algorithm}`);
    }
  }
  
  private async compressWithGzip(inputDir: string, outputFile: string): Promise<CompressionResult> {
    return new Promise((resolve, reject) => {
      const archive = archiver('tar', {
        gzip: true,
        gzipOptions: { level: 6 }
      });
      
      const output = fs.createWriteStream(outputFile);
      let originalSize = 0;
      
      archive.pipe(output);
      
      archive.on('error', reject);
      archive.on('warning', (err) => {
        if (err.code === 'ENOENT') {
          console.warn(`Warning: ${err.message}`);
        } else {
          reject(err);
        }
      });
      
      archive.on('progress', (progress) => {
        originalSize = progress.entries.processed;
      });
      
      output.on('close', () => {
        const compressedSize = archive.pointer();
        const ratio = compressedSize / originalSize;
        
        resolve({
          originalSize,
          compressedSize,
          compressionRatio: ratio,
          algorithm: 'gzip'
        });
      });
      
      archive.directory(inputDir, false);
      archive.finalize();
    });
  }
}

export class EncryptionService {
  private readonly algorithm = 'aes-256-gcm';
  private readonly keyService: KeyManagementService;
  
  async encryptFile(inputFile: string, outputFile: string): Promise<EncryptionResult> {
    const key = await this.keyService.getBackupEncryptionKey();
    const iv = crypto.randomBytes(16);
    
    return new Promise((resolve, reject) => {
      const cipher = crypto.createCipher(this.algorithm, key, iv);
      const input = fs.createReadStream(inputFile);
      const output = fs.createWriteStream(outputFile);
      
      // IV를 파일 시작에 저장
      output.write(iv);
      
      input.pipe(cipher).pipe(output);
      
      output.on('finish', () => {
        const authTag = cipher.getAuthTag();
        
        resolve({
          success: true,
          algorithm: this.algorithm,
          keyId: key.id,
          iv: iv.toString('hex'),
          authTag: authTag.toString('hex')
        });
      });
      
      input.on('error', reject);
      cipher.on('error', reject);
      output.on('error', reject);
    });
  }
  
  async calculateChecksum(filePath: string): Promise<string> {
    return new Promise((resolve, reject) => {
      const hash = crypto.createHash('sha256');
      const stream = fs.createReadStream(filePath);
      
      stream.on('data', (data) => hash.update(data));
      stream.on('end', () => resolve(hash.digest('hex')));
      stream.on('error', reject);
    });
  }
}
```

```typescript
export class BackupProgressTracker {
  private readonly eventEmitter = new EventEmitter();
  private readonly progressCache = new Map<string, BackupProgress>();
  
  updateProgress(backupId: string, progress: Partial<BackupProgress>): void {
    const current = this.progressCache.get(backupId) || {
      backupId,
      stage: 'initializing',
      percentage: 0,
      startedAt: new Date(),
      estimatedCompletion: null,
      currentTask: '',
      errors: []
    };
    
    const updated = { ...current, ...progress };
    this.progressCache.set(backupId, updated);
    
    // 실시간 업데이트를 위한 이벤트 발송
    this.eventEmitter.emit('progress', updated);
    
    // WebSocket을 통한 클라이언트 알림
    this.notifyClients(updated);
  }
  
  getProgress(backupId: string): BackupProgress | null {
    return this.progressCache.get(backupId) || null;
  }
  
  private notifyClients(progress: BackupProgress): void {
    // WebSocket 구현
    this.webSocketService.broadcast('backup-progress', progress);
  }
}

interface BackupProgress {
  backupId: string;
  stage: 'initializing' | 'database' | 'files' | 'config' | 'compressing' | 'encrypting' | 'uploading' | 'completed' | 'failed';
  percentage: number;
  startedAt: Date;
  estimatedCompletion: Date | null;
  currentTask: string;
  errors: string[];
}
```

```typescript
export class ParallelBackupExecutor {
  async executeParallelBackup(job: BackupJob, workDir: string): Promise<BackupResult[]> {
    const tasks = [
      () => this.databaseBackup.backupDatabase(job, workDir),
      () => this.fileSystemBackup.backupFiles(job, workDir),
      () => this.configBackup.backupConfigurations(job, workDir)
    ];
    
    // 최대 3개 작업 병렬 실행
    const results = await Promise.allSettled(
      tasks.map(task => this.executeWithTimeout(task, 30 * 60 * 1000)) // 30분 타임아웃
    );
    
    return results.map((result, index) => {
      if (result.status === 'fulfilled') {
        return result.value;
      } else {
        throw new Error(`Task ${index} failed: ${result.reason}`);
      }
    });
  }
  
  private async executeWithTimeout<T>(task: () => Promise<T>, timeoutMs: number): Promise<T> {
    return Promise.race([
      task(),
      new Promise<never>((_, reject) => 
        setTimeout(() => reject(new Error('Task timeout')), timeoutMs)
      )
    ]);
  }
}
```

```typescript
export class StreamingBackupHandler {
  async streamBackupToS3(
    sourceStream: Readable, 
    s3Key: string,
    progressCallback?: (progress: number) => void
  ): Promise<string> {
    const upload = new AWS.S3.ManagedUpload({
      params: {
        Bucket: this.bucketName,
        Key: s3Key,
        Body: sourceStream
      }
    });
    
    // 진행률 추적
    upload.on('httpUploadProgress', (progress) => {
      const percentage = (progress.loaded / progress.total) * 100;
      progressCallback?.(percentage);
    });
    
    const result = await upload.promise();
    return result.Location;
  }
}
```

### 1. 재시도 로직
```typescript
export class RetryableBackupExecutor {
  async executeWithRetry<T>(
    operation: () => Promise<T>,
    maxRetries: number = 3,
    backoffMs: number = 1000
  ): Promise<T> {
    let lastError: Error;
    
    for (let attempt = 1; attempt <= maxRetries; attempt++) {
      try {
        return await operation();
      } catch (error) {
        lastError = error;
        
        if (attempt === maxRetries) {
          break;
        }
        
        // 지수 백오프
        const delay = backoffMs * Math.pow(2, attempt - 1);
        await this.sleep(delay);
        
        this.logger.warn(`Backup attempt ${attempt} failed, retrying in ${delay}ms: ${error.message}`);
      }
    }
    
    throw new Error(`Backup failed after ${maxRetries} attempts: ${lastError.message}`);
  }
  
  private sleep(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

```typescript
    backupId: string, 
    completedTasks: BackupTask[], 
    failedTasks: BackupTask[]
  ): Promise<void> {
    // 성공한 부분은 보존
```
