### 2.2 수동 전체 백업 실행
```bash
#!/bin/bash
# manual-full-backup.sh

echo "Starting manual full backup..."

# 1. 백업 작업 생성
kubectl create job --from=cronjob/vanillameta-full-backup manual-backup-$(date +%Y%m%d-%H%M%S) -n vanillameta

# 2. 백업 진행 상황 모니터링
JOB_NAME=$(kubectl get jobs -n vanillameta --sort-by=.metadata.creationTimestamp | tail -1 | awk '{print $1}')
echo "Monitoring job: $JOB_NAME"

while true; do
    STATUS=$(kubectl get job $JOB_NAME -n vanillameta -o jsonpath='{.status.conditions[0].type}')
    if [ "$STATUS" = "Complete" ]; then
        echo "Backup completed successfully"
        break
    elif [ "$STATUS" = "Failed" ]; then
        echo "Backup failed"
        kubectl logs job/$JOB_NAME -n vanillameta
        exit 1
    fi
    sleep 30
done

# 3. 백업 결과 확인
kubectl logs job/$JOB_NAME -n vanillameta | tail -20
```

### 2.3 백업 검증
```bash
#!/bin/bash
# verify-backup.sh

BACKUP_ID=$1
if [ -z "$BACKUP_ID" ]; then
    echo "Usage: $0 <backup_id>"
    exit 1
fi

echo "Verifying backup: $BACKUP_ID"

# 1. 백업 메타데이터 조회
curl -X GET "http://admin-api.vanillameta.com/api/admin/backup/$BACKUP_ID" \
     -H "Authorization: Bearer $ADMIN_TOKEN" | jq .

# 2. 체크섬 검증
echo "Verifying checksum..."
EXPECTED_CHECKSUM=$(curl -s "http://admin-api.vanillameta.com/api/admin/backup/$BACKUP_ID" \
                   -H "Authorization: Bearer $ADMIN_TOKEN" | jq -r '.checksum')

aws s3 cp s3://vanillameta-backups/full/$(date +%Y/%m/%d)/backup_${BACKUP_ID}.tar.gz.enc /tmp/
ACTUAL_CHECKSUM=$(sha256sum /tmp/backup_${BACKUP_ID}.tar.gz.enc | awk '{print $1}')

if [ "$EXPECTED_CHECKSUM" = "$ACTUAL_CHECKSUM" ]; then
    echo "✓ Checksum verification passed"
else
    echo "✗ Checksum verification failed"
    exit 1
fi

# 3. 압축 파일 무결성 확인
echo "Verifying archive integrity..."
gpg --decrypt /tmp/backup_${BACKUP_ID}.tar.gz.enc | tar -tzf - > /dev/null
if [ $? -eq 0 ]; then
    echo "✓ Archive integrity verified"
else
    echo "✗ Archive integrity check failed"
    exit 1
fi

echo "Backup verification completed successfully"
```

## 3. 복구 절차

### 3.1 전체 시스템 복구
```bash
#!/bin/bash
# full-system-restore.sh

BACKUP_ID=$1
if [ -z "$BACKUP_ID" ]; then
    echo "Usage: $0 <backup_id>"
    exit 1
fi

echo "WARNING: This will restore the entire system from backup $BACKUP_ID"
echo "Current data will be overwritten. Continue? (yes/no)"
read -r CONFIRM

if [ "$CONFIRM" != "yes" ]; then
    echo "Recovery cancelled"
    exit 0
fi

# 1. 현재 상태 스냅샷 생성
echo "Creating current state snapshot..."
kubectl create job --from=cronjob/vanillameta-full-backup pre-restore-snapshot-$(date +%Y%m%d-%H%M%S) -n vanillameta

# 2. 서비스 중지
echo "Stopping services..."
kubectl scale deployment vanillameta-backend --replicas=0 -n vanillameta
kubectl scale deployment vanillameta-frontend --replicas=0 -n vanillameta

# 3. 복구 작업 실행
echo "Starting restore process..."
kubectl run restore-job --image=vanillameta/restore:latest \
    --env="BACKUP_ID=$BACKUP_ID" \
    --env="RESTORE_TYPE=full" \
    -n vanillameta

# 4. 복구 진행 상황 모니터링
POD_NAME=$(kubectl get pods -n vanillameta -l run=restore-job -o jsonpath='{.items[0].metadata.name}')
kubectl logs -f $POD_NAME -n vanillameta

# 5. 복구 완료 후 서비스 재시작
echo "Restore completed. Restarting services..."
kubectl scale deployment vanillameta-backend --replicas=3 -n vanillameta
kubectl scale deployment vanillameta-frontend --replicas=2 -n vanillameta

# 6. 서비스 상태 확인
echo "Waiting for services to be ready..."
kubectl wait --for=condition=available --timeout=300s deployment/vanillameta-backend -n vanillameta
kubectl wait --for=condition=available --timeout=300s deployment/vanillameta-frontend -n vanillameta

echo "System restore completed successfully"
```

# health-check.sh - 시스템 헬스체크 스크립트
#!/bin/bash
health_check() {
    echo "=== VanillaMeta 헬스체크 $(date) ==="
    
    # 웹사이트 접근성 확인
    if curl -s --max-time 10 https://app.vanillameta.com/health | grep -q "OK"; then
        echo "✓ 웹사이트 접근 가능"
    else
        echo "✗ 웹사이트 접근 불가"
        send_alert "Website unreachable"
    fi
    
    # API 응답 확인
    response_time=$(curl -w "%{time_total}" -s -o /dev/null https://app.vanillameta.com/api/health)
    if (( $(echo "$response_time < 5.0" | bc -l) )); then
        echo "✓ API 응답 시간: ${response_time}s"
    else
        echo "⚠ API 응답 느림: ${response_time}s"
        send_alert "Slow API response: ${response_time}s"
    fi
    
    # 데이터베이스 연결 확인
    if mysql -h $DB_HOST -u $DB_USER -p$DB_PASSWORD -e "SELECT 1" 2>/dev/null; then
        echo "✓ 데이터베이스 연결 정상"
    else
        echo "✗ 데이터베이스 연결 실패"
        send_alert "Database connection failed"
    fi
    
    # 디스크 공간 확인
    disk_usage=$(df -h / | awk 'NR==2 {print $5}' | sed 's/%//')
    if [ "$disk_usage" -lt 80 ]; then
        echo "✓ 디스크 사용률: ${disk_usage}%"
    else
        echo "⚠ 디스크 사용률 높음: ${disk_usage}%"
        send_alert "High disk usage: ${disk_usage}%"
    fi
    
    # 메모리 사용률 확인
    memory_usage=$(free | grep Mem | awk '{printf "%.1f", $3/$2 * 100.0}')
    if (( $(echo "$memory_usage < 80.0" | bc -l) )); then
        echo "✓ 메모리 사용률: ${memory_usage}%"
    else
        echo "⚠ 메모리 사용률 높음: ${memory_usage}%"
        send_alert "High memory usage: ${memory_usage}%"
    fi
}

#!/bin/bash
performance_monitoring() {
    echo "=== 성능 모니터링 $(date) ==="
    
    # API 응답 시간 측정
    echo "API 응답 시간 측정 중..."
    for endpoint in "/api/health" "/api/dashboards" "/api/users/me"; do
        response_time=$(curl -w "%{time_total}" -s -o /dev/null \
            -H "Authorization: Bearer $TEST_TOKEN" \
            "https://app.vanillameta.com$endpoint")
        echo "$endpoint: ${response_time}s"
    done
    
    # 데이터베이스 성능 확인
    echo "데이터베이스 성능 확인..."
    mysql -h $DB_HOST -u $DB_USER -p$DB_PASSWORD -e "
        SELECT 
            COUNT(*) as active_connections,
            AVG(TIME) as avg_query_time
        FROM INFORMATION_SCHEMA.PROCESSLIST 
        WHERE COMMAND != 'Sleep';"
    
    # 느린 쿼리 확인
    slow_queries=$(mysql -h $DB_HOST -u $DB_USER -p$DB_PASSWORD -e "
        SELECT COUNT(*) 
        FROM mysql.slow_log 
        WHERE start_time > NOW() - INTERVAL 1 HOUR;" 2>/dev/null | tail -1)
    
    if [ "$slow_queries" -gt 10 ]; then
        echo "⚠ 최근 1시간 느린 쿼리: $slow_queries개"
        send_alert "High number of slow queries: $slow_queries"
    fi
    
    # Redis 성능 확인
    redis_info=$(redis-cli -h $REDIS_HOST info stats)
    hit_rate=$(echo "$redis_info" | grep "keyspace_hit" | cut -d: -f2)
    echo "Redis 히트율: $hit_rate"
}

#!/bin/bash
log_analysis() {
    echo "=== 로그 분석 $(date) ==="
    
    # 에러 로그 분석
    echo "최근 1시간 에러 로그 분석..."
    kubectl logs --since=1h deployment/vanillameta-backend -n vanillameta | \
        grep -i error | \
        sort | uniq -c | sort -nr | head -10
    
    # 404 에러 분석
    echo "404 에러 분석..."
    kubectl logs --since=1h deployment/vanillameta-backend -n vanillameta | \
        grep "404" | \
        awk '{print $7}' | \
        sort | uniq -c | sort -nr | head -10
    
    # 사용자 활동 분석
    echo "활성 사용자 수..."
    kubectl logs --since=1h deployment/vanillameta-backend -n vanillameta | \
        grep "user_id" | \
        awk -F'user_id":"' '{print $2}' | \
        awk -F'"' '{print $1}' | \
        sort | uniq | wc -l
    
    # 보안 이벤트 확인
    echo "보안 이벤트 확인..."
    security_events=$(kubectl logs --since=1h deployment/vanillameta-backend -n vanillameta | \
        grep -E "(authentication failed|unauthorized|suspicious)" | wc -l)
    
    if [ "$security_events" -gt 5 ]; then
        echo "⚠ 보안 이벤트 감지: $security_events건"
        send_alert "Security events detected: $security_events"
    fi
}

#!/bin/bash
cleanup_old_data() {

    # 30일 이상 된 감사 로그 아카이브

    logrotate /etc/logrotate.d/vanillameta

    echo "사용하지 않는 Docker 이미지 정리..."
    docker image prune -a -f --filter "until=168h"
    
    # 백업 파일 정리 (90일 이상)
    echo "오래된 백업 파일 정리..."
    aws s3 ls s3://vanillameta-backups/ --recursive | \
        awk '$1 <= "'$(date -d '90 days ago' '+%Y-%m-%d')'" {print $4}' | \
        xargs -I {} aws s3 rm s3://vanillameta-backups/{}
}

# auto-scaling.sh - 자동 스케일링 스크립트
#!/bin/bash
auto_scaling() {
    echo "=== 자동 스케일링 체크 $(date) ==="
    
    # CPU 사용률 확인
    cpu_usage=$(kubectl top pods -n vanillameta --no-headers | \
        grep vanillameta-backend | \
        awk '{print $3}' | \
        sed 's/m//' | \
        awk '{sum+=$1} END {print sum/NR/1000}')
    
    current_replicas=$(kubectl get deployment vanillameta-backend -n vanillameta -o jsonpath='{.status.replicas}')
    
    if (( $(echo "$cpu_usage > 70" | bc -l) )); then
        new_replicas=$((current_replicas + 2))
        if [ "$new_replicas" -le 10 ]; then
            echo "CPU 사용률 높음 ($cpu_usage%). 스케일 업: $current_replicas -> $new_replicas"
            kubectl scale deployment vanillameta-backend --replicas=$new_replicas -n vanillameta
        fi
    elif (( $(echo "$cpu_usage < 30" | bc -l) )); then
        new_replicas=$((current_replicas - 1))
        if [ "$new_replicas" -ge 2 ]; then
            echo "CPU 사용률 낮음 ($cpu_usage%). 스케일 다운: $current_replicas -> $new_replicas"
            kubectl scale deployment vanillameta-backend --replicas=$new_replicas -n vanillameta
        fi
    fi
}

    # 이메일 알림 (선택적)
    if [ "$SEND_EMAIL" = "true" ]; then
        echo "$message" | mail -s "VanillaMeta Alert" $ALERT_EMAIL
    fi
}
