## 태스크 개요
- **Sprint**: S10 - Admin 운영 기능
- **Priority**: High
- **Estimated Time**: 4일
- **Dependencies**: T01_S10 (백업 시스템 설계), T02_S10 (백업 실행 엔진), T03_S10 (백업 관리 UI)

## 목표
백업된 데이터로부터 시스템을 복구할 수 있는 종합적인 복구 시스템을 구현합니다. 복구 포인트 선택, 선택적 복구, 복구 전 검증, 복구 롤백 기능을 포함합니다.

## 상세 요구사항

### 1. 복구 포인트 선택
- **백업 목록**: 복구 가능한 백업 목록 표시
- **복구 유형**: 전체/선택적/포인트 타임 복구
- **영향도 분석**: 복구 시 영향받는 데이터 범위 표시
- **의존성 검증**: 복구 가능성 사전 검증

### 2. 선택적 복구 (특정 대시보드)
- **대시보드 선택**: 복구할 특정 대시보드 선택
- **관련 데이터**: 위젯, 데이터소스, 사용자 설정 포함
- **충돌 해결**: 기존 데이터와의 충돌 처리
- **미리보기**: 복구 후 예상 결과 미리보기

### 3. 복구 전 검증
- **백업 무결성**: 체크섬 및 파일 무결성 검증
- **환경 호환성**: 복구 대상 환경과의 호환성 확인
- **리소스 용량**: 필요한 디스크/메모리 용량 확인
- **시뮬레이션**: 복구 시뮬레이션 실행

### 4. 복구 롤백 기능
- **복구 전 스냅샷**: 복구 전 현재 상태 백업
- **단계별 롤백**: 복구 과정의 각 단계별 롤백
- **자동 롤백**: 복구 실패 시 자동 롤백
- **수동 롤백**: 관리자 판단에 의한 수동 롤백

```typescript
// RecoveryService - 메인 복구 서비스
export class RecoveryService {
  private readonly backupService: BackupService;
  private readonly databaseRecovery: DatabaseRecoveryHandler;
  private readonly fileRecovery: FileRecoveryHandler;
  private readonly configRecovery: ConfigRecoveryHandler;
  private readonly validationService: RecoveryValidationService;
  private readonly rollbackService: RollbackService;

  async executeRecovery(request: RecoveryRequest): Promise<RecoveryResult> {
    const recoveryId = this.generateRecoveryId();
    const workDir = `/tmp/recovery_${recoveryId}`;
    
    try {
      // 1. 복구 전 검증
      await this.validateRecovery(request);
      
      // 2. 현재 상태 스냅샷 (롤백용)
      const snapshotId = await this.createSnapshot();
      
      // 3. 백업 파일 준비
      const backupData = await this.prepareBackupData(request.backupId, workDir);
      
      // 4. 복구 실행
      const result = await this.performRecovery(request, backupData, workDir);
      
      // 5. 복구 후 검증
      await this.validateRecoveryResult(result);
      
      return {
        success: true,
        recoveryId,
        snapshotId,
        restoredItems: result.items,
        duration: result.duration
      };
      
    } catch (error) {
      // 자동 롤백
      if (snapshotId) {
        await this.rollbackService.rollback(snapshotId);
      }
      throw error;
    } finally {
      // 임시 파일 정리
      await this.cleanup(workDir);
    }
  }
}
```

```typescript
export class DatabaseRecoveryHandler {
  private readonly connectionService: DatabaseConnectionService;
  private readonly backupService: BackupService;
  
  async recoverDatabase(request: DatabaseRecoveryRequest): Promise<DatabaseRecoveryResult> {
    const connection = await this.connectionService.getConnection();
    
    try {
      // 1. 복구 전 현재 스키마 백업
      const currentSchema = await this.backupCurrentSchema();
      
      // 2. 복구 방식에 따른 처리
      switch (request.type) {
        case 'full':
          return await this.performFullRestore(request, connection);
        case 'selective':
          return await this.performSelectiveRestore(request, connection);
        case 'point-in-time':
          return await this.performPointInTimeRestore(request, connection);
        default:
          throw new Error(`Unsupported recovery type: ${request.type}`);
      }
      
    } catch (error) {
      // 복구 실패 시 원래 스키마로 롤백
      await this.restoreSchema(currentSchema);
      throw error;
    }
  }
  
  private async performFullRestore(
    request: DatabaseRecoveryRequest, 
    connection: DatabaseConnection
  ): Promise<DatabaseRecoveryResult> {
    const sqlFile = request.backupPath + '/database.sql';
    
    // 1. 기존 데이터 백업 (롤백용)
    const rollbackFile = await this.createRollbackDump(connection);
    
    // 2. 데이터베이스 초기화
    if (request.dropExisting) {
      await this.dropAllTables(connection);
    }
    
    // 3. SQL 파일 실행
    const result = await this.executeSqlFile(connection, sqlFile);
    
    // 4. 결과 검증
    await this.validateRestore(connection, request.expectedTables);
    
    return {
      success: true,
      restoredTables: result.tables,
      restoredRows: result.rows,
      rollbackFile,
      duration: result.duration
    };
  }
  
  private async performSelectiveRestore(
    request: DatabaseRecoveryRequest,
    connection: DatabaseConnection
  ): Promise<DatabaseRecoveryResult> {
    const tables = request.selectedTables || [];
    const restoredTables: string[] = [];
    const rollbackData: Map<string, string> = new Map();
    
    for (const table of tables) {
      try {
        // 1. 기존 테이블 백업
        const tableBackup = await this.backupTable(connection, table);
        rollbackData.set(table, tableBackup);
        
        // 2. 테이블 복구
        await this.restoreTable(connection, request.backupPath, table);
        restoredTables.push(table);
        
        this.logger.info(`Table restored: ${table}`);
        
      } catch (error) {
        this.logger.error(`Failed to restore table ${table}: ${error.message}`);
        
        // 부분 롤백
        if (rollbackData.has(table)) {
          await this.restoreTableFromBackup(connection, table, rollbackData.get(table));
        }
        
        throw error;
      }
    }
    
    return {
      success: true,
      restoredTables,
      restoredRows: await this.countRows(connection, restoredTables),
      rollbackData: Array.from(rollbackData.entries()),
      duration: 0 // 계산 필요
    };
  }
  
  private async executeSqlFile(
    connection: DatabaseConnection, 
    sqlFile: string
  ): Promise<SqlExecutionResult> {
    return new Promise((resolve, reject) => {
      const mysql = spawn('mysql', [
        '-h', connection.host,
        '-P', connection.port.toString(),
        '-u', connection.username,
        '-p' + connection.password,
        connection.database
      ]);
      
      const fileStream = fs.createReadStream(sqlFile);
      fileStream.pipe(mysql.stdin);
      
      let stderr = '';
      let tableCount = 0;
      let rowCount = 0;
      
      mysql.stderr.on('data', (data) => {
        stderr += data.toString();
      });
      
      mysql.stdout.on('data', (data) => {
        const output = data.toString();
        // 진행률 파싱
        if (output.includes('CREATE TABLE')) {
          tableCount++;
        }
        if (output.includes('INSERT INTO')) {
          rowCount++;
        }
      });
      
      mysql.on('close', (code) => {
        if (code === 0) {
          resolve({ 
            tables: tableCount, 
            rows: rowCount, 
            duration: 0 
          });
        } else {
          reject(new Error(`MySQL restore failed: ${stderr}`));
        }
      });
    });
  }
}
```

```typescript
export class FileRecoveryHandler {
  private readonly logger = new Logger(FileRecoveryHandler.name);
  
  async recoverFiles(request: FileRecoveryRequest): Promise<FileRecoveryResult> {
    const results: FileRecoveryResult = {
      uploadFiles: { restored: 0, failed: 0 },
      configFiles: { restored: 0, failed: 0 },
      logFiles: { restored: 0, failed: 0 },
      templates: { restored: 0, failed: 0 }
    };
    
    try {
      // 1. 업로드 파일 복구
      if (request.includeUploads) {
        results.uploadFiles = await this.recoverUploadFiles(request);
      }
      
      // 2. 설정 파일 복구
      if (request.includeConfig) {
        results.configFiles = await this.recoverConfigFiles(request);
      }
      
      // 3. 로그 파일 복구
      if (request.includeLogs) {
        results.logFiles = await this.recoverLogFiles(request);
      }
      
      // 4. 템플릿 복구
      if (request.includeTemplates) {
        results.templates = await this.recoverTemplates(request);
      }
      
      return results;
      
    } catch (error) {
      this.logger.error(`File recovery failed: ${error.message}`);
      throw error;
    }
  }
  
  private async recoverUploadFiles(request: FileRecoveryRequest): Promise<RecoveryStats> {
    const sourceDir = path.join(request.backupPath, 'uploads');
    const targetDir = process.env.UPLOADS_DIR || '/app/uploads';
    
    return this.recoverDirectory(sourceDir, targetDir, {
      overwrite: request.overwriteExisting,
      backup: true, // 기존 파일 백업
      filter: request.fileFilters?.uploads
    });
  }
  
  private async recoverDirectory(
    sourceDir: string,
    targetDir: string,
    options: RecoveryOptions
  ): Promise<RecoveryStats> {
    let restored = 0;
    let failed = 0;
    const rollbackData: Array<{ action: string; path: string; backup?: string }> = [];
    
    try {
      const files = await this.getAllFiles(sourceDir);
      
      for (const file of files) {
        try {
          const relativePath = path.relative(sourceDir, file.path);
          const targetPath = path.join(targetDir, relativePath);
          
          // 필터 적용
          if (options.filter && !options.filter(relativePath)) {
            continue;
          }
          
          // 기존 파일 처리
          if (await fs.pathExists(targetPath)) {
            if (!options.overwrite) {
              this.logger.warn(`File exists, skipping: ${targetPath}`);
              continue;
            }
            
            // 백업 생성
            if (options.backup) {
              const backupPath = `${targetPath}.backup.${Date.now()}`;
              await fs.copy(targetPath, backupPath);
              rollbackData.push({
                action: 'restore',
                path: targetPath,
                backup: backupPath
              });
            }
          } else {
            rollbackData.push({
              action: 'delete',
              path: targetPath
            });
          }
          
          // 파일 복사
          await fs.ensureDir(path.dirname(targetPath));
          await fs.copy(file.path, targetPath);
          
          restored++;
          this.logger.debug(`File restored: ${targetPath}`);
          
        } catch (error) {
          failed++;
          this.logger.error(`Failed to restore file ${file.path}: ${error.message}`);
        }
      }
      
      // 롤백 정보 저장
      await this.saveRollbackData(rollbackData);
      
      return { restored, failed };
      
    } catch (error) {
      // 복구 실패 시 롤백
      await this.rollbackFileChanges(rollbackData);
      throw error;
    }
  }
  
  private async getAllFiles(dir: string): Promise<Array<{ path: string; stats: fs.Stats }>> {
    const files: Array<{ path: string; stats: fs.Stats }> = [];
    
    const traverse = async (currentDir: string) => {
      const items = await fs.readdir(currentDir);
      
      for (const item of items) {
        const itemPath = path.join(currentDir, item);
        const stats = await fs.stat(itemPath);
        
        if (stats.isDirectory()) {
          await traverse(itemPath);
        } else {
          files.push({ path: itemPath, stats });
        }
      }
    };
    
    await traverse(dir);
    return files;
  }
}
```

### 4. 설정 복구 핸들러

```typescript
export class ConfigRecoveryHandler {
  private readonly dashboardService: DashboardService;
  private readonly widgetService: WidgetService;
  private readonly datasourceService: DatasourceService;
  private readonly encryptionService: EncryptionService;
  
  async recoverConfigurations(request: ConfigRecoveryRequest): Promise<ConfigRecoveryResult> {
    const configPath = path.join(request.backupPath, 'config');
    const results: ConfigRecoveryResult = {
      dashboards: { restored: 0, failed: 0, conflicts: [] },
      widgets: { restored: 0, failed: 0, conflicts: [] },
      datasources: { restored: 0, failed: 0, conflicts: [] },
      userSettings: { restored: 0, failed: 0, conflicts: [] }
    };
    
    try {
      // 1. 대시보드 복구
      if (request.includeDashboards) {
        results.dashboards = await this.recoverDashboards(configPath, request);
      }
      
      // 2. 위젯 복구
      if (request.includeWidgets) {
        results.widgets = await this.recoverWidgets(configPath, request);
      }
      
      // 3. 데이터소스 복구
      if (request.includeDatasources) {
        results.datasources = await this.recoverDatasources(configPath, request);
      }
      
      // 4. 사용자 설정 복구
      if (request.includeUserSettings) {
        results.userSettings = await this.recoverUserSettings(configPath, request);
      }
      
      return results;
      
    } catch (error) {
      this.logger.error(`Configuration recovery failed: ${error.message}`);
      throw error;
    }
  }
  
  private async recoverDashboards(
    configPath: string, 
    request: ConfigRecoveryRequest
  ): Promise<ConfigRecoveryStats> {
    const dashboardsFile = path.join(configPath, 'dashboards.json');
    
    if (!await fs.pathExists(dashboardsFile)) {
      throw new Error('Dashboards backup file not found');
    }
    
    const backupData = await fs.readJson(dashboardsFile);
    const dashboards = backupData.dashboards;
    
    let restored = 0;
    let failed = 0;
    const conflicts: Array<{ id: string; reason: string; resolution: string }> = [];
    
    for (const dashboard of dashboards) {
      try {
        // 선택적 복구 필터 적용
        if (request.selectedDashboards && 
            !request.selectedDashboards.includes(dashboard.id)) {
          continue;
        }
        
        // 기존 대시보드 확인
        const existing = await this.dashboardService.findById(dashboard.id);
        
        if (existing) {
          // 충돌 처리
          const resolution = await this.resolveDashboardConflict(
            existing, 
            dashboard, 
            request.conflictResolution
          );
          
          if (resolution.action === 'skip') {
            conflicts.push({
              id: dashboard.id,
              reason: 'Dashboard already exists',
              resolution: 'skipped'
            });
            continue;
          } else if (resolution.action === 'merge') {
            dashboard = resolution.merged;
          }
        }
        
        // 대시보드 복구
        await this.dashboardService.create(dashboard);
        restored++;
        
        this.logger.info(`Dashboard restored: ${dashboard.id}`);
        
      } catch (error) {
        failed++;
        this.logger.error(`Failed to restore dashboard ${dashboard.id}: ${error.message}`);
      }
    }
    
    return { restored, failed, conflicts };
  }
  
  private async recoverDatasources(
    configPath: string,
    request: ConfigRecoveryRequest
  ): Promise<ConfigRecoveryStats> {
    const datasourcesFile = path.join(configPath, 'datasources.json');
    
    if (!await fs.pathExists(datasourcesFile)) {
      throw new Error('Datasources backup file not found');
    }
    
    const backupData = await fs.readJson(datasourcesFile);
    const datasources = backupData.datasources;
    
    let restored = 0;
    let failed = 0;
    const conflicts: Array<{ id: string; reason: string; resolution: string }> = [];
    
    for (const datasource of datasources) {
      try {
        // 암호화된 정보 복호화
        if (datasource.connectionString) {
          datasource.connectionString = await this.encryptionService.decrypt(
            datasource.connectionString
          );
        }
        
        if (datasource.password) {
          datasource.password = await this.encryptionService.decrypt(
            datasource.password
          );
        }
        
        // 기존 데이터소스 확인
        const existing = await this.datasourceService.findById(datasource.id);
        
        if (existing && request.conflictResolution === 'skip') {
          conflicts.push({
            id: datasource.id,
            reason: 'Datasource already exists',
            resolution: 'skipped'
          });
          continue;
        }
        
        // 연결 테스트
        if (request.testConnections) {
          await this.datasourceService.testConnection(datasource);
        }
        
        // 데이터소스 복구
        if (existing) {
          await this.datasourceService.update(datasource.id, datasource);
        } else {
          await this.datasourceService.create(datasource);
        }
        
        restored++;
        this.logger.info(`Datasource restored: ${datasource.id}`);
        
      } catch (error) {
        failed++;
        this.logger.error(`Failed to restore datasource ${datasource.id}: ${error.message}`);
      }
    }
    
    return { restored, failed, conflicts };
  }
  
  private async resolveDashboardConflict(
    existing: Dashboard,
    backup: Dashboard,
    strategy: ConflictResolution
  ): Promise<{ action: 'skip' | 'overwrite' | 'merge'; merged?: Dashboard }> {
    switch (strategy) {
      case 'skip':
        return { action: 'skip' };
        
      case 'overwrite':
        return { action: 'overwrite' };
        
      case 'merge':
        // 스마트 머지: 최신 수정일을 기준으로 필드별 머지
        const merged = {
          ...existing,
          title: backup.updatedAt > existing.updatedAt ? backup.title : existing.title,
          description: backup.description || existing.description,
          layout: this.mergeDashboardLayout(existing.layout, backup.layout),
          widgets: this.mergeDashboardWidgets(existing.widgets, backup.widgets),
          updatedAt: new Date()
        };
        
        return { action: 'merge', merged };
        
      default:
        return { action: 'skip' };
    }
  }
}
```

### 5. 복구 검증 서비스

```typescript
export class RecoveryValidationService {
  async validateRecovery(request: RecoveryRequest): Promise<ValidationResult> {
    const validations: ValidationCheck[] = [];
    
    // 1. 백업 파일 무결성 검증
    validations.push(await this.validateBackupIntegrity(request.backupId));
    
    // 2. 환경 호환성 검증
    validations.push(await this.validateEnvironmentCompatibility(request));
    
    // 3. 리소스 용량 검증
    validations.push(await this.validateResourceCapacity(request));
    
    // 4. 의존성 검증
    validations.push(await this.validateDependencies(request));
    
    const failed = validations.filter(v => !v.passed);
    
    return {
      passed: failed.length === 0,
      checks: validations,
      errors: failed.map(f => f.error),
      warnings: validations.filter(v => v.warning).map(v => v.warning)
    };
  }
  
  private async validateBackupIntegrity(backupId: string): Promise<ValidationCheck> {
    try {
      const metadata = await this.backupService.getMetadata(backupId);
      const backupFile = await this.s3Service.downloadFile(metadata.s3Location);
      
      // 체크섬 검증
      const calculatedChecksum = await this.calculateChecksum(backupFile);
      
      if (calculatedChecksum !== metadata.checksum) {
        return {
          name: 'backup_integrity',
          passed: false,
          error: 'Backup file checksum mismatch - file may be corrupted'
        };
      }
      
      // 압축 파일 구조 검증
      const structure = await this.validateArchiveStructure(backupFile);
      
      if (!structure.valid) {
        return {
          name: 'backup_integrity',
          passed: false,
          error: `Invalid backup structure: ${structure.error}`
        };
      }
      
      return {
        name: 'backup_integrity',
        passed: true,
        message: 'Backup file integrity verified'
      };
      
    } catch (error) {
      return {
        name: 'backup_integrity',
        passed: false,
        error: `Failed to validate backup integrity: ${error.message}`
      };
    }
  }
  
  private async validateEnvironmentCompatibility(request: RecoveryRequest): Promise<ValidationCheck> {
    try {
      const metadata = await this.backupService.getMetadata(request.backupId);
      const currentEnv = await this.getCurrentEnvironmentInfo();
      
      const warnings: string[] = [];
      
      // 데이터베이스 버전 호환성
      if (metadata.metadata.databaseVersion !== currentEnv.databaseVersion) {
        warnings.push(
          `Database version mismatch: backup (${metadata.metadata.databaseVersion}) vs current (${currentEnv.databaseVersion})`
        );
      }
      
      // 애플리케이션 버전 호환성
      if (metadata.metadata.appVersion !== currentEnv.appVersion) {
        warnings.push(
          `Application version mismatch: backup (${metadata.metadata.appVersion}) vs current (${currentEnv.appVersion})`
        );
      }
      
      return {
        name: 'environment_compatibility',
        passed: true,
        message: 'Environment compatibility checked',
        warning: warnings.length > 0 ? warnings.join('; ') : undefined
      };
      
    } catch (error) {
      return {
        name: 'environment_compatibility',
        passed: false,
        error: `Failed to validate environment compatibility: ${error.message}`
      };
    }
  }
  
  private async validateResourceCapacity(request: RecoveryRequest): Promise<ValidationCheck> {
    try {
      const metadata = await this.backupService.getMetadata(request.backupId);
      const systemInfo = await this.getSystemResourceInfo();
      
      // 디스크 공간 확인 (압축 해제 후 크기 고려)
      const requiredSpace = metadata.fileSize * 3; // 압축 해제 + 여유 공간
      
      if (systemInfo.availableDisk < requiredSpace) {
        return {
          name: 'resource_capacity',
          passed: false,
          error: `Insufficient disk space: required ${formatFileSize(requiredSpace)}, available ${formatFileSize(systemInfo.availableDisk)}`
        };
      }
      
      // 메모리 확인 (대용량 복구 시)
      if (metadata.fileSize > 1024 * 1024 * 1024 && systemInfo.availableMemory < 512 * 1024 * 1024) {
        return {
          name: 'resource_capacity',
          passed: true,
          warning: 'Large backup detected but limited memory available - recovery may be slow'
        };
      }
      
      return {
        name: 'resource_capacity',
        passed: true,
        message: 'Sufficient resources available'
      };
      
    } catch (error) {
      return {
        name: 'resource_capacity',
        passed: false,
        error: `Failed to validate resource capacity: ${error.message}`
      };
    }
  }
}
```

### 6. 롤백 서비스

```typescript
export class RollbackService {
  private readonly snapshotService: SnapshotService;
  private readonly logger = new Logger(RollbackService.name);
  
  async createSnapshot(): Promise<string> {
    const snapshotId = this.generateSnapshotId();
    
    try {
      // 1. 데이터베이스 스냅샷
      const dbSnapshot = await this.createDatabaseSnapshot();
      
      // 2. 파일 시스템 스냅샷
      const fileSnapshot = await this.createFileSystemSnapshot();
      
      // 3. 설정 스냅샷
      const configSnapshot = await this.createConfigSnapshot();
      
      // 4. 스냅샷 메타데이터 저장
      await this.snapshotService.saveSnapshot(snapshotId, {
        databaseSnapshot: dbSnapshot,
        fileSnapshot: fileSnapshot,
        configSnapshot: configSnapshot,
        createdAt: new Date()
      });
      
      return snapshotId;
      
    } catch (error) {
      this.logger.error(`Failed to create snapshot: ${error.message}`);
      throw error;
    }
  }
  
  async rollback(snapshotId: string): Promise<void> {
    try {
      const snapshot = await this.snapshotService.getSnapshot(snapshotId);
      
      if (!snapshot) {
        throw new Error(`Snapshot not found: ${snapshotId}`);
      }
      
      this.logger.info(`Starting rollback to snapshot: ${snapshotId}`);
      
      // 1. 데이터베이스 롤백
      if (snapshot.databaseSnapshot) {
        await this.rollbackDatabase(snapshot.databaseSnapshot);
      }
      
      // 2. 파일 시스템 롤백
      if (snapshot.fileSnapshot) {
        await this.rollbackFileSystem(snapshot.fileSnapshot);
      }
      
      // 3. 설정 롤백
      if (snapshot.configSnapshot) {
        await this.rollbackConfig(snapshot.configSnapshot);
      }
      
      this.logger.info(`Rollback completed: ${snapshotId}`);
      
    } catch (error) {
      this.logger.error(`Rollback failed: ${error.message}`);
      throw error;
    }
  }
  
  private async createDatabaseSnapshot(): Promise<DatabaseSnapshot> {
    const connection = await this.connectionService.getConnection();
    const snapshotFile = `/tmp/snapshot_${Date.now()}.sql`;
    
    // mysqldump로 현재 데이터베이스 덤프
    await this.executeMysqlDump(connection, snapshotFile);
    
    return {
      file: snapshotFile,
      size: await this.getFileSize(snapshotFile),
      checksum: await this.calculateChecksum(snapshotFile)
    };
  }
  
  private async rollbackDatabase(snapshot: DatabaseSnapshot): Promise<void> {
    const connection = await this.connectionService.getConnection();
    
    // 체크섬 검증
    const currentChecksum = await this.calculateChecksum(snapshot.file);
    if (currentChecksum !== snapshot.checksum) {
      throw new Error('Snapshot file corrupted');
    }
    
    // 데이터베이스 복원
    await this.executeMysqlRestore(connection, snapshot.file);
  }
}
```
