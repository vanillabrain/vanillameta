```typescript
@Entity('notices')
export class Notice {
  @PrimaryGeneratedColumn('uuid')
  id: string;

  @Column({ length: 200 })
  title: string;

  @Column({ type: 'text' })
  content: string; // HTML 또는 Markdown 형식

  @Column({ type: 'enum', enum: ContentType, default: ContentType.MARKDOWN })
  contentType: ContentType;

  @Column({ type: 'text', nullable: true })
  excerpt: string; // 요약문 (선택사항)

  @ManyToOne(() => User, { onDelete: 'CASCADE' })
  @JoinColumn({ name: 'authorId' })
  author: User;

  @Column({ type: 'uuid' })
  authorId: string;

  @Column({ type: 'enum', enum: NoticeType })
  type: NoticeType;

  @Column({ type: 'enum', enum: NoticePriority, default: NoticePriority.NORMAL })
  priority: NoticePriority;

  @Column({ type: 'enum', enum: NoticeStatus, default: NoticeStatus.DRAFT })
  status: NoticeStatus;

  @Column({ type: 'json', nullable: true })
  targetAudience: TargetAudience;

  @Column({ type: 'json', nullable: true })
  displaySettings: DisplaySettings;

  @Column({ type: 'timestamp', nullable: true })
  scheduledAt: Date; // 예약 발송 시간

  @Column({ type: 'timestamp', nullable: true })
  publishedAt: Date;

  @Column({ type: 'timestamp', nullable: true })
  expiresAt: Date;

  @OneToMany(() => NoticeAttachment, attachment => attachment.notice)
  attachments: NoticeAttachment[];

  @OneToMany(() => NoticeRead, read => read.notice)
  reads: NoticeRead[];

  @Column({ type: 'boolean', default: false })
  isPinned: boolean; // 상단 고정

  @Column({ type: 'boolean', default: false })
  requiresAcknowledgment: boolean; // 확인 필요

  @Column({ type: 'json', nullable: true })
  metadata: NoticeMetadata;

  @CreateDateColumn()
  createdAt: Date;

  @UpdateDateColumn()
  updatedAt: Date;
}

enum ContentType {
  MARKDOWN = 'markdown',
  HTML = 'html',
  RICH_TEXT = 'rich_text'
}

enum NoticeType {
  ANNOUNCEMENT = 'announcement',    // 일반 공지
  MAINTENANCE = 'maintenance',      // 점검 공지
  UPDATE = 'update',               // 업데이트 공지
  SECURITY = 'security',           // 보안 공지
  EVENT = 'event',                 // 이벤트 공지
  POLICY = 'policy'                // 정책 공지
}

enum NoticePriority {
  LOW = 'low',
  NORMAL = 'normal',
  HIGH = 'high',
  URGENT = 'urgent'
}

enum NoticeStatus {
  DRAFT = 'draft',
  SCHEDULED = 'scheduled',
  PUBLISHED = 'published',
  ARCHIVED = 'archived'
}

interface TargetAudience {
  type: 'all' | 'teams' | 'departments' | 'roles' | 'users';
  targets: string[]; // team/department/role/user IDs
  excludes?: string[]; // 제외할 대상들
}

interface DisplaySettings {
  showAsPopup: boolean;
  showAsBanner: boolean;
  popupDelay: number; // 초 단위
  bannerPosition: 'top' | 'bottom';
  autoCloseAfter: number; // 초 단위, 0이면 수동 닫기
  allowDismiss: boolean;
}

interface NoticeMetadata {
  tags: string[];
  category: string;
  language: string;
  estimatedReadTime: number; // 분 단위
  relatedNotices: string[]; // 관련 공지 ID들
}
```

#### NoticeService
```typescript
// backend-api/src/modules/notice/services/notice.service.ts
@Injectable()
export class NoticeService {
  constructor(
    @InjectRepository(Notice) private noticeRepo: Repository<Notice>,
    @InjectRepository(NoticeRead) private noticeReadRepo: Repository<NoticeRead>,
    @InjectRepository(NoticeAttachment) private attachmentRepo: Repository<NoticeAttachment>,
    private fileUploadService: FileUploadService,
    private notificationService: NotificationService,
    private eventEmitter: EventEmitter2
  ) {}

  async createNotice(createDto: CreateNoticeDto, authorId: string): Promise<Notice> {
    const notice = this.noticeRepo.create({
      ...createDto,
      authorId,
      status: createDto.scheduledAt ? NoticeStatus.SCHEDULED : NoticeStatus.DRAFT
    });

    const saved = await this.noticeRepo.save(notice);

    // 예약 발송인 경우 스케줄링
    if (createDto.scheduledAt) {
      await this.scheduleNotice(saved.id, createDto.scheduledAt);
    }

    return saved;
  }

  async publishNotice(noticeId: string, publisherId: string): Promise<Notice> {
    const notice = await this.noticeRepo.findOne({
      where: { id: noticeId },
      relations: ['author']
    });

    if (!notice) {
      throw new NotFoundException('공지사항을 찾을 수 없습니다.');
    }

    if (notice.status === NoticeStatus.PUBLISHED) {
      throw new BadRequestException('이미 발행된 공지사항입니다.');
    }

    notice.status = NoticeStatus.PUBLISHED;
    notice.publishedAt = new Date();

    const updated = await this.noticeRepo.save(notice);

    // 대상 사용자들에게 알림 발송
    await this.sendNoticeNotifications(updated);

    // 이벤트 발행
    this.eventEmitter.emit('notice.published', { notice: updated, publisherId });

    return updated;
  }

  async getNoticesForUser(userId: string, filters: NoticeFilters): Promise<PaginatedResult<Notice>> {
    // 사용자가 볼 수 있는 공지사항 조회
    const userTeams = await this.getUserTeams(userId);
    const userDepartments = await this.getUserDepartments(userId);
    const userRoles = await this.getUserRoles(userId);

    const query = this.noticeRepo.createQueryBuilder('notice')
      .leftJoinAndSelect('notice.author', 'author')
      .leftJoinAndSelect('notice.reads', 'reads', 'reads.userId = :userId', { userId })
      .where('notice.status = :status', { status: NoticeStatus.PUBLISHED })
      .andWhere('(notice.expiresAt IS NULL OR notice.expiresAt > :now)', { now: new Date() });

    // 대상 필터링
    this.applyAudienceFilter(query, userId, userTeams, userDepartments, userRoles);

    // 추가 필터 적용
    this.applyNoticeFilters(query, filters);

    // 정렬: 고정 > 우선순위 > 발행일
    query.orderBy('notice.isPinned', 'DESC')
      .addOrderBy('notice.priority', 'DESC')
      .addOrderBy('notice.publishedAt', 'DESC');

    const [items, total] = await query
      .skip((filters.page - 1) * filters.limit)
      .take(filters.limit)
      .getManyAndCount();

    return {
      items,
      total,
      page: filters.page,
      limit: filters.limit,
      totalPages: Math.ceil(total / filters.limit)
    };
  }

  async markAsRead(noticeId: string, userId: string): Promise<void> {
    const existingRead = await this.noticeReadRepo.findOne({
      where: { noticeId, userId }
    });

    if (existingRead) {
      existingRead.readCount++;
      existingRead.readAt = new Date();
      await this.noticeReadRepo.save(existingRead);
    } else {
      const noticeRead = this.noticeReadRepo.create({
        noticeId,
        userId,
        readAt: new Date()
      });
      await this.noticeReadRepo.save(noticeRead);
    }

    // 읽음 통계 업데이트 이벤트
    this.eventEmitter.emit('notice.read', { noticeId, userId });
  }

  async acknowledgeNotice(noticeId: string, userId: string): Promise<void> {
    const noticeRead = await this.noticeReadRepo.findOne({
      where: { noticeId, userId }
    });

    if (!noticeRead) {
      // 읽지 않은 경우 먼저 읽음 처리
      await this.markAsRead(noticeId, userId);
      
      const newNoticeRead = await this.noticeReadRepo.findOne({
        where: { noticeId, userId }
      });
      
      newNoticeRead.acknowledged = true;
      newNoticeRead.acknowledgedAt = new Date();
      await this.noticeReadRepo.save(newNoticeRead);
    } else {
      noticeRead.acknowledged = true;
      noticeRead.acknowledgedAt = new Date();
      await this.noticeReadRepo.save(noticeRead);
    }

    this.eventEmitter.emit('notice.acknowledged', { noticeId, userId });
  }

  async uploadAttachment(noticeId: string, file: Express.Multer.File): Promise<NoticeAttachment> {
    const notice = await this.noticeRepo.findOne({ where: { id: noticeId } });
    if (!notice) {
      throw new NotFoundException('공지사항을 찾을 수 없습니다.');
    }

    // 파일 업로드
    const uploadResult = await this.fileUploadService.uploadFile(file, 'notices');

    const attachment = this.attachmentRepo.create({
      noticeId,
      fileName: uploadResult.fileName,
      originalFileName: file.originalname,
      mimeType: file.mimetype,
      fileSize: file.size,
      filePath: uploadResult.filePath
    });

    return this.attachmentRepo.save(attachment);
  }

  @Cron('*/5 * * * *') // 5분마다 실행
  async processScheduledNotices(): Promise<void> {
    const scheduledNotices = await this.noticeRepo.find({
      where: {
        status: NoticeStatus.SCHEDULED,
        scheduledAt: LessThanOrEqual(new Date())
      }
    });

    for (const notice of scheduledNotices) {
      try {
        await this.publishNotice(notice.id, notice.authorId);
      } catch (error) {
        console.error(`Failed to publish scheduled notice ${notice.id}:`, error);
      }
    }
  }

  async getNoticeStats(noticeId: string): Promise<NoticeStats> {
    const notice = await this.noticeRepo.findOne({
      where: { id: noticeId },
      relations: ['reads']
    });

    if (!notice) {
      throw new NotFoundException('공지사항을 찾을 수 없습니다.');
    }

    // 대상 사용자 수 계산
    const targetUserCount = await this.calculateTargetUserCount(notice.targetAudience);
    
    // 읽은 사용자 수
    const readUserCount = notice.reads.length;
    
    // 확인한 사용자 수
    const acknowledgedUserCount = notice.reads.filter(read => read.acknowledged).length;

    return {
      targetUserCount,
      readUserCount,
      acknowledgedUserCount,
      readRate: targetUserCount > 0 ? (readUserCount / targetUserCount) * 100 : 0,
      acknowledgmentRate: targetUserCount > 0 ? (acknowledgedUserCount / targetUserCount) * 100 : 0,
      totalDownloads: await this.getTotalDownloads(noticeId)
    };
  }

  private async sendNoticeNotifications(notice: Notice): Promise<void> {
    // 대상 사용자 목록 조회
    const targetUsers = await this.getTargetUsers(notice.targetAudience);

    // 각 사용자에게 알림 발송
    const notifications = targetUsers.map(userId => ({
      userId,
      title: `새 공지사항: ${notice.title}`,
      message: notice.excerpt || notice.title,
      type: 'notice',
      relatedId: notice.id,
      priority: this.mapNoticePriorityToNotificationPriority(notice.priority)
    }));

}
```

```typescript
import { Modal, Form, Input, Select, DatePicker, Switch, Upload, Button, Alert } from 'antd';
import { Editor } from '@tinymce/tinymce-react';
import { FileText, Calendar, Users, AlertCircle } from 'lucide-react';
import { Notice, CreateNoticeDto, TargetAudience } from '@/types/notice';

interface NoticeEditorProps {
  visible: boolean;
  notice?: Notice | null;
  onSubmit: (data: CreateNoticeDto) => Promise<void>;
  onCancel: () => void;
}

export const NoticeEditor: React.FC<NoticeEditorProps> = ({
  visible,
  notice,
  onSubmit,
  onCancel
}) => {
  const [form] = Form.useForm();
  const [content, setContent] = useState('');
  const [loading, setLoading] = useState(false);
  const [attachments, setAttachments] = useState<any[]>([]);

  const isEdit = !!notice;

  useEffect(() => {
    if (visible && notice) {
      form.setFieldsValue({
        title: notice.title,
        type: notice.type,
        priority: notice.priority,
        targetAudience: notice.targetAudience,
        displaySettings: notice.displaySettings,
        scheduledAt: notice.scheduledAt ? moment(notice.scheduledAt) : null,
        expiresAt: notice.expiresAt ? moment(notice.expiresAt) : null,
        isPinned: notice.isPinned,
        requiresAcknowledgment: notice.requiresAcknowledgment
      });
      setContent(notice.content);
    } else if (visible) {
      form.resetFields();
      setContent('');
      setAttachments([]);
    }
  }, [visible, notice, form]);

  const handleSubmit = async () => {
    try {
      setLoading(true);
      const values = await form.validateFields();
      
      const submitData: CreateNoticeDto = {
        ...values,
        content,
        contentType: 'rich_text',
        excerpt: this.generateExcerpt(content)
      };

      await onSubmit(submitData);
      onCancel();
    } catch (error) {
      console.error('Notice submission failed:', error);
    } finally {
      setLoading(false);
    }
  };

  const generateExcerpt = (htmlContent: string): string => {
    // HTML 태그 제거하고 첫 150자만 추출
    const textContent = htmlContent.replace(/<[^>]*>/g, '');
    return textContent.length > 150 ? 
      textContent.substring(0, 150) + '...' : 
      textContent;
  };

  return (
    <Modal
      title={isEdit ? '공지사항 수정' : '새 공지사항 작성'}
      open={visible}
      onCancel={onCancel}
      width={1000}
      footer={[
        <Button key="cancel" onClick={onCancel}>
          취소
        </Button>,
        <Button key="draft" onClick={() => handleSubmit('draft')}>
          임시저장
        </Button>,
        <Button key="schedule" type="default" onClick={() => handleSubmit('schedule')}>
          예약 발송
        </Button>,
        <Button key="publish" type="primary" loading={loading} onClick={() => handleSubmit('publish')}>
          즉시 발행
        </Button>
      ]}
      destroyOnClose
    >
      <Form
        form={form}
        layout="vertical"
        requiredMark={false}
      >
        <div className="grid grid-cols-3 gap-4">
          {/* 메인 콘텐츠 영역 */}
          <div className="col-span-2">
            <Form.Item
              name="title"
              label="제목"
              rules={[{ required: true, message: '제목을 입력해주세요.' }]}
            >
              <Input 
                placeholder="공지사항 제목을 입력하세요"
                size="large"
              />
            </Form.Item>

            <Form.Item label="내용">
              <Editor
                apiKey="your-tinymce-api-key"
                value={content}
                onEditorChange={setContent}
                init={{
                  height: 400,
                  menubar: false,
                  plugins: [
                    'advlist autolink lists link image charmap print preview anchor',
                    'searchreplace visualblocks code fullscreen',
                    'insertdatetime media table paste code help wordcount'
                  ],
                  toolbar: 'undo redo | formatselect | bold italic backcolor | \
                    alignleft aligncenter alignright alignjustify | \
                    bullist numlist outdent indent | removeformat | help',
                  content_style: 'body { font-family: -apple-system, BlinkMacSystemFont, San Francisco, Segoe UI, Roboto, Helvetica Neue, sans-serif; font-size: 14px }'
                }}
              />
            </Form.Item>

            <Form.Item label="첨부파일">
              <Upload
                fileList={attachments}
                onChange={({ fileList }) => setAttachments(fileList)}
                beforeUpload={() => false} // 수동 업로드
                multiple
              >
```

1. Notice, NoticeRead, NoticeAttachment 엔티티 구현
2. NoticeService 기본 CRUD 기능
3. 예약 발송 스케줄링 로직

### Phase 2: 고급 기능 (2일)
1. 대상 사용자 필터링 로직
2. 읽음/확인 추적 시스템
3. 첨부파일 관리
4. 통계 및 분석 기능

### Phase 3: 프론트엔드 에디터 (1일)
1. NoticeEditor 컴포넌트 (TinyMCE 통합)
2. 대상 선택 UI
3. 예약 발송 설정 UI
