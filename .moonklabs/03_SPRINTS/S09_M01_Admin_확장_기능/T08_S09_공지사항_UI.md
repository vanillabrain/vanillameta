### 1. 공지사항 목록 페이지

#### NoticeListPage 컴포넌트
```typescript
// frontend-web/src/pages/notice/NoticeListPage.tsx
import React, { useState, useEffect } from 'react';
import { List, Card, Badge, Tag, Button, Select, Input, DatePicker, Empty } from 'antd';
import { Pin, Eye, Calendar, User, Search } from 'lucide-react';
import { Notice, NoticeFilters } from '@/types/notice';
import { useNotices } from '@/hooks/useNotices';

export const NoticeListPage: React.FC = () => {
  const [filters, setFilters] = useState<NoticeFilters>({
    page: 1,
    limit: 10,
    status: 'published'
  });

  const {
    notices,
    loading,
    total,
    loadNotices,
    markAsRead
  } = useNotices(filters);

  useEffect(() => {
    loadNotices();
  }, [filters, loadNotices]);

  const getNoticeTypeColor = (type: string) => {
    const colorMap = {
      announcement: 'blue',
      maintenance: 'orange',
      update: 'green',
      security: 'red',
      event: 'purple',
      policy: 'gold'
    };
    return colorMap[type] || 'default';
  };

  const getNoticeTypeLabel = (type: string) => {
    const labelMap = {
      announcement: '일반 공지',
      maintenance: '점검 공지',
      update: '업데이트',
      security: '보안 공지',
      event: '이벤트',
      policy: '정책 공지'
    };
    return labelMap[type] || type;
  };

  const getPriorityIcon = (priority: string) => {
    switch (priority) {
      case 'urgent':
        return <span className="text-red-500">🚨</span>;
      case 'high':
        return <span className="text-orange-500">⚡</span>;
      case 'normal':
        return <span className="text-blue-500">📢</span>;
      case 'low':
        return <span className="text-gray-500">📝</span>;
      default:
        return null;
    }
  };

  const handleNoticeClick = async (notice: Notice) => {
    // 읽음 처리
    if (!notice.isRead) {
      await markAsRead(notice.id);
    }
    
    // 상세 페이지로 이동
    window.location.href = `/notices/${notice.id}`;
  };

  return (
    <div className="notice-list-page max-w-6xl mx-auto p-6">
      <div className="page-header mb-6">
        <div className="flex justify-between items-center">
          <h1 className="text-2xl font-bold">공지사항</h1>
          <div className="text-sm text-gray-500">
            총 {total}개의 공지사항
          </div>
        </div>
      </div>

      {/* 필터 영역 */}
      <Card className="mb-6">
        <div className="grid grid-cols-1 md:grid-cols-4 gap-4">
          <Input
            placeholder="제목 검색"
            prefix={<Search size={16} />}
            value={filters.searchTerm}
            onChange={(e) => setFilters({ ...filters, searchTerm: e.target.value, page: 1 })}
          />
          
          <Select
            placeholder="공지 유형"
            value={filters.type}
            onChange={(value) => setFilters({ ...filters, type: value, page: 1 })}
            allowClear
          >
            <Select.Option value="announcement">일반 공지</Select.Option>
            <Select.Option value="maintenance">점검 공지</Select.Option>
            <Select.Option value="update">업데이트</Select.Option>
            <Select.Option value="security">보안 공지</Select.Option>
            <Select.Option value="event">이벤트</Select.Option>
            <Select.Option value="policy">정책 공지</Select.Option>
          </Select>

          <Select
            placeholder="우선순위"
            value={filters.priority}
            onChange={(value) => setFilters({ ...filters, priority: value, page: 1 })}
            allowClear
          >
            <Select.Option value="urgent">긴급</Select.Option>
            <Select.Option value="high">높음</Select.Option>
            <Select.Option value="normal">보통</Select.Option>
            <Select.Option value="low">낮음</Select.Option>
          </Select>

          <DatePicker.RangePicker
            placeholder={['시작일', '종료일']}
            onChange={(dates) => setFilters({
              ...filters,
              dateFrom: dates?.[0]?.toDate(),
              dateTo: dates?.[1]?.toDate(),
              page: 1
            })}
          />
        </div>
      </Card>

      {/* 공지사항 목록 */}
      {notices.length === 0 && !loading ? (
        <Empty 
          description="공지사항이 없습니다"
          image={Empty.PRESENTED_IMAGE_SIMPLE}
        />
      ) : (
        <List
          loading={loading}
          dataSource={notices}
          pagination={{
            current: filters.page,
            pageSize: filters.limit,
            total: total,
            showSizeChanger: true,
            showQuickJumper: true,
            onChange: (page, size) => setFilters({ ...filters, page, limit: size })
          }}
          renderItem={(notice) => (
            <NoticeListItem
              key={notice.id}
              notice={notice}
              onClick={() => handleNoticeClick(notice)}
            />
          )}
        />
      )}
    </div>
  );
};

const NoticeListItem: React.FC<{
  notice: Notice;
  onClick: () => void;
}> = ({ notice, onClick }) => {
  const formatDate = (dateString: string) => {
    return new Date(dateString).toLocaleDateString('ko-KR', {
      year: 'numeric',
      month: 'long',
      day: 'numeric'
    });
  };

  return (
    <List.Item className="notice-list-item">
      <Card
        hoverable
        className={`w-full cursor-pointer transition-all duration-200 ${
          !notice.isRead ? 'border-l-4 border-l-blue-500 bg-blue-50' : ''
        } ${notice.isPinned ? 'border-t-2 border-t-yellow-400' : ''}`}
        onClick={onClick}
      >
        <div className="flex items-start justify-between">
          <div className="flex-1">
            <div className="flex items-center space-x-2 mb-2">
              {notice.isPinned && (
                <Pin size={16} className="text-yellow-500" />
              )}
              {getPriorityIcon(notice.priority)}
              <Tag color={getNoticeTypeColor(notice.type)}>
                {getNoticeTypeLabel(notice.type)}
              </Tag>
              {!notice.isRead && (
                <Badge status="processing" text="새 공지" />
              )}
            </div>

            <h3 className={`text-lg mb-2 ${!notice.isRead ? 'font-bold' : 'font-medium'}`}>
              {notice.title}
            </h3>

            {notice.excerpt && (
              <p className="text-gray-600 text-sm mb-3 line-clamp-2">
                {notice.excerpt}
              </p>
            )}

            <div className="flex items-center space-x-4 text-sm text-gray-500">
              <div className="flex items-center space-x-1">
                <User size={14} />
                <span>{notice.author.name}</span>
              </div>
              <div className="flex items-center space-x-1">
                <Calendar size={14} />
                <span>{formatDate(notice.publishedAt)}</span>
              </div>
              <div className="flex items-center space-x-1">
                <Eye size={14} />
                <span>조회 {notice.viewCount || 0}</span>
              </div>
            </div>
          </div>

          {notice.hasAttachments && (
            <div className="ml-4">
              <Tag color="cyan">📎 첨부파일</Tag>
            </div>
          )}
        </div>
      </Card>
    </List.Item>
  );
};
```

```typescript
// frontend-web/src/pages/notice/NoticeDetailPage.tsx
import React, { useState, useEffect } from 'react';
import { useParams } from 'react-router-dom';
import { Card, Typography, Tag, Button, Divider, List, Modal, message } from 'antd';
import { ArrowLeft, Download, Eye, Calendar, User, MessageCircle } from 'lucide-react';
import { Notice, NoticeAttachment } from '@/types/notice';
import { useNoticeDetail } from '@/hooks/useNoticeDetail';

const { Title, Paragraph } = Typography;

export const NoticeDetailPage: React.FC = () => {
  const { id } = useParams<{ id: string }>();
  const [acknowledgmentModalVisible, setAcknowledgmentModalVisible] = useState(false);

  const {
    notice,
    loading,
    loadNotice,
    acknowledgeNotice,
    downloadAttachment
  } = useNoticeDetail(id!);

  useEffect(() => {
    if (id) {
      loadNotice();
    }
  }, [id, loadNotice]);

  useEffect(() => {
    // 확인 필요한 공지사항인 경우 모달 표시
    if (notice && notice.requiresAcknowledgment && !notice.isAcknowledged) {
      setAcknowledgmentModalVisible(true);
    }
  }, [notice]);

  const handleAcknowledge = async () => {
    try {
      await acknowledgeNotice();
      setAcknowledgmentModalVisible(false);
      message.success('공지사항을 확인했습니다.');
    } catch (error) {
      message.error('확인 처리에 실패했습니다.');
    }
  };

  const handleDownload = async (attachment: NoticeAttachment) => {
    try {
      await downloadAttachment(attachment.id);
    } catch (error) {
      message.error('파일 다운로드에 실패했습니다.');
    }
  };

  if (loading || !notice) {
    return <div className="p-6">로딩 중...</div>;
  }

  return (
    <div className="notice-detail-page max-w-4xl mx-auto p-6">
      {/* 헤더 */}
      <div className="mb-6">
        <Button 
          icon={<ArrowLeft size={16} />} 
          onClick={() => window.history.back()}
          className="mb-4"
        >
          목록으로
        </Button>
        
        <div className="flex items-start justify-between">
          <div className="flex-1">
            <div className="flex items-center space-x-2 mb-2">
              <Tag color={getNoticeTypeColor(notice.type)}>
                {getNoticeTypeLabel(notice.type)}
              </Tag>
              {notice.isPinned && (
                <Tag color="gold">📌 고정</Tag>
              )}
              {notice.priority === 'urgent' && (
                <Tag color="red">🚨 긴급</Tag>
              )}
              {notice.requiresAcknowledgment && (
                <Tag color={notice.isAcknowledged ? 'green' : 'orange'}>
                  {notice.isAcknowledged ? '✅ 확인완료' : '⏳ 확인필요'}
                </Tag>
              )}
            </div>
            
            <Title level={2} className="mb-2">
              {notice.title}
            </Title>
            
            <div className="flex items-center space-x-4 text-gray-500 text-sm">
              <div className="flex items-center space-x-1">
                <User size={14} />
                <span>{notice.author.name}</span>
              </div>
              <div className="flex items-center space-x-1">
                <Calendar size={14} />
                <span>{new Date(notice.publishedAt).toLocaleString('ko-KR')}</span>
              </div>
              <div className="flex items-center space-x-1">
                <Eye size={14} />
                <span>조회 {notice.viewCount}</span>
              </div>
            </div>
          </div>
          
          {notice.requiresAcknowledgment && !notice.isAcknowledged && (
            <Button 
              type="primary" 
              onClick={() => setAcknowledgmentModalVisible(true)}
            >
              확인하기
            </Button>
          )}
        </div>
      </div>

      {/* 본문 */}
      <Card>
        <div 
          className="notice-content prose max-w-none"
          dangerouslySetInnerHTML={{ __html: notice.content }}
        />
        
        {/* 첨부파일 */}
        {notice.attachments && notice.attachments.length > 0 && (
          <>
            <Divider />
            <div className="attachments">
              <h4 className="mb-3">첨부파일</h4>
              <List
                dataSource={notice.attachments}
                renderItem={(attachment) => (
                  <List.Item
                    actions={[
                      <Button
                        key="download"
                        type="link"
                        icon={<Download size={16} />}
                        onClick={() => handleDownload(attachment)}
                      >
                        다운로드 ({attachment.downloadCount})
                      </Button>
                    ]}
                  >
                    <List.Item.Meta
                      title={attachment.originalFileName}
                      description={`${formatFileSize(attachment.fileSize)} • ${attachment.mimeType}`}
                    />
                  </List.Item>
                )}
              />
            </div>
          </>
        )}
      </Card>

      {/* 확인 모달 */}
      <Modal
        title="공지사항 확인"
        open={acknowledgmentModalVisible}
        onOk={handleAcknowledge}
        onCancel={() => setAcknowledgmentModalVisible(false)}
        okText="확인"
        cancelText="나중에"
        centered
      >
        <div className="py-4">
          <Paragraph>
            이 공지사항을 읽고 확인했음을 알려주세요.
            확인 후에는 다시 되돌릴 수 없습니다.
          </Paragraph>
          
          <div className="bg-blue-50 p-3 rounded">
            <Paragraph strong className="mb-0">
              "{notice.title}"
            </Paragraph>
          </div>
        </div>
      </Modal>
    </div>
  );
};

const formatFileSize = (bytes: number): string => {
  if (bytes === 0) return '0 Bytes';
  const k = 1024;
  const sizes = ['Bytes', 'KB', 'MB', 'GB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));
  return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
};

  const colorMap = {

    maintenance: 'orange',
    update: 'green',
    security: 'red',
    event: 'purple',
    policy: 'gold'
  };
  return colorMap[type] || 'default';
};

    security: '보안 공지',

};
```

### 3. 팝업/배너 표시 시스템

#### NoticePopupManager 컴포넌트
```typescript
// frontend-web/src/components/notice/NoticePopupManager.tsx
import React, { useState, useEffect } from 'react';
import { Modal, Alert, Button, Checkbox } from 'antd';
import { X, Pin, ExternalLink } from 'lucide-react';
import { Notice } from '@/types/notice';
import { useNoticePopups } from '@/hooks/useNoticePopups';

export const NoticePopupManager: React.FC = () => {
  const {
    popupNotices,
    bannerNotices,
    dismissNotice,
    markAsRead
  } = useNoticePopups();

  return (
    <>
      {/* 배너 공지사항 */}
      <NoticeBanners 
        notices={bannerNotices}
        onDismiss={dismissNotice}
        onMarkAsRead={markAsRead}
      />
      
      {/* 팝업 공지사항 */}
      <NoticePopups 
        notices={popupNotices}
        onDismiss={dismissNotice}
        onMarkAsRead={markAsRead}
      />
    </>
  );
};

const NoticeBanners: React.FC<{
  notices: Notice[];
  onDismiss: (noticeId: string) => void;
  onMarkAsRead: (noticeId: string) => void;
}> = ({ notices, onDismiss, onMarkAsRead }) => {
  if (notices.length === 0) return null;

  return (
    <div className="notice-banners">
      {notices.map((notice) => (
        <NoticeBanner
          key={notice.id}
          notice={notice}
          onDismiss={onDismiss}
          onMarkAsRead={onMarkAsRead}
        />
      ))}
    </div>
  );
};

const NoticeBanner: React.FC<{
  notice: Notice;
  onDismiss: (noticeId: string) => void;
  onMarkAsRead: (noticeId: string) => void;
}> = ({ notice, onDismiss, onMarkAsRead }) => {
  const [visible, setVisible] = useState(true);

  useEffect(() => {
    // 읽음 처리
    if (!notice.isRead) {
      onMarkAsRead(notice.id);
    }
  }, [notice.id, notice.isRead, onMarkAsRead]);

  const handleClose = () => {
    setVisible(false);
    if (notice.displaySettings?.allowDismiss) {
      onDismiss(notice.id);
    }
  };

  const getBannerType = (priority: string) => {
    switch (priority) {
      case 'urgent':
        return 'error';
      case 'high':
        return 'warning';
      default:
        return 'info';
    }
  };

  if (!visible) return null;

  return (
    <div 
      className={`notice-banner ${
        notice.displaySettings?.bannerPosition === 'bottom' ? 'bottom-0' : 'top-0'
      } fixed left-0 right-0 z-50`}
    >
      <Alert
        message={
          <div className="flex items-center justify-between">
            <div className="flex items-center space-x-2">
              {notice.isPinned && <Pin size={16} />}
              <span className="font-medium">{notice.title}</span>
            </div>
            <div className="flex items-center space-x-2">
              {notice.data?.actionUrl && (
                <Button
                  type="link"
                  size="small"
                  icon={<ExternalLink size={14} />}
                  onClick={() => window.open(notice.data.actionUrl, '_blank')}
                >
                  자세히 보기
                </Button>
              )}
              {notice.displaySettings?.allowDismiss && (
                <Button
                  type="text"
                  size="small"
                  icon={<X size={14} />}
                  onClick={handleClose}
                />
              )}
            </div>
          </div>
        }
        type={getBannerType(notice.priority)}
        banner
        closable={false}
      />
    </div>
  );
};

const NoticePopups: React.FC<{
  notices: Notice[];
  onDismiss: (noticeId: string) => void;
  onMarkAsRead: (noticeId: string) => void;
}> = ({ notices, onDismiss, onMarkAsRead }) => {
  const [currentIndex, setCurrentIndex] = useState(0);
  const [dontShowToday, setDontShowToday] = useState(false);

  const currentNotice = notices[currentIndex];

  useEffect(() => {
    if (currentNotice && !currentNotice.isRead) {
      onMarkAsRead(currentNotice.id);
    }
  }, [currentNotice, onMarkAsRead]);

  useEffect(() => {
    // 자동 닫기 설정이 있는 경우
    if (currentNotice?.displaySettings?.autoCloseAfter) {
      const timer = setTimeout(() => {
        handleNext();
      }, currentNotice.displaySettings.autoCloseAfter * 1000);

      return () => clearTimeout(timer);
    }
  }, [currentNotice]);

  const handleNext = () => {
    if (currentIndex < notices.length - 1) {
      setCurrentIndex(currentIndex + 1);
    } else {
      handleClose();
    }
  };

  const handleClose = () => {
    if (currentNotice) {
      onDismiss(currentNotice.id);
      
      if (dontShowToday) {
        // 오늘 하루 보지 않기 설정 저장
        localStorage.setItem(
          `notice_dont_show_${currentNotice.id}`,
          new Date().toDateString()
        );
      }
    }
    
    setCurrentIndex(0);
  };

  if (!currentNotice) return null;

  return (
    <Modal
      title={
        <div className="flex items-center space-x-2">
          {currentNotice.isPinned && <Pin size={16} className="text-yellow-500" />}
          <span>{currentNotice.title}</span>
          {notices.length > 1 && (
            <span className="text-sm text-gray-500">
              ({currentIndex + 1}/{notices.length})
            </span>
          )}
        </div>
      }
      open={true}
      onCancel={handleClose}
      footer={[
        <div key="footer" className="flex justify-between items-center w-full">
          <Checkbox
            checked={dontShowToday}
            onChange={(e) => setDontShowToday(e.target.checked)}
          >
            오늘 하루 보지 않기
          </Checkbox>
          
          <div className="space-x-2">
            {currentNotice.data?.actionUrl && (
              <Button
                type="primary"
                onClick={() => {
                  window.open(currentNotice.data.actionUrl, '_blank');
                  handleNext();
                }}
              >
                {currentNotice.data.actionText || '자세히 보기'}
              </Button>
            )}
            <Button onClick={handleNext}>
              {currentIndex < notices.length - 1 ? '다음' : '닫기'}
            </Button>
          </div>
        </div>
      ]}
      width={600}
      centered
    >
      <div className="notice-popup-content">
        <div 
          className="prose max-w-none"
          dangerouslySetInnerHTML={{ __html: currentNotice.content }}
        />
      </div>
    </Modal>
  );
};
```

### 4. 읽음 확인 위젯

#### NoticeReadStatus 컴포넌트
```typescript
// frontend-web/src/components/notice/NoticeReadStatus.tsx
import React, { useState, useEffect } from 'react';
import { Progress, List, Avatar, Tag, Modal, Button } from 'antd';
import { Eye, Users, Clock } from 'lucide-react';
import { NoticeStats, NoticeReadInfo } from '@/types/notice';

interface NoticeReadStatusProps {
  noticeId: string;
  visible: boolean;
  onCancel: () => void;
}

export const NoticeReadStatus: React.FC<NoticeReadStatusProps> = ({
  noticeId,
  visible,
  onCancel
}) => {
  const [stats, setStats] = useState<NoticeStats | null>(null);
  const [readList, setReadList] = useState<NoticeReadInfo[]>([]);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    if (visible && noticeId) {
      loadNoticeStats();
      loadReadList();
    }
  }, [visible, noticeId]);

  const loadNoticeStats = async () => {
    try {
      setLoading(true);
      const response = await fetch(`/api/notices/${noticeId}/stats`);
      const data = await response.json();
      setStats(data);
    } catch (error) {
      console.error('Failed to load notice stats:', error);
    } finally {
      setLoading(false);
    }
  };

  const loadReadList = async () => {
    try {
      const response = await fetch(`/api/notices/${noticeId}/reads`);
      const data = await response.json();
      setReadList(data);
    } catch (error) {
      console.error('Failed to load read list:', error);
    }
  };

  return (
    <Modal
      title="읽음 현황"
      open={visible}
      onCancel={onCancel}
      footer={[
        <Button key="close" onClick={onCancel}>
          닫기
        </Button>
      ]}
      width={600}
    >
      {stats && (
        <div className="space-y-6">
          {/* 통계 요약 */}
          <div className="grid grid-cols-3 gap-4">
            <div className="text-center">
              <div className="text-2xl font-bold text-blue-600">
                {stats.readUserCount}
              </div>
              <div className="text-sm text-gray-500">읽은 사용자</div>
            </div>
            <div className="text-center">
              <div className="text-2xl font-bold text-green-600">
                {stats.acknowledgedUserCount}
              </div>

                {stats.targetUserCount}
              </div>

              </div>
              <Progress percent={stats.acknowledgmentRate} strokeColor="#52c41a" />
            </div>
          </div>

          {/* 읽음 목록 */}
          <div>

                        {readInfo.acknowledged && (

                        )}
                      </div>
                    }

                        <div className="flex items-center space-x-1">
                          <Eye size={12} />
                          <span>{new Date(readInfo.readAt).toLocaleString('ko-KR')}</span>

                        {readInfo.acknowledgedAt && (
                          <div className="flex items-center space-x-1">
                            <Clock size={12} />
                            <span>확인: {new Date(readInfo.acknowledgedAt).toLocaleString('ko-KR')}</span>
                          </div>
                        )}
                      </div>
                    }
                  />
                </List.Item>

            />
          </div>
        </div>
      )}
    </Modal>
  );
};
```
