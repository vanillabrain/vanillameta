```typescript
// backend-api/src/modules/dashboard/services/team-isolation.service.ts
@Injectable()
export class TeamIsolationService {
  constructor(
    @InjectRepository(Dashboard) private dashboardRepo: Repository<Dashboard>,
    @InjectRepository(DashboardTeamAccess) private accessRepo: Repository<DashboardTeamAccess>,
    @InjectRepository(TeamMember) private teamMemberRepo: Repository<TeamMember>,
    private permissionService: PermissionResolverService
  ) {}

  async getUserAccessibleDashboards(userId: string): Promise<Dashboard[]> {
    // 사용자가 속한 팀들 조회
    const userTeams = await this.teamMemberRepo.find({
      where: { userId, isActive: true },
      relations: ['team']
    });

    const teamIds = userTeams.map(tm => tm.teamId);
    
    // 팀별 대시보드와 공유된 대시보드 조회
    const query = this.dashboardRepo.createQueryBuilder('dashboard')
      .leftJoinAndSelect('dashboard.team', 'team')
      .leftJoinAndSelect('dashboard.teamAccesses', 'access')
      .leftJoinAndSelect('access.team', 'accessTeam')
      .where('dashboard.teamId IN (:...teamIds)', { teamIds })
      .orWhere('access.teamId IN (:...teamIds)', { teamIds })
      .orWhere('dashboard.accessScope = :orgScope', { orgScope: AccessScope.ORGANIZATION })
      .andWhere('dashboard.isDeleted = false');

    const dashboards = await query.getMany();

    // 권한 필터링
    return this.filterByPermissions(dashboards, userId, userTeams);
  }

  async canUserAccessDashboard(userId: string, dashboardId: string): Promise<boolean> {
    const dashboard = await this.dashboardRepo.findOne({
      where: { id: dashboardId },
      relations: ['team', 'teamAccesses', 'teamAccesses.team']
    });

    if (!dashboard) return false;

    // 개인 대시보드인 경우 소유자 확인
    if (dashboard.accessScope === AccessScope.PERSONAL) {
      return dashboard.createdBy === userId;
    }

    // 조직 전체 공개인 경우
    if (dashboard.accessScope === AccessScope.ORGANIZATION) {
      return true;
    }

    // 팀 기반 권한 확인
    const userTeams = await this.getUserTeams(userId);
    return this.checkTeamBasedAccess(dashboard, userTeams);
  }

  async grantTeamAccess(
    dashboardId: string,
    teamId: string,
    accessLevel: AccessLevel,
    grantedBy: string
  ): Promise<void> {
    // 기존 접근 권한 확인
    const existingAccess = await this.accessRepo.findOne({
      where: { dashboardId, teamId }
    });

    if (existingAccess) {
      existingAccess.accessLevel = accessLevel;
      existingAccess.grantedBy = grantedBy;
      existingAccess.grantedAt = new Date();
      await this.accessRepo.save(existingAccess);
    } else {
      const newAccess = this.accessRepo.create({
        dashboardId,
        teamId,
        accessLevel,
        grantedBy,
        grantedAt: new Date()
      });
      await this.accessRepo.save(newAccess);
    }
  }

  private async filterByPermissions(
    dashboards: Dashboard[],
    userId: string,
    userTeams: TeamMember[]
  ): Promise<Dashboard[]> {
    const filtered = [];

    for (const dashboard of dashboards) {
      const hasPermission = await this.checkDashboardPermission(
        dashboard,
        userId,
        userTeams,
        'read'
      );
      
      if (hasPermission) {
        filtered.push(dashboard);
      }
    }

    return filtered;
  }

  private async checkDashboardPermission(
    dashboard: Dashboard,
    userId: string,
    userTeams: TeamMember[],
    permission: string
  ): Promise<boolean> {
    // 대시보드 소유자인 경우
    if (dashboard.createdBy === userId) {
      return true;
    }

    // 팀 권한 확인
    const userTeamIds = userTeams.map(tm => tm.teamId);
    
    // 대시보드가 속한 팀의 멤버인지 확인
    if (dashboard.teamId && userTeamIds.includes(dashboard.teamId)) {
      const userTeam = userTeams.find(tm => tm.teamId === dashboard.teamId);
      return this.checkRolePermission(dashboard.teamPermissions, userTeam.role, permission);
    }

    // 공유된 팀 접근 권한 확인
    for (const access of dashboard.teamAccesses) {
      if (userTeamIds.includes(access.teamId)) {
        return this.checkAccessLevelPermission(access.accessLevel, permission);
      }
    }

    return false;
  }
}
```

#### 위임 관리 서비스
```typescript
// backend-api/src/modules/admin/services/delegation.service.ts
@Injectable()
export class DelegationService {
  constructor(
    @InjectRepository(TeamLeaderDelegation) private delegationRepo: Repository<TeamLeaderDelegation>,
    @InjectRepository(TeamMember) private teamMemberRepo: Repository<TeamMember>,
    private notificationService: NotificationService
  ) {}

  async createDelegation(request: CreateDelegationRequest): Promise<TeamLeaderDelegation> {
    // 위임자가 팀 리더/관리자인지 확인
    await this.validateDelegator(request.delegatedBy, request.teamId);

    // 피위임자가 해당 팀 멤버인지 확인
    await this.validateDelegatee(request.delegatedTo, request.teamId);

    const delegation = this.delegationRepo.create({
      ...request,
      isActive: true
    });

    const saved = await this.delegationRepo.save(delegation);

    // 알림 발송
    await this.notificationService.sendDelegationNotification(saved);

    return saved;
  }

  async getActiveDelegations(teamId: string): Promise<TeamLeaderDelegation[]> {
    return this.delegationRepo.find({
      where: {
        teamId,
        isActive: true,
        validFrom: LessThanOrEqual(new Date()),
        validTo: MoreThanOrEqual(new Date())
      },
      relations: ['delegatedToUser', 'delegatedByUser', 'team']
    });
  }

  async revokeDelegation(delegationId: string, revokedBy: string): Promise<void> {
    const delegation = await this.delegationRepo.findOne({
      where: { id: delegationId }
    });

    if (!delegation) {
      throw new NotFoundException('위임을 찾을 수 없습니다.');
    }

    // 권한 확인 (위임자 또는 팀 관리자만 가능)
    await this.validateRevocationPermission(delegation, revokedBy);

    delegation.isActive = false;
    delegation.validTo = new Date();
    
    await this.delegationRepo.save(delegation);

    // 철회 알림 발송
    await this.notificationService.sendDelegationRevocationNotification(delegation);
  }

  async getUserDelegatedPermissions(userId: string, teamId: string): Promise<DelegatedPermissions | null> {
    const activeDelegation = await this.delegationRepo.findOne({
      where: {
        delegatedTo: userId,
        teamId,
        isActive: true,
        validFrom: LessThanOrEqual(new Date()),
        validTo: MoreThanOrEqual(new Date())
      }
    });

    return activeDelegation?.permissions || null;
  }

  private async validateDelegator(userId: string, teamId: string): Promise<void> {
    const member = await this.teamMemberRepo.findOne({
      where: { userId, teamId, isActive: true }
    });

    if (!member || !['leader', 'admin'].includes(member.role)) {
      throw new ForbiddenException('권한 위임 권한이 없습니다.');
    }
  }

  private async validateDelegatee(userId: string, teamId: string): Promise<void> {
    const member = await this.teamMemberRepo.findOne({
      where: { userId, teamId, isActive: true }
    });

    if (!member) {
      throw new BadRequestException('해당 팀의 멤버가 아닙니다.');
    }
  }
}
```

#### 사용량 집계 서비스
```typescript
// backend-api/src/modules/analytics/services/team-usage.service.ts
@Injectable()
export class TeamUsageService {
  constructor(
    @InjectRepository(TeamUsageStats) private statsRepo: Repository<TeamUsageStats>,
    @InjectRepository(Dashboard) private dashboardRepo: Repository<Dashboard>,
    @InjectRepository(TeamMember) private teamMemberRepo: Repository<TeamMember>
  ) {}

  @Cron('0 2 * * *') // 매일 새벽 2시 실행
  async aggregateDailyUsage(): Promise<void> {
    const yesterday = new Date();
    yesterday.setDate(yesterday.getDate() - 1);

    const teams = await this.teamMemberRepo.createQueryBuilder('tm')
      .select('DISTINCT tm.teamId')
      .getRawMany();

    for (const { teamId } of teams) {
      const stats = await this.calculateTeamStats(teamId, yesterday);
      
      await this.statsRepo.upsert(
        {
          teamId,
          date: yesterday,
          stats
        },
        ['teamId', 'date']
      );
    }
  }

  async getTeamUsageReport(teamId: string, period: 'week' | 'month' | 'quarter'): Promise<TeamUsageReport> {
    const endDate = new Date();
    const startDate = this.getStartDate(endDate, period);

    const stats = await this.statsRepo.find({
      where: {
        teamId,
        date: Between(startDate, endDate)
      },
      order: { date: 'ASC' }
    });

    return this.generateUsageReport(stats, period);
  }

  private async calculateTeamStats(teamId: string, date: Date): Promise<UsageStatistics> {
    const startOfDay = new Date(date);
    startOfDay.setHours(0, 0, 0, 0);
    
    const endOfDay = new Date(date);
    endOfDay.setHours(23, 59, 59, 999);

    // 대시보드 수
    const dashboardCount = await this.dashboardRepo.count({
      where: { teamId, isDeleted: false }
    });

    // 활성 사용자 수 (해당일에 로그인한 팀 멤버)
    const activeUsers = await this.getActiveUsersCount(teamId, startOfDay, endOfDay);

    // 총 조회수
    const totalViews = await this.getTotalViews(teamId, startOfDay, endOfDay);

    // 공유된 대시보드 수
    const sharedDashboards = await this.getSharedDashboardsCount(teamId);

    // 기타 통계들...
    const dataConnections = await this.getDataConnectionsCount(teamId);
    const storageUsed = await this.getStorageUsed(teamId);
    const queryCount = await this.getQueryCount(teamId, startOfDay, endOfDay);
    const avgResponseTime = await this.getAvgResponseTime(teamId, startOfDay, endOfDay);

    return {
      dashboardCount,
      activeUsers,
      totalViews,
      sharedDashboards,
      dataConnections,
      storageUsed,
      queryCount,
      avgResponseTime
    };
  }

  async getTeamQuota(teamId: string): Promise<TeamQuota> {
    const team = await this.teamMemberRepo.findOne({
      where: { id: teamId },
      relations: ['department', 'department.organization']
    });

    // 조직/부서별 기본 할당량 + 팀별 추가 할당량
    return this.calculateTeamQuota(team);
  }

  async checkTeamLimits(teamId: string): Promise<LimitCheckResult> {
    const quota = await this.getTeamQuota(teamId);
    const currentUsage = await this.getCurrentUsage(teamId);

    return {
      withinLimits: this.isWithinLimits(currentUsage, quota),
      warnings: this.generateWarnings(currentUsage, quota),
      limits: quota,
      usage: currentUsage
    };
  }
}
```
