// backend-api/src/modules/sharing/entities/share-link.entity.ts
@Entity('share_links')
export class ShareLink {
  @PrimaryGeneratedColumn('uuid')
  id: string;

  @Column({ length: 32, unique: true })
  token: string; // 공유 토큰

  @Column({ length: 200 })
  title: string; // 공유 제목

  @Column({ type: 'text', nullable: true })
  description: string;

  @ManyToOne(() => Dashboard, { onDelete: 'CASCADE' })
  @JoinColumn({ name: 'dashboardId' })
  dashboard: Dashboard;

  @Column({ type: 'uuid' })
  dashboardId: string;

  @ManyToOne(() => User, { onDelete: 'CASCADE' })
  @JoinColumn({ name: 'createdBy' })
  creator: User;

  @Column({ type: 'uuid' })
  createdBy: string;

  @Column({ type: 'enum', enum: ShareStatus, default: ShareStatus.ACTIVE })
  status: ShareStatus;

  @Column({ type: 'enum', enum: ShareLevel })
  shareLevel: ShareLevel;

  @Column({ type: 'json' })
  settings: ShareLinkSettings;

  @Column({ type: 'timestamp', nullable: true })
  expiresAt: Date;

  @Column({ type: 'timestamp', nullable: true })
  lastAccessedAt: Date;

  @Column({ type: 'int', default: 0 })
  accessCount: number;

  @OneToMany(() => ShareLinkAccess, access => access.shareLink)
  accesses: ShareLinkAccess[];

  @OneToMany(() => ShareLinkAudit, audit => audit.shareLink)
  audits: ShareLinkAudit[];

  @CreateDateColumn()
  createdAt: Date;

  @UpdateDateColumn()
  updatedAt: Date;
}

enum ShareStatus {
  ACTIVE = 'active',
  INACTIVE = 'inactive',
  EXPIRED = 'expired',
  REVOKED = 'revoked'
}

enum ShareLevel {
  PUBLIC = 'public',
  INTERNAL = 'internal',
  SPECIFIC_USERS = 'specific_users',
  PASSWORD_PROTECTED = 'password_protected'
}

interface ShareLinkSettings {
  allowDownload: boolean;
  allowComments: boolean;
  allowEmbedding: boolean;
  requireAuth: boolean;
  ipWhitelist: string[];
  maxViewers: number;
  watermark: boolean;
  trackingEnabled: boolean;
}

#### ShareLinkManagementService
```typescript
// backend-api/src/modules/sharing/services/share-link-management.service.ts
@Injectable()
export class ShareLinkManagementService {
  constructor(
    @InjectRepository(ShareLink) private shareLinkRepo: Repository<ShareLink>,
    @InjectRepository(ShareLinkAccess) private accessRepo: Repository<ShareLinkAccess>,
    @InjectRepository(ShareLinkAudit) private auditRepo: Repository<ShareLinkAudit>,
    private eventEmitter: EventEmitter2
  ) {}

  async getAllShareLinks(filters: ShareLinkFilters): Promise<PaginatedResult<ShareLink>> {
    const query = this.shareLinkRepo.createQueryBuilder('shareLink')
      .leftJoinAndSelect('shareLink.dashboard', 'dashboard')
      .leftJoinAndSelect('shareLink.creator', 'creator')
      .leftJoinAndSelect('dashboard.team', 'team');

    // 필터 적용
    this.applyFilters(query, filters);

    // 정렬
    query.orderBy('shareLink.createdAt', 'DESC');

    // 페이지네이션
    const [items, total] = await query
      .skip((filters.page - 1) * filters.limit)
      .take(filters.limit)
      .getManyAndCount();

    return {
      items,
      total,
      page: filters.page,
      limit: filters.limit,
      totalPages: Math.ceil(total / filters.limit)
    };
  }

  async getShareLinkStats(): Promise<ShareLinkStats> {
    const totalCount = await this.shareLinkRepo.count();
    const activeCount = await this.shareLinkRepo.count({
      where: { status: ShareStatus.ACTIVE }
    });
    
    const expiredCount = await this.shareLinkRepo.count({
      where: { status: ShareStatus.EXPIRED }
    });

    const revokedCount = await this.shareLinkRepo.count({
      where: { status: ShareStatus.REVOKED }
    });

    // 총 접근 수
    const totalAccesses = await this.accessRepo
      .createQueryBuilder('access')
      .select('SUM(shareLink.accessCount)', 'total')
      .leftJoin('access.shareLink', 'shareLink')
      .getRawOne();

    // 최근 7일 접근 수
    const recentAccesses = await this.accessRepo.count({
      where: {
        accessedAt: MoreThanOrEqual(new Date(Date.now() - 7 * 24 * 60 * 60 * 1000))
      }
    });

    return {
      totalCount,
      activeCount,
      expiredCount,
      revokedCount,
      totalAccesses: parseInt(totalAccesses?.total || '0'),
      recentAccesses
    };
  }

  async bulkUpdateStatus(
    shareLinkIds: string[],
    status: ShareStatus,
    performedBy: string,
    reason?: string
  ): Promise<void> {
    await this.shareLinkRepo.update(
      { id: In(shareLinkIds) },
      { status, updatedAt: new Date() }
    );

    // 감사 로그 생성
    const auditLogs = shareLinkIds.map(shareLinkId => ({
      shareLinkId,
      performedBy,
      action: this.getAuditActionFromStatus(status),
      reason,
      ipAddress: '127.0.0.1' // 실제로는 요청에서 추출
    }));

    await this.auditRepo.save(auditLogs);

    // 이벤트 발송
    this.eventEmitter.emit('shareLink.bulkStatusUpdate', {
      shareLinkIds,
      status,
      performedBy
    });
  }

  async revokeShareLink(shareLinkId: string, performedBy: string, reason: string): Promise<void> {
    const shareLink = await this.shareLinkRepo.findOne({
      where: { id: shareLinkId }
    });

    if (!shareLink) {
      throw new NotFoundException('공유 링크를 찾을 수 없습니다.');
    }

    const oldStatus = shareLink.status;
    shareLink.status = ShareStatus.REVOKED;
    
    await this.shareLinkRepo.save(shareLink);

    // 감사 로그
    await this.auditRepo.save({
      shareLinkId,
      performedBy,
      action: AuditAction.REVOKED,
      oldValues: { status: oldStatus },
      newValues: { status: ShareStatus.REVOKED },
      reason,
      ipAddress: '127.0.0.1'
    });

    // 알림 발송 (공유 링크 생성자에게)
    this.eventEmitter.emit('shareLink.revoked', {
      shareLink,
      performedBy,
      reason
    });
  }

  async extendExpiryDate(
    shareLinkId: string,
    newExpiryDate: Date,
    performedBy: string
  ): Promise<void> {
    const shareLink = await this.shareLinkRepo.findOne({
      where: { id: shareLinkId }
    });

    if (!shareLink) {
      throw new NotFoundException('공유 링크를 찾을 수 없습니다.');
    }

    const oldExpiryDate = shareLink.expiresAt;
    shareLink.expiresAt = newExpiryDate;
    
    // 만료된 링크라면 다시 활성화
    if (shareLink.status === ShareStatus.EXPIRED) {
      shareLink.status = ShareStatus.ACTIVE;
    }

    await this.shareLinkRepo.save(shareLink);

    // 감사 로그
    await this.auditRepo.save({
      shareLinkId,
      performedBy,
      action: AuditAction.UPDATED,
      oldValues: { expiresAt: oldExpiryDate },
      newValues: { expiresAt: newExpiryDate },
      ipAddress: '127.0.0.1'
    });
  }

  @Cron('0 1 * * *') // 매일 새벽 1시 실행
  async checkExpiredLinks(): Promise<void> {
    const expiredLinks = await this.shareLinkRepo.find({
      where: {
        status: ShareStatus.ACTIVE,
        expiresAt: LessThan(new Date())
      }
    });

    if (expiredLinks.length > 0) {
      await this.shareLinkRepo.update(
        { id: In(expiredLinks.map(link => link.id)) },
        { status: ShareStatus.EXPIRED }
      );

      // 만료 알림 발송
      this.eventEmitter.emit('shareLink.expired', { links: expiredLinks });
    }
  }

  private applyFilters(query: SelectQueryBuilder<ShareLink>, filters: ShareLinkFilters): void {
    if (filters.status) {
      query.andWhere('shareLink.status = :status', { status: filters.status });
    }

    if (filters.shareLevel) {
      query.andWhere('shareLink.shareLevel = :shareLevel', { shareLevel: filters.shareLevel });
    }

    if (filters.createdBy) {
      query.andWhere('shareLink.createdBy = :createdBy', { createdBy: filters.createdBy });
    }

    if (filters.teamId) {
      query.andWhere('team.id = :teamId', { teamId: filters.teamId });
    }

    if (filters.searchTerm) {
      query.andWhere(
        '(shareLink.title ILIKE :search OR shareLink.description ILIKE :search)',
        { search: `%${filters.searchTerm}%` }
      );
    }

    if (filters.dateFrom) {
      query.andWhere('shareLink.createdAt >= :dateFrom', { dateFrom: filters.dateFrom });
    }

    if (filters.dateTo) {
      query.andWhere('shareLink.createdAt <= :dateTo', { dateTo: filters.dateTo });
    }
  }
}
```
