```typescript
  readonly?: boolean;

  readonly = false

  const [expandedNodes, setExpandedNodes] = useState<Set<string>>(new Set());
  const [selectedNode, setSelectedNode] = useState<string | null>(null);

  const handleDragEnd = useCallback((result: any) => {

    const changes = calculateStructureChanges(result);
    onStructureChange(changes);
  }, [onStructureChange]);

  const renderDepartmentNode = (dept: Department, level: number) => (
    <DepartmentNode
      key={dept.id}
      department={dept}

      isExpanded={expandedNodes.has(dept.id)}

          key={team.id}

          level={level + 1}

          readonly={readonly}

                      ref={provided.innerRef}

    </div>
```

```typescript
// frontend-web/src/components/admin/organization/DepartmentNode.tsx
import React from 'react';
import { ChevronRight, ChevronDown, Users, Settings } from 'lucide-react';
import { Department } from '@/types/admin';

interface DepartmentNodeProps {
  department: Department;
  level: number;
  isExpanded: boolean;
  isSelected: boolean;
  onToggle: () => void;
  onSelect: () => void;
  readonly: boolean;
  children?: React.ReactNode;
}

export const DepartmentNode: React.FC<DepartmentNodeProps> = ({
  department,
  level,
  isExpanded,
  isSelected,
  onToggle,
  onSelect,
  readonly,
  children
}) => {
  const hasChildren = department.children?.length > 0 || department.teams?.length > 0;

  return (
    <div className={`department-node level-${level}`}>
      <div
        className={`node-header ${isSelected ? 'selected' : ''} ${!department.isActive ? 'inactive' : ''}`}
        onClick={onSelect}
        style={{ paddingLeft: `${level * 24 + 8}px` }}
      >
        <button
          className="expand-button"
          onClick={(e) => {
            e.stopPropagation();
            onToggle();
          }}
          disabled={!hasChildren}
        >
          {hasChildren ? (
            isExpanded ? <ChevronDown size={16} /> : <ChevronRight size={16} />
          ) : (
            <div className="spacer" />
          )}
        </button>

        <div className="node-icon">
          <Users size={18} className="department-icon" />
        </div>

        <div className="node-content">
          <div className="node-title">
            <span className="department-name">{department.name}</span>
            <span className="department-code">({department.code})</span>
          </div>
          <div className="node-meta">
            <span className="member-count">
              {department.teams?.reduce((sum, team) => sum + (team.memberCount || 0), 0)} 명
            </span>
            <span className="team-count">
              {department.teams?.length || 0} 팀
            </span>
          </div>
        </div>

        {!readonly && (
          <div className="node-actions">
            <button
              className="action-button"
              onClick={(e) => {
                e.stopPropagation();
                // 부서 설정 모달 열기
              }}
            >
              <Settings size={14} />
            </button>
          </div>
        )}
      </div>

      {isExpanded && children && (
        <div className="node-children">
          {children}
        </div>
      )}
    </div>
  );
};
```

```typescript
// frontend-web/src/components/admin/team/TeamManagementModal.tsx
import React, { useState, useEffect } from 'react';
import { Modal, Form, Input, Select, Button, Alert } from 'antd';
import { Team, Department, CreateTeamRequest, UpdateTeamRequest } from '@/types/admin';

interface TeamManagementModalProps {
  visible: boolean;
  team?: Team | null; // null이면 생성, 값이 있으면 수정
  departments: Department[];
  onSubmit: (data: CreateTeamRequest | UpdateTeamRequest) => Promise<void>;
  onCancel: () => void;
}

export const TeamManagementModal: React.FC<TeamManagementModalProps> = ({
  visible,
  team,
  departments,
  onSubmit,
  onCancel
}) => {
  const [form] = Form.useForm();
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const isEdit = !!team;

  useEffect(() => {
    if (visible) {
      if (team) {
        form.setFieldsValue({
          name: team.name,
          code: team.code,
          description: team.description,
          departmentId: team.departmentId,
          permissions: team.permissions
        });
      } else {
        form.resetFields();
      }
      setError(null);
    }
  }, [visible, team, form]);

  const handleSubmit = async () => {
    try {
      setLoading(true);
      setError(null);

      const values = await form.validateFields();
      await onSubmit(values);
      
      onCancel(); // 성공시 모달 닫기
    } catch (err: any) {
      if (err.errorFields) {
        // 폼 검증 오류는 자동으로 표시됨
        return;
      }
      setError(err.message || '오류가 발생했습니다.');
    } finally {
      setLoading(false);
    }
  };

  return (
    <Modal
      title={isEdit ? '팀 수정' : '새 팀 생성'}
      open={visible}
      onCancel={onCancel}
      footer={[
        <Button key="cancel" onClick={onCancel}>
          취소
        </Button>,
        <Button
          key="submit"
          type="primary"
          loading={loading}
          onClick={handleSubmit}
        >
          {isEdit ? '수정' : '생성'}
        </Button>
      ]}
      destroyOnClose
    >
      {error && (
        <Alert
          message={error}
          type="error"
          closable
          onClose={() => setError(null)}
          style={{ marginBottom: 16 }}
        />
      )}

      <Form
        form={form}
        layout="vertical"
        requiredMark={false}
      >
        <Form.Item
          name="name"
          label="팀 이름"
          rules={[
            { required: true, message: '팀 이름을 입력해주세요.' },
            { max: 100, message: '팀 이름은 100자 이하로 입력해주세요.' }
          ]}
        >
          <Input placeholder="예: 프론트엔드 개발팀" />
        </Form.Item>

        <Form.Item
          name="code"
          label="팀 코드"
          rules={[
            { required: true, message: '팀 코드를 입력해주세요.' },
            { max: 10, message: '팀 코드는 10자 이하로 입력해주세요.' },
            { pattern: /^[A-Z0-9]+$/, message: '영문 대문자와 숫자만 사용 가능합니다.' }
          ]}
        >
          <Input placeholder="예: FE01" style={{ textTransform: 'uppercase' }} />
        </Form.Item>

        <Form.Item
          name="departmentId"
          label="소속 부서"
          rules={[{ required: true, message: '소속 부서를 선택해주세요.' }]}
        >
          <Select
            placeholder="부서를 선택하세요"
            showSearch
            optionFilterProp="children"
          >
            {departments.map(dept => (
              <Select.Option key={dept.id} value={dept.id}>
                {dept.name} ({dept.code})
              </Select.Option>
            ))}
          </Select>
        </Form.Item>

        <Form.Item
          name="description"
          label="팀 설명"
        >
          <Input.TextArea
            rows={3}
            placeholder="팀의 역할과 책임에 대해 설명해주세요"
            maxLength={500}
          />
        </Form.Item>

        <TeamPermissionsForm />
      </Form>
    </Modal>
  );
};
```

#### TeamMemberManagement 컴포넌트
```typescript
// frontend-web/src/components/admin/team/TeamMemberManagement.tsx
import React, { useState } from 'react';
import { Table, Button, Modal, Select, Tag, Avatar, Popconfirm } from 'antd';
import { UserPlus, Crown, Shield, User, Trash2 } from 'lucide-react';
import { Team, User, TeamMember, TeamRole } from '@/types/admin';

interface TeamMemberManagementProps {
  team: Team;
  members: TeamMember[];
  availableUsers: User[];
  onAddMember: (userId: string, role: TeamRole) => Promise<void>;
  onUpdateMemberRole: (memberId: string, role: TeamRole) => Promise<void>;
  onRemoveMember: (memberId: string) => Promise<void>;
}

export const TeamMemberManagement: React.FC<TeamMemberManagementProps> = ({
  team,
  members,
  availableUsers,
  onAddMember,
  onUpdateMemberRole,
  onRemoveMember
}) => {
  const [addMemberModalVisible, setAddMemberModalVisible] = useState(false);
  const [selectedUserId, setSelectedUserId] = useState<string>();
  const [selectedRole, setSelectedRole] = useState<TeamRole>(TeamRole.MEMBER);

  const getRoleIcon = (role: TeamRole) => {
    switch (role) {
      case TeamRole.ADMIN:
        return <Crown size={16} className="text-yellow-500" />;
      case TeamRole.LEADER:
        return <Shield size={16} className="text-blue-500" />;
      default:
        return <User size={16} className="text-gray-500" />;
    }
  };

  const getRoleColor = (role: TeamRole) => {
    switch (role) {
      case TeamRole.ADMIN:
        return 'gold';
      case TeamRole.LEADER:
        return 'blue';
      default:
        return 'default';
    }
  };

  const columns = [
    {
      title: '멤버',
      dataIndex: 'user',
      key: 'user',
      render: (user: User) => (
        <div className="flex items-center space-x-3">
          <Avatar src={user.profileImage} size={32}>
            {user.name.charAt(0)}
          </Avatar>
          <div>
            <div className="font-medium">{user.name}</div>
            <div className="text-sm text-gray-500">{user.email}</div>
          </div>
        </div>
      )
    },
    {
      title: '역할',
      dataIndex: 'role',
      key: 'role',
      render: (role: TeamRole, record: TeamMember) => (
        <Select
          value={role}
          style={{ width: 120 }}
          onChange={(newRole) => onUpdateMemberRole(record.id, newRole)}
          size="small"
        >
          <Select.Option value={TeamRole.MEMBER}>
            <div className="flex items-center space-x-2">
              {getRoleIcon(TeamRole.MEMBER)}
              <span>멤버</span>
            </div>
          </Select.Option>
          <Select.Option value={TeamRole.LEADER}>
            <div className="flex items-center space-x-2">
              {getRoleIcon(TeamRole.LEADER)}
              <span>리더</span>
            </div>
          </Select.Option>
          <Select.Option value={TeamRole.ADMIN}>
            <div className="flex items-center space-x-2">
              {getRoleIcon(TeamRole.ADMIN)}
              <span>관리자</span>
            </div>
          </Select.Option>
        </Select>
      )
    },
    {
      title: '참여일',
      dataIndex: 'joinedAt',
      key: 'joinedAt',
      render: (date: string) => new Date(date).toLocaleDateString('ko-KR')
    },
    {
      title: '상태',
      dataIndex: 'isActive',
      key: 'isActive',
      render: (isActive: boolean) => (
        <Tag color={isActive ? 'green' : 'red'}>
          {isActive ? '활성' : '비활성'}
        </Tag>
      )
    },
    {
      title: '작업',
      key: 'actions',
      render: (record: TeamMember) => (
        <Popconfirm
          title="멤버를 팀에서 제거하시겠습니까?"
          onConfirm={() => onRemoveMember(record.id)}
          okText="제거"
          cancelText="취소"
        >
          <Button
            type="text"
            size="small"
            icon={<Trash2 size={16} />}
            danger
          />
        </Popconfirm>
      )
    }
  ];

  const handleAddMember = async () => {
    if (!selectedUserId) return;
    
    await onAddMember(selectedUserId, selectedRole);
    setAddMemberModalVisible(false);
    setSelectedUserId(undefined);
    setSelectedRole(TeamRole.MEMBER);
  };

  return (
    <div className="team-member-management">
      <div className="flex justify-between items-center mb-4">
        <h3 className="text-lg font-medium">팀 멤버 ({members.length}명)</h3>
        <Button
          type="primary"
          icon={<UserPlus size={16} />}
          onClick={() => setAddMemberModalVisible(true)}
        >
          멤버 추가
        </Button>
      </div>

      <Table
        columns={columns}
        dataSource={members}
        rowKey="id"
        pagination={false}
        size="middle"
      />

      <Modal
        title="팀 멤버 추가"
        open={addMemberModalVisible}
        onOk={handleAddMember}
        onCancel={() => setAddMemberModalVisible(false)}
        okText="추가"
        cancelText="취소"
        okButtonProps={{ disabled: !selectedUserId }}
      >
        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium mb-2">사용자 선택</label>
            <Select
              placeholder="추가할 사용자를 선택하세요"
              style={{ width: '100%' }}
              value={selectedUserId}
              onChange={setSelectedUserId}
              showSearch
              optionFilterProp="children"
            >
              {availableUsers.map(user => (
                <Select.Option key={user.id} value={user.id}>
                  <div className="flex items-center space-x-2">
                    <Avatar src={user.profileImage} size={24}>
                      {user.name.charAt(0)}
                    </Avatar>
                    <span>{user.name} ({user.email})</span>
                  </div>
                </Select.Option>
              ))}
            </Select>
          </div>

          <div>
            <label className="block text-sm font-medium mb-2">역할</label>
            <Select
              value={selectedRole}
              onChange={setSelectedRole}
              style={{ width: '100%' }}
            >
              <Select.Option value={TeamRole.MEMBER}>멤버</Select.Option>
              <Select.Option value={TeamRole.LEADER}>리더</Select.Option>
              <Select.Option value={TeamRole.ADMIN}>관리자</Select.Option>
            </Select>
          </div>
        </div>
      </Modal>
    </div>
  );
};
```

#### AdminOrganizationPage 컴포넌트
```typescript
// frontend-web/src/pages/admin/organization/AdminOrganizationPage.tsx
import React, { useState, useEffect } from 'react';
import { Card, Tabs, Button, message } from 'antd';
import { Building, Users, Settings } from 'lucide-react';
import { OrganizationTreeView } from '@/components/admin/organization/OrganizationTreeView';
import { TeamManagementModal } from '@/components/admin/team/TeamManagementModal';
import { TeamMemberManagement } from '@/components/admin/team/TeamMemberManagement';
import { useAdminOrganization } from '@/hooks/admin/useAdminOrganization';

export const AdminOrganizationPage: React.FC = () => {
  const {
    organization,
    departments,
    teams,
    users,
    loading,
    createTeam,
    updateTeam,
    deleteTeam,
    addTeamMember,
    updateTeamMemberRole,
    removeTeamMember,
    updateOrganizationStructure
  } = useAdminOrganization();

  const [selectedTeam, setSelectedTeam] = useState<Team | null>(null);
  const [teamModalVisible, setTeamModalVisible] = useState(false);
  const [editingTeam, setEditingTeam] = useState<Team | null>(null);

  const handleStructureChange = async (changes: StructureChange[]) => {
    try {
      await updateOrganizationStructure(changes);
      message.success('조직 구조가 업데이트되었습니다.');
    } catch (error) {
      message.error('조직 구조 업데이트에 실패했습니다.');
    }
  };

  const handleTeamSubmit = async (data: CreateTeamRequest | UpdateTeamRequest) => {
    try {
      if (editingTeam) {
        await updateTeam(editingTeam.id, data as UpdateTeamRequest);
        message.success('팀이 수정되었습니다.');
      } else {
        await createTeam(data as CreateTeamRequest);
        message.success('팀이 생성되었습니다.');
      }
    } catch (error) {
      throw new Error('팀 처리에 실패했습니다.');
    }
  };

  return (
    <div className="admin-organization-page">
      <div className="page-header mb-6">
        <h1 className="text-2xl font-bold flex items-center space-x-2">
          <Building size={24} />
          <span>조직 관리</span>
        </h1>
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
        {/* 조직 구조 트리 */}
        <div className="lg:col-span-2">
          <Card
            title="조직 구조"
            extra={
              <Button
                type="primary"
                onClick={() => {
                  setEditingTeam(null);
                  setTeamModalVisible(true);
                }}
              >
                팀 추가
              </Button>
            }
          >
            {organization && (
              <OrganizationTreeView
                organization={organization}
                onStructureChange={handleStructureChange}
              />
            )}
          </Card>
        </div>

        {/* 선택된 팀 상세 */}
        <div>
          {selectedTeam ? (
            <Card
              title={`${selectedTeam.name} 팀`}
              extra={
                <Button
                  size="small"
                  onClick={() => {
                    setEditingTeam(selectedTeam);
                    setTeamModalVisible(true);
                  }}
                >
                  <Settings size={16} />
                </Button>
              }
            >
              <Tabs
                items={[
                  {
                    key: 'members',
                    label: '멤버',
                    icon: <Users size={16} />,
                    children: (
                      <TeamMemberManagement
                        team={selectedTeam}
                        members={selectedTeam.members || []}
                        availableUsers={users}
                        onAddMember={addTeamMember}
                        onUpdateMemberRole={updateTeamMemberRole}
                        onRemoveMember={removeTeamMember}
                      />
                    )
                  }
                ]}
              />
            </Card>
          ) : (
            <Card>
              <div className="text-center text-gray-500 py-8">
                팀을 선택하여 상세 정보를 확인하세요
              </div>
            </Card>
          )}
        </div>
      </div>

      <TeamManagementModal
        visible={teamModalVisible}
        team={editingTeam}
        departments={departments}
        onSubmit={handleTeamSubmit}
        onCancel={() => {
          setTeamModalVisible(false);
          setEditingTeam(null);
        }}
      />
    </div>
  );
};
```
