### 2. 알림 서비스 구현

#### NotificationService
```typescript
// backend-api/src/modules/notification/services/notification.service.ts
@Injectable()
export class NotificationService {
  constructor(
    @InjectRepository(Notification) private notificationRepo: Repository<Notification>,
    @InjectRepository(NotificationDelivery) private deliveryRepo: Repository<NotificationDelivery>,
    @InjectRepository(NotificationPreference) private preferenceRepo: Repository<NotificationPreference>,
    private emailService: EmailService,
    private webSocketGateway: NotificationGateway,
    private pushService: PushNotificationService,
    private templateService: NotificationTemplateService,
    private eventEmitter: EventEmitter2
  ) {}

  async createNotification(createDto: CreateNotificationDto): Promise<Notification> {
    const notification = this.notificationRepo.create(createDto);
    const saved = await this.notificationRepo.save(notification);

    // 사용자 설정에 따른 채널 결정
    const userPreferences = await this.getUserPreferences(createDto.userId, createDto.type);
    const enabledChannels = this.getEnabledChannels(userPreferences, createDto.priority);

    // 각 채널로 알림 전송
    await this.deliverNotification(saved, enabledChannels);

    return saved;
  }

  async sendBulkNotifications(notifications: CreateNotificationDto[]): Promise<void> {
    // 배치 처리로 대량 알림 전송
    const batchSize = 100;
    const batches = this.chunkArray(notifications, batchSize);

    for (const batch of batches) {
      await Promise.all(
        batch.map(notificationDto => this.createNotification(notificationDto))
      );
    }
  }

  async deliverNotification(notification: Notification, channels: NotificationChannel[]): Promise<void> {
    const deliveryPromises = channels.map(channel => 
      this.deliverToChannel(notification, channel)
    );

    await Promise.allSettled(deliveryPromises);
  }

  private async deliverToChannel(notification: Notification, channel: NotificationChannel): Promise<void> {
    const delivery = await this.createDeliveryRecord(notification.id, channel);

    try {
      switch (channel) {
        case NotificationChannel.IN_APP:
          await this.deliverInApp(notification);
          break;
        case NotificationChannel.EMAIL:
          await this.deliverEmail(notification, delivery);
          break;
        case NotificationChannel.PUSH:
          await this.deliverPush(notification, delivery);
          break;
        case NotificationChannel.SMS:
          await this.deliverSMS(notification, delivery);
          break;
      }

      delivery.status = DeliveryStatus.SENT;
      delivery.sentAt = new Date();
    } catch (error) {
      delivery.status = DeliveryStatus.FAILED;
      delivery.errorMessage = error.message;
      delivery.retryCount++;
    }

    await this.deliveryRepo.save(delivery);
  }

  private async deliverInApp(notification: Notification): Promise<void> {
    // WebSocket을 통한 실시간 알림 전송
    await this.webSocketGateway.sendNotificationToUser(
      notification.userId,
      {
        id: notification.id,
        title: notification.title,
        message: notification.message,
        type: notification.type,
        priority: notification.priority,
        data: notification.data,
        createdAt: notification.createdAt
      }
    );
  }

  private async deliverEmail(notification: Notification, delivery: NotificationDelivery): Promise<void> {
    const user = await this.getUserById(notification.userId);
    if (!user?.email) {
      throw new Error('User email not found');
    }

    // 템플릿 기반 이메일 생성
    const emailContent = await this.templateService.generateEmailTemplate(
      notification.type,
      {
        title: notification.title,
        message: notification.message,
        userName: user.name,
        actionUrl: notification.data?.actionUrl,
        actionText: notification.data?.actionText
      }
    );

    delivery.recipient = user.email;
    delivery.deliveryData = {
      subject: notification.title,
      template: emailContent.template,
      variables: emailContent.variables
    };

    await this.emailService.sendEmail({
      to: user.email,
      subject: notification.title,
      html: emailContent.html,
      trackingId: delivery.id
    });
  }

  async markAsRead(notificationId: string, userId: string): Promise<void> {
    const notification = await this.notificationRepo.findOne({
      where: { id: notificationId, userId }
    });

    if (!notification) {
      throw new NotFoundException('알림을 찾을 수 없습니다.');
    }

    if (notification.status === NotificationStatus.UNREAD) {
      notification.status = NotificationStatus.READ;
      notification.readAt = new Date();
      await this.notificationRepo.save(notification);

      // 읽음 이벤트 발행
      this.eventEmitter.emit('notification.read', {
        notificationId,
        userId,
        readAt: notification.readAt
      });
    }
  }

  async markAllAsRead(userId: string): Promise<void> {
    await this.notificationRepo.update(
      { 
        userId, 
        status: NotificationStatus.UNREAD 
      },
      { 
        status: NotificationStatus.READ,
        readAt: new Date()
      }
    );

    // WebSocket으로 클라이언트에 알림
    this.webSocketGateway.sendNotificationUpdate(userId, { allRead: true });
  }

  async getUserNotifications(
    userId: string, 
    filters: NotificationFilters
  ): Promise<PaginatedResult<Notification>> {
    const query = this.notificationRepo.createQueryBuilder('notification')
      .where('notification.userId = :userId', { userId })
      .andWhere('(notification.expiresAt IS NULL OR notification.expiresAt > :now)', { now: new Date() });

    // 필터 적용
    if (filters.status) {
      query.andWhere('notification.status = :status', { status: filters.status });
    }

    if (filters.type) {
      query.andWhere('notification.type = :type', { type: filters.type });
    }

    if (filters.priority) {
      query.andWhere('notification.priority = :priority', { priority: filters.priority });
    }

    // 정렬: 읽지 않음 > 우선순위 > 생성일
    query.orderBy('notification.status', 'ASC')
      .addOrderBy('notification.priority', 'DESC')
      .addOrderBy('notification.createdAt', 'DESC');

    const [items, total] = await query
      .skip((filters.page - 1) * filters.limit)
      .take(filters.limit)
      .getManyAndCount();

    return {
      items,
      total,
      page: filters.page,
      limit: filters.limit,
      totalPages: Math.ceil(total / filters.limit)
    };
  }

  async getUnreadCount(userId: string): Promise<number> {
    return this.notificationRepo.count({
      where: {
        userId,
        status: NotificationStatus.UNREAD,
        expiresAt: IsNull() || MoreThan(new Date())
      }
    });
  }

  async updateUserPreferences(
    userId: string, 
    preferences: UpdatePreferenceDto[]
  ): Promise<void> {
    // 기존 설정 삭제
    await this.preferenceRepo.delete({ userId });

    // 새 설정 저장
    const newPreferences = preferences.map(pref => 
      this.preferenceRepo.create({ ...pref, userId })
    );

    await this.preferenceRepo.save(newPreferences);
  }

  async getUserPreferences(userId: string, type?: NotificationType): Promise<NotificationPreference[]> {
    const query = this.preferenceRepo.createQueryBuilder('pref')
      .where('pref.userId = :userId', { userId });

    if (type) {
      query.andWhere('pref.notificationType = :type', { type });
    }

    return query.getMany();
  }

  @Cron('*/5 * * * *') // 5분마다 실행
  async retryFailedDeliveries(): Promise<void> {
    const failedDeliveries = await this.deliveryRepo.find({
      where: {
        status: DeliveryStatus.FAILED,
        retryCount: LessThan(3) // 최대 3회 재시도
      },
      relations: ['notification']
    });

    for (const delivery of failedDeliveries) {
      try {
        await this.deliverToChannel(delivery.notification, delivery.channel);
      } catch (error) {
        console.error(`Retry failed for delivery ${delivery.id}:`, error);
      }
    }
  }

  @Cron('0 2 * * *') // 매일 새벽 2시 실행
  async cleanupExpiredNotifications(): Promise<void> {
    const expiredDate = new Date();
    expiredDate.setDate(expiredDate.getDate() - 30); // 30일 이전

    await this.notificationRepo.delete({
      expiresAt: LessThan(new Date()),
      status: In([NotificationStatus.READ, NotificationStatus.DISMISSED])
    });

    // 오래된 알림도 정리 (읽은 알림 중 30일 이전)
    await this.notificationRepo.delete({
      status: NotificationStatus.READ,
      readAt: LessThan(expiredDate)
    });
  }

  private getEnabledChannels(
    preferences: NotificationPreference[], 
    priority: NotificationPriority
  ): NotificationChannel[] {
    const channels: NotificationChannel[] = [];
    
    // 긴급 알림은 모든 채널로 전송
    if (priority === NotificationPriority.URGENT) {
      return [
        NotificationChannel.IN_APP,
        NotificationChannel.EMAIL,
        NotificationChannel.PUSH
      ];
    }

    // 사용자 설정에 따른 채널 선택
    preferences.forEach(pref => {
      if (pref.isEnabled) {
        if (pref.channelSettings.inApp) channels.push(NotificationChannel.IN_APP);
        if (pref.channelSettings.email) channels.push(NotificationChannel.EMAIL);
        if (pref.channelSettings.push) channels.push(NotificationChannel.PUSH);
        if (pref.channelSettings.sms) channels.push(NotificationChannel.SMS);
      }
    });

    // 기본값: 인앱 알림은 항상 포함
    if (!channels.includes(NotificationChannel.IN_APP)) {
      channels.push(NotificationChannel.IN_APP);
    }

    return [...new Set(channels)]; // 중복 제거
  }
}
```

#### NotificationGateway
```typescript
// backend-api/src/modules/notification/gateways/notification.gateway.ts
@WebSocketGateway({
  namespace: 'notifications',
  cors: {
    origin: process.env.FRONTEND_URL,
    credentials: true
  }
})
export class NotificationGateway implements OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer()
  server: Server;

  private connectedUsers = new Map<string, string[]>(); // userId -> socketIds[]

  constructor(
    private jwtService: JwtService,
    private notificationService: NotificationService
  ) {}

  async handleConnection(client: Socket) {
    try {
      const token = client.handshake.auth.token;
      const payload = this.jwtService.verify(token);
      const userId = payload.sub;

      // 사용자별 소켓 연결 관리
      const userSockets = this.connectedUsers.get(userId) || [];
      userSockets.push(client.id);
      this.connectedUsers.set(userId, userSockets);

      // 클라이언트에 연결 확인
      client.emit('connected', { userId });

      // 읽지 않은 알림 수 전송
      const unreadCount = await this.notificationService.getUnreadCount(userId);
      client.emit('unreadCount', { count: unreadCount });

      console.log(`User ${userId} connected with socket ${client.id}`);
    } catch (error) {
      console.error('Socket connection error:', error);
      client.disconnect();
    }
  }

  handleDisconnect(client: Socket) {
    // 연결된 사용자 목록에서 제거
    this.connectedUsers.forEach((sockets, userId) => {
      const index = sockets.indexOf(client.id);
      if (index !== -1) {
        sockets.splice(index, 1);
        if (sockets.length === 0) {
          this.connectedUsers.delete(userId);
        } else {
          this.connectedUsers.set(userId, sockets);
        }
        console.log(`User ${userId} disconnected socket ${client.id}`);
      }
    });
  }

  @SubscribeMessage('markAsRead')
  async handleMarkAsRead(
    client: Socket,
    payload: { notificationId: string }
  ) {
    try {
      const token = client.handshake.auth.token;
      const jwtPayload = this.jwtService.verify(token);
      const userId = jwtPayload.sub;

      await this.notificationService.markAsRead(payload.notificationId, userId);
      
      // 읽음 처리 완료 알림
      client.emit('notificationRead', { 
        notificationId: payload.notificationId 
      });

      // 업데이트된 읽지 않은 알림 수 전송
      const unreadCount = await this.notificationService.getUnreadCount(userId);
      client.emit('unreadCount', { count: unreadCount });
    } catch (error) {
      client.emit('error', { message: error.message });
    }
  }

  @SubscribeMessage('markAllAsRead')
  async handleMarkAllAsRead(client: Socket) {
    try {
      const token = client.handshake.auth.token;
      const jwtPayload = this.jwtService.verify(token);
      const userId = jwtPayload.sub;

      await this.notificationService.markAllAsRead(userId);
      
      client.emit('allNotificationsRead');
      client.emit('unreadCount', { count: 0 });
    } catch (error) {
      client.emit('error', { message: error.message });
    }
  }

  // 특정 사용자에게 알림 전송
  async sendNotificationToUser(userId: string, notification: any): Promise<void> {
    const userSockets = this.connectedUsers.get(userId);
    if (userSockets && userSockets.length > 0) {
      userSockets.forEach(socketId => {
        this.server.to(socketId).emit('newNotification', notification);
      });

      // 읽지 않은 알림 수 업데이트
      const unreadCount = await this.notificationService.getUnreadCount(userId);
      userSockets.forEach(socketId => {
        this.server.to(socketId).emit('unreadCount', { count: unreadCount });
      });
    }
  }

  // 알림 업데이트 전송
  sendNotificationUpdate(userId: string, update: any): void {
    const userSockets = this.connectedUsers.get(userId);
    if (userSockets && userSockets.length > 0) {
      userSockets.forEach(socketId => {
        this.server.to(socketId).emit('notificationUpdate', update);
      });
    }
  }

  // 브로드캐스트 알림 (모든 연결된 사용자)
  broadcastNotification(notification: any): void {
    this.server.emit('broadcastNotification', notification);
  }
}
```

```typescript
    if (priority === 'urgent') {
      return `🚨 ${emoji}`;
    } else if (priority === 'high') {
      return `⚡ ${emoji}`;
    }
    
    return emoji;
  };

  const formatTimeAgo = (date: string) => {
    const now = new Date();
    const notificationDate = new Date(date);
    const diffInSeconds = Math.floor((now.getTime() - notificationDate.getTime()) / 1000);

    if (diffInSeconds < 60) return '방금 전';
    if (diffInSeconds < 3600) return `${Math.floor(diffInSeconds / 60)}분 전`;
    if (diffInSeconds < 86400) return `${Math.floor(diffInSeconds / 3600)}시간 전`;
    return `${Math.floor(diffInSeconds / 86400)}일 전`;
  };

  const notificationContent = (
    <div className="notification-center" style={{ width: 400, maxHeight: 500 }}>
      <div className="notification-header p-3 border-b flex justify-between items-center">
        <div className="flex items-center space-x-2">
          <Typography.Text strong>알림</Typography.Text>
          <div className={`w-2 h-2 rounded-full ${isConnected ? 'bg-green-500' : 'bg-red-500'}`} />
        </div>
        <div className="flex space-x-2">
          {unreadCount > 0 && (
            <Button
              size="small"
              type="text"
              icon={<CheckCheck size={16} />}
              onClick={markAllAsRead}
            >
              모두 읽음
            </Button>
          )}
          <Button
            size="small"
            type="text"
            icon={<Settings size={16} />}
            onClick={() => {
              setVisible(false);
              // 알림 설정 페이지로 이동
            }}
          />
        </div>
      </div>

      <div className="notification-list" style={{ maxHeight: 400, overflowY: 'auto' }}>
        {loading ? (
          <div className="p-4 text-center">
            <Spin />
          </div>
        ) : notifications.length === 0 ? (
          <Empty 
            description="새로운 알림이 없습니다"
            image={Empty.PRESENTED_IMAGE_SIMPLE}
            className="py-8"
          />
        ) : (
          <List
            dataSource={notifications}
            renderItem={(notification) => (
              <List.Item
                className={`notification-item cursor-pointer hover:bg-gray-50 transition-colors ${
                  notification.status === 'unread' ? 'bg-blue-50 border-l-4 border-l-blue-500' : ''
                }`}
                onClick={() => handleNotificationClick(notification)}
              >
                <div className="flex items-start space-x-3 w-full p-2">
                  <div className="text-xl">
                    {getNotificationIcon(notification.type, notification.priority)}
                  </div>
                  
                  <div className="flex-1 min-w-0">
                    <div className="flex justify-between items-start">
                      <Typography.Text 
                        strong={notification.status === 'unread'}
                        className="text-sm"
                      >
                        {notification.title}
                      </Typography.Text>
                      <Typography.Text 
                        type="secondary" 
                        className="text-xs whitespace-nowrap ml-2"
                      >
                        {formatTimeAgo(notification.createdAt)}
                      </Typography.Text>
                    </div>
                    
                    <Typography.Paragraph 
                      className="text-sm text-gray-600 mb-0"
                      ellipsis={{ rows: 2 }}
                    >
                      {notification.message}
                    </Typography.Paragraph>

                    {notification.data?.actionText && (
                      <Button 
                        size="small" 
                        type="link" 
                        className="p-0 h-auto text-xs"
                      >
                        {notification.data.actionText}
                      </Button>
                    )}
                  </div>

                  {notification.status === 'unread' && (
                    <div className="w-2 h-2 bg-blue-500 rounded-full mt-1 flex-shrink-0" />
                  )}
                </div>
              </List.Item>
            )}
          />
        )}
      </div>
    </div>
  );

  return (
    <Popover
      content={notificationContent}
      trigger="click"
      placement="bottomRight"
      visible={visible}
      onVisibleChange={setVisible}
      overlayClassName="notification-popover"
    >
      <Badge count={unreadCount} size="small" offset={[-2, 2]}>
        <Button
          type="text"
          icon={<Bell size={20} />}
          className="notification-bell"
        />
      </Badge>
    </Popover>
  );
};
```
