@Entity('organizations')
export class Organization {
  @PrimaryGeneratedColumn('uuid')
  id: string;

  @Column({ length: 100 })
  name: string;

  @Column({ length: 10, unique: true })
  code: string; // 조직 코드 (예: COMP01)

  @Column({ type: 'json', nullable: true })

  @Column({ type: 'boolean', default: true })

  @OneToMany(() => Department, department => department.organization)

  @OneToMany(() => User, user => user.organization)
  users: User[];

  @CreateDateColumn()
  createdAt: Date;

  @UpdateDateColumn()
  updatedAt: Date;
}

interface OrganizationSettings {
  maxDepth: number; // 최대 조직 깊이 (기본: 4)
  allowCrossDepartmentAccess: boolean;
  defaultTeamPermissions: string[];
}

@Entity('team_members')
export class TeamMember {
  @PrimaryGeneratedColumn('uuid')
  id: string;

  @ManyToOne(() => Team, team => team.members, { onDelete: 'CASCADE' })
  @JoinColumn({ name: 'teamId' })
  team: Team;

  @Column({ type: 'uuid' })
  teamId: string;

  @ManyToOne(() => User, user => user.teamMemberships, { onDelete: 'CASCADE' })
  @JoinColumn({ name: 'userId' })
  user: User;

  @Column({ type: 'uuid' })
  userId: string;

  @Column({ 
    type: 'enum', 
    enum: ['member', 'leader', 'admin'],
    default: 'member'
  })
  role: TeamRole;

  @Column({ type: 'timestamp', nullable: true })
  joinedAt: Date;

  @Column({ type: 'timestamp', nullable: true })
  leftAt: Date;

  @Column({ type: 'boolean', default: true })
  isActive: boolean;

  @CreateDateColumn()
  createdAt: Date;

  @UpdateDateColumn()
  updatedAt: Date;
}

enum TeamRole {
  MEMBER = 'member',
  LEADER = 'leader',
  ADMIN = 'admin'
}

### 4. 권한 상속 구조

#### 권한 해결 서비스
```typescript
// backend-api/src/modules/admin/services/permission-resolver.service.ts
@Injectable()
export class PermissionResolverService {
  constructor(
    @InjectRepository(Team) private teamRepo: Repository<Team>,
    @InjectRepository(Department) private deptRepo: Repository<Department>,
    @InjectRepository(Organization) private orgRepo: Repository<Organization>
  ) {}

  async resolveUserPermissions(userId: string): Promise<ResolvedPermissions> {
    const userTeams = await this.getUserActiveTeams(userId);
    const permissions = new Set<string>();

    for (const team of userTeams) {
      const teamPermissions = await this.resolveTeamPermissions(team);
      teamPermissions.forEach(p => permissions.add(p));
    }

    return {
      permissions: Array.from(permissions),
      teams: userTeams.map(t => ({
        teamId: t.id,
        teamName: t.name,
        role: t.members[0].role, // 사용자의 해당 팀에서의 역할
        permissions: this.getTeamSpecificPermissions(t)
      }))
    };
  }

  private async resolveTeamPermissions(team: Team): Promise<string[]> {
    let permissions = [...(team.permissions?.permissions || [])];

    if (team.permissions?.inheritFromParent) {
      const dept = await this.deptRepo.findOne({
        where: { id: team.departmentId },
        relations: ['parent', 'organization']
      });

      if (dept?.parent) {
        const parentPermissions = await this.resolveDepartmentPermissions(dept.parent);
        permissions = [...permissions, ...parentPermissions];
      }

      // 조직 레벨 권한도 상속
      const orgPermissions = await this.resolveOrganizationPermissions(dept.organization);
      permissions = [...permissions, ...orgPermissions];
    }

    return [...new Set(permissions)]; // 중복 제거
  }
}
```
