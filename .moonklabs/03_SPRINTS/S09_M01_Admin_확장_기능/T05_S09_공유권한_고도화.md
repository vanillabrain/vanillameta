#### 패스워드 보호 서비스
```typescript
// backend-api/src/modules/sharing/services/password-protection.service.ts
@Injectable()
export class PasswordProtectionService {
  constructor(
    @InjectRepository(ShareLinkPassword) private passwordRepo: Repository<ShareLinkPassword>,
    @InjectRepository(ShareLink) private shareLinkRepo: Repository<ShareLink>
  ) {}

  async setPassword(shareLinkId: string, password: string, policy?: PasswordPolicy): Promise<void> {
    // 패스워드 정책 검증
    this.validatePassword(password, policy);

    const salt = this.generateSalt();
    const passwordHash = await bcrypt.hash(password + salt, 12);

    const existingPassword = await this.passwordRepo.findOne({
      where: { shareLinkId }
    });

    if (existingPassword) {
      existingPassword.passwordHash = passwordHash;
      existingPassword.salt = salt;
      existingPassword.passwordPolicy = policy;
      existingPassword.lastUpdated = new Date();
      existingPassword.failedAttempts = 0;
      existingPassword.lockedUntil = null;
      
      await this.passwordRepo.save(existingPassword);
    } else {
      const newPassword = this.passwordRepo.create({
        shareLinkId,
        passwordHash,
        salt,
        passwordPolicy: policy,
        lastUpdated: new Date()
      });
      
      await this.passwordRepo.save(newPassword);
    }

    // ShareLink 상태 업데이트
    await this.shareLinkRepo.update(shareLinkId, {
      shareLevel: ShareLevel.PASSWORD_PROTECTED
    });
  }

  async verifyPassword(shareLinkId: string, password: string, ipAddress: string): Promise<boolean> {
    const passwordRecord = await this.passwordRepo.findOne({
      where: { shareLinkId }
    });

    if (!passwordRecord) {
      throw new UnauthorizedException('패스워드가 설정되지 않았습니다.');
    }

    // 잠금 확인
    if (passwordRecord.lockedUntil && passwordRecord.lockedUntil > new Date()) {
      throw new TooManyRequestsException('계정이 잠겨있습니다. 나중에 다시 시도해주세요.');
    }

    // 패스워드 검증
    const isValid = await bcrypt.compare(
      password + passwordRecord.salt,
      passwordRecord.passwordHash
    );

    if (isValid) {
      // 성공 시 실패 횟수 초기화
      passwordRecord.failedAttempts = 0;
      passwordRecord.lockedUntil = null;
      await this.passwordRepo.save(passwordRecord);
      
      return true;
    } else {
      // 실패 시 카운트 증가
      passwordRecord.failedAttempts++;
      
      const policy = passwordRecord.passwordPolicy;
      const maxAttempts = policy?.maxAttempts || 5;
      
      if (passwordRecord.failedAttempts >= maxAttempts) {
        const lockoutDuration = policy?.lockoutDuration || 30; // 분
        passwordRecord.lockedUntil = new Date(Date.now() + lockoutDuration * 60 * 1000);
      }
      
      await this.passwordRepo.save(passwordRecord);
      
      throw new UnauthorizedException('패스워드가 올바르지 않습니다.');
    }
  }

  async removePassword(shareLinkId: string): Promise<void> {
    await this.passwordRepo.delete({ shareLinkId });
    
    // ShareLink 레벨 변경
    await this.shareLinkRepo.update(shareLinkId, {
      shareLevel: ShareLevel.INTERNAL
    });
  }

  private validatePassword(password: string, policy?: PasswordPolicy): void {
    if (!policy) return;

    if (password.length < policy.minLength) {
      throw new BadRequestException(`패스워드는 최소 ${policy.minLength}자 이상이어야 합니다.`);
    }

    if (policy.requireUppercase && !/[A-Z]/.test(password)) {
      throw new BadRequestException('패스워드에 대문자를 포함해야 합니다.');
    }

    if (policy.requireLowercase && !/[a-z]/.test(password)) {
      throw new BadRequestException('패스워드에 소문자를 포함해야 합니다.');
    }

    if (policy.requireNumbers && !/\d/.test(password)) {
      throw new BadRequestException('패스워드에 숫자를 포함해야 합니다.');
    }

    if (policy.requireSpecialChars && !/[!@#$%^&*(),.?":{}|<>]/.test(password)) {
      throw new BadRequestException('패스워드에 특수문자를 포함해야 합니다.');
    }
  }

  private generateSalt(): string {
    return crypto.randomBytes(32).toString('hex');
  }
}
```

#### 정책 템플릿 서비스
```typescript
// backend-api/src/modules/sharing/services/share-policy-template.service.ts
@Injectable()
export class SharePolicyTemplateService {
  constructor(
    @InjectRepository(SharePolicyTemplate) private templateRepo: Repository<SharePolicyTemplate>,
    @InjectRepository(ShareLink) private shareLinkRepo: Repository<ShareLink>
  ) {}

  async createTemplate(request: CreateTemplateRequest): Promise<SharePolicyTemplate> {
    // 기본 템플릿은 하나만 허용
    if (request.isDefault) {
      await this.clearDefaultTemplate(request.teamId);
    }

    const template = this.templateRepo.create(request);
    return this.templateRepo.save(template);
  }

  async getTemplates(teamId?: string): Promise<SharePolicyTemplate[]> {
    const query = this.templateRepo.createQueryBuilder('template')
      .where('template.isActive = true')
      .orderBy('template.isDefault', 'DESC')
      .addOrderBy('template.name', 'ASC');

    if (teamId) {
      query.andWhere('(template.teamId = :teamId OR template.teamId IS NULL)', { teamId });
    } else {
      query.andWhere('template.teamId IS NULL');
    }

    return query.getMany();
  }

  async applyTemplate(shareLinkId: string, templateId: string): Promise<void> {
    const template = await this.templateRepo.findOne({
      where: { id: templateId, isActive: true }
    });

    if (!template) {
      throw new NotFoundException('템플릿을 찾을 수 없습니다.');
    }

    const shareLink = await this.shareLinkRepo.findOne({
      where: { id: shareLinkId }
    });

    if (!shareLink) {
      throw new NotFoundException('공유 링크를 찾을 수 없습니다.');
    }

    // 정책 적용
    await this.applyPolicyToShareLink(shareLink, template.policy);
  }

  async getDefaultTemplate(teamId?: string): Promise<SharePolicyTemplate | null> {
    const query = this.templateRepo.createQueryBuilder('template')
      .where('template.isDefault = true')
      .andWhere('template.isActive = true');

    if (teamId) {
      query.andWhere('(template.teamId = :teamId OR template.teamId IS NULL)', { teamId })
        .orderBy('template.teamId', 'DESC'); // 팀 템플릿 우선
    } else {
      query.andWhere('template.teamId IS NULL');
    }

    return query.getOne();
  }

  private async applyPolicyToShareLink(shareLink: ShareLink, policy: SharePolicyConfig): Promise<void> {
    // ShareLink 설정 업데이트
    shareLink.shareLevel = policy.accessControl.shareLevel;
    shareLink.settings = {
      ...shareLink.settings,
      allowDownload: policy.features.allowDownload,
      allowComments: policy.features.allowComments,
      allowEmbedding: policy.features.allowEmbedding,
      watermark: policy.features.watermark,
      trackingEnabled: policy.tracking.enableAnalytics,
      maxViewers: policy.security.maxConcurrentUsers
    };

    // 만료 시간 설정
    if (policy.timeRestrictions.expiryType === 'duration') {
      const expiryDate = new Date();
      expiryDate.setHours(expiryDate.getHours() + Number(policy.timeRestrictions.expiryValue));
      shareLink.expiresAt = expiryDate;
    } else if (policy.timeRestrictions.expiryType === 'date') {
      shareLink.expiresAt = new Date(policy.timeRestrictions.expiryValue);
    }

    await this.shareLinkRepo.save(shareLink);

    // 패스워드 설정
    if (policy.security.requirePassword) {
      // 임시 패스워드 생성 또는 기존 패스워드 유지
      // 실제로는 사용자가 설정해야 함
    }
  }

  private async clearDefaultTemplate(teamId?: string): Promise<void> {
    const query = this.templateRepo.createQueryBuilder()
      .update(SharePolicyTemplate)
      .set({ isDefault: false })
      .where('isDefault = true');

    if (teamId) {
      query.andWhere('teamId = :teamId', { teamId });
    } else {
      query.andWhere('teamId IS NULL');
    }

    await query.execute();
  }
}
```
