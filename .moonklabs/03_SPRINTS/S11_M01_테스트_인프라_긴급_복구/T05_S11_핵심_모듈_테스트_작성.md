```typescript
// backend-api/src/auth/auth.service.spec.ts
import { Test, TestingModule } from '@nestjs/testing';
import { JwtService } from '@nestjs/jwt';
import { AuthService } from './auth.service';
import { UserService } from '../user/user.service';
import { MockDataFactory } from '../../../shared/test-utils/factories';
import * as bcrypt from 'bcrypt';

describe('AuthService', () => {
  let service: AuthService;
  let userService: UserService;
  let jwtService: JwtService;

  const mockUserService = {
    findByEmail: jest.fn(),
    create: jest.fn(),
    findById: jest.fn(),
  };

  const mockJwtService = {
    sign: jest.fn(),
    verify: jest.fn(),
  };

  beforeEach(async () => {
    const module: TestingModule = await Test.createTestingModule({
      providers: [
        AuthService,
        { provide: UserService, useValue: mockUserService },
        { provide: JwtService, useValue: mockJwtService },
      ],
    }).compile();

    service = module.get<AuthService>(AuthService);
    userService = module.get<UserService>(UserService);
    jwtService = module.get<JwtService>(JwtService);
  });

  describe('login', () => {
    it('사용자가 올바른 자격 증명으로 로그인할 수 있어야 한다', async () => {
      // Given
      const email = 'test@example.com';
      const password = 'password123';
      const hashedPassword = await bcrypt.hash(password, 10);
      const user = MockDataFactory.createUser({ 
        email, 
        password: hashedPassword 
      });
      const token = 'jwt-token';

      mockUserService.findByEmail.mockResolvedValue(user);
      mockJwtService.sign.mockReturnValue(token);

      // When
      const result = await service.login(email, password);

      // Then
      expect(result).toEqual({
        access_token: token,
        user: expect.objectContaining({ email })
      });
      expect(userService.findByEmail).toHaveBeenCalledWith(email);
      expect(jwtService.sign).toHaveBeenCalledWith({ 
        sub: user.id, 
        email: user.email 
      });
    });

    it('잘못된 비밀번호로 로그인 시 실패해야 한다', async () => {
      // Given
      const email = 'test@example.com';
      const password = 'wrongpassword';
      const hashedPassword = await bcrypt.hash('correctpassword', 10);
      const user = MockDataFactory.createUser({ 
        email, 
        password: hashedPassword 
      });

      mockUserService.findByEmail.mockResolvedValue(user);

      // When & Then
      await expect(service.login(email, password))
        .rejects.toThrow('Invalid credentials');
    });

    it('존재하지 않는 사용자로 로그인 시 실패해야 한다', async () => {
      // Given
      const email = 'nonexistent@example.com';
      const password = 'password123';

      mockUserService.findByEmail.mockResolvedValue(null);

      // When & Then
      await expect(service.login(email, password))
        .rejects.toThrow('User not found');
    });
  });

  describe('validateToken', () => {
    it('유효한 토큰을 검증할 수 있어야 한다', async () => {
      // Given
      const token = 'valid-jwt-token';
      const payload = { sub: 1, email: 'test@example.com' };
      const user = MockDataFactory.createUser({ id: 1, email: 'test@example.com' });

      mockJwtService.verify.mockReturnValue(payload);
      mockUserService.findById.mockResolvedValue(user);

      // When
      const result = await service.validateToken(token);

      // Then
      expect(result).toEqual(user);
      expect(jwtService.verify).toHaveBeenCalledWith(token);
      expect(userService.findById).toHaveBeenCalledWith(1);
    });

    it('유효하지 않은 토큰 검증 시 실패해야 한다', async () => {
      // Given
      const token = 'invalid-jwt-token';

      mockJwtService.verify.mockImplementation(() => {
        throw new Error('Invalid token');
      });

      // When & Then
      await expect(service.validateToken(token))
        .rejects.toThrow('Invalid token');
    });
  });
});
```

```typescript
        title: 'New Dashboard',
        description: 'Dashboard description'
      };
      const dashboard = MockDataFactory.createDashboard({
        ...createDto,
        userId: user.id
      });

      mockRepository.create.mockReturnValue(dashboard);
      mockRepository.save.mockResolvedValue(dashboard);

      // When
      const result = await service.createDashboard(createDto, user);

      // Then
      expect(result).toEqual(dashboard);
      expect(repository.create).toHaveBeenCalledWith({
        ...createDto,
        userId: user.id
      });
      expect(repository.save).toHaveBeenCalledWith(dashboard);
    });
  });

  describe('getUserDashboards', () => {
    it('사용자의 대시보드 목록을 가져올 수 있어야 한다', async () => {
      // Given
      const userId = 1;
      const dashboards = Array.from({ length: 3 }, () =>
        MockDataFactory.createDashboard({ userId })
      );

      mockRepository.find.mockResolvedValue(dashboards);

      // When
      const result = await service.getUserDashboards(userId);

      // Then
      expect(result).toEqual(dashboards);
      expect(repository.find).toHaveBeenCalledWith({
        where: { userId },
        relations: ['widgets']
      });
    });
  });

  describe('updateDashboard', () => {
    it('대시보드를 업데이트할 수 있어야 한다', async () => {
      // Given
      const dashboardId = 1;
      const userId = 1;
      const updateDto = { title: 'Updated Dashboard' };
      const existingDashboard = MockDataFactory.createDashboard({
        id: dashboardId,
        userId
      });
      const updatedDashboard = { ...existingDashboard, ...updateDto };

      mockRepository.findOne.mockResolvedValue(existingDashboard);
      mockRepository.save.mockResolvedValue(updatedDashboard);

      // When
      const result = await service.updateDashboard(dashboardId, updateDto, userId);

      // Then
      expect(result).toEqual(updatedDashboard);
      expect(repository.findOne).toHaveBeenCalledWith({
        where: { id: dashboardId, userId }
      });
    });

    it('권한이 없는 사용자가 대시보드 업데이트 시 실패해야 한다', async () => {
      // Given

      const unauthorizedUserId = 2;

        service.updateDashboard(dashboardId, updateDto, unauthorizedUserId)
      ).rejects.toThrow('Dashboard not found or access denied');
    });
  });
```

### E2E 테스트 예시

#### 사용자 플로우 E2E 테스트
```typescript
// backend-api/test/e2e/user-flow.e2e-spec.ts
import { Test, TestingModule } from '@nestjs/testing';
import { INestApplication } from '@nestjs/common';
import * as request from 'supertest';
import { AppModule } from '../../src/app.module';
import { DatabaseTestHelper } from '../helpers/database-helper';

describe('User Flow (e2e)', () => {
  let app: INestApplication;
  let dbHelper: DatabaseTestHelper;

  beforeAll(async () => {
    const moduleFixture: TestingModule = await Test.createTestingModule({
      imports: [AppModule],
    }).compile();

    app = moduleFixture.createNestApplication();
    await app.init();

    dbHelper = DatabaseTestHelper.getInstance();
    await dbHelper.setupDatabase();
  });

  afterAll(async () => {
    await dbHelper.closeDatabase();
    await app.close();
  });

  beforeEach(async () => {
    await dbHelper.cleanDatabase();
  });

  it('사용자가 로그인하여 대시보드를 생성하고 위젯을 추가할 수 있어야 한다', async () => {
    // 1. 사용자 로그인
    const loginResponse = await request(app.getHttpServer())
      .post('/auth/login')
      .send({
        email: 'admin@test.com',
        password: 'password123'
      })
      .expect(200);

    const { access_token } = loginResponse.body;

    // 2. 대시보드 생성
    const dashboardResponse = await request(app.getHttpServer())
      .post('/dashboards')
      .set('Authorization', `Bearer ${access_token}`)
      .send({
        title: 'E2E Test Dashboard',
        description: 'Dashboard created in E2E test'
      })
      .expect(201);

    const dashboard = dashboardResponse.body;
    expect(dashboard.title).toBe('E2E Test Dashboard');

    // 3. 위젯 추가
    const widgetResponse = await request(app.getHttpServer())
      .post('/widgets')
      .set('Authorization', `Bearer ${access_token}`)
      .send({
        dashboardId: dashboard.id,
        title: 'E2E Test Widget',
        type: 'chart',
        config: {
          chartType: 'line',
          query: 'SELECT * FROM test_data'
        },
        position: { x: 0, y: 0, w: 4, h: 3 }
      })
      .expect(201);

    const widget = widgetResponse.body;
    expect(widget.title).toBe('E2E Test Widget');
    expect(widget.dashboardId).toBe(dashboard.id);

    // 4. 대시보드 조회 (위젯 포함)
    const dashboardWithWidgets = await request(app.getHttpServer())
      .get(`/dashboards/${dashboard.id}`)
      .set('Authorization', `Bearer ${access_token}`)
      .expect(200);

    expect(dashboardWithWidgets.body.widgets).toHaveLength(1);
    expect(dashboardWithWidgets.body.widgets[0].id).toBe(widget.id);
  });
});
```
