// src/test/utils/permission-test.utils.ts
export class PermissionTestUtils {
  static async createTestUser(
    roleNames: string[],
    customPermissions?: string[],
  ): Promise<{ user: User; tokens: AuthTokens }> {
    const user = await this.createUser({
      email: `test-${Date.now()}@example.com`,
      name: 'Test User',
      password: 'test123',
    });

    // 역할 할당
    for (const roleName of roleNames) {
      const role = await this.findRoleByName(roleName);
      if (role) {
        await this.assignRoleToUser(user.id, role.id);
      }
    }

    // 커스텀 권한 추가 (테스트용)
    if (customPermissions) {
      await this.assignCustomPermissions(user.id, customPermissions);
    }

    const tokens = await this.generateTokens(user);
    return { user, tokens };
  }

  static async testEndpointPermission(
    endpoint: string,
    method: 'GET' | 'POST' | 'PUT' | 'DELETE',
    requiredPermissions: string[],
    testScenarios: PermissionTestScenario[],
  ): Promise<PermissionTestResult[]> {
    const results: PermissionTestResult[] = [];

    for (const scenario of testScenarios) {
      const { user, tokens } = await this.createTestUser(
        scenario.roles,
        scenario.customPermissions,
      );

      try {
        const response = await this.makeRequest(endpoint, method, {
          headers: { Authorization: `Bearer ${tokens.accessToken}` },
          data: scenario.requestData,
        });

        results.push({
          scenario: scenario.name,
          expected: scenario.expectedResult,
          actual: response.status,
          passed: response.status === scenario.expectedResult,
          details: response.data,
        });
      } catch (error) {
        results.push({
          scenario: scenario.name,
          expected: scenario.expectedResult,
          actual: error.response?.status || 500,
          passed: error.response?.status === scenario.expectedResult,
          error: error.message,
        });
      } finally {
        await this.cleanupTestUser(user.id);
      }
    }

    return results;
  }

  static generatePermissionMatrix(): PermissionMatrix {
    const endpoints = this.getAllEndpoints();
    const roles = this.getAllRoles();
    const matrix: PermissionMatrix = {};

    for (const endpoint of endpoints) {
      matrix[endpoint] = {};
      for (const role of roles) {
        matrix[endpoint][role] = this.checkRoleHasPermission(
          role,
          endpoint.requiredPermissions,
        );
      }
    }

    return matrix;
  }
}

interface PermissionTestScenario {
  name: string;
  roles: string[];
  customPermissions?: string[];
  requestData?: any;
  expectedResult: number; // HTTP status code
}

interface PermissionTestResult {
  scenario: string;
  expected: number;
  actual: number;
  passed: boolean;
  details?: any;
  error?: string;
}

// src/test/scripts/generate-permission-matrix.ts
interface PermissionMatrixEntry {
  endpoint: string;
  method: string;
  requiredPermissions: string[];
  roles: {
    [roleName: string]: {
      hasAccess: boolean;
      reason: string;
    };
  };
}

export class PermissionMatrixGenerator {
  private readonly app: INestApplication;
  private readonly reflector: Reflector;

  constructor(app: INestApplication) {
    this.app = app;
    this.reflector = app.get(Reflector);
  }

  async generateMatrix(): Promise<PermissionMatrixEntry[]> {
    const routes = this.extractRoutes();
    const roles = await this.getAllRoles();
    const matrix: PermissionMatrixEntry[] = [];

    for (const route of routes) {
      const entry: PermissionMatrixEntry = {
        endpoint: route.path,
        method: route.method,
        requiredPermissions: route.permissions,
        roles: {},
      };

      for (const role of roles) {
        const rolePermissions = await this.getRolePermissions(role.name);
        const hasAccess = this.checkPermissions(
          route.permissions,
          rolePermissions,
        );

        entry.roles[role.name] = {
          hasAccess,
          reason: hasAccess
            ? 'Has required permissions'
            : `Missing: ${route.permissions.filter(p => !rolePermissions.includes(p)).join(', ')}`,
        };
      }

      matrix.push(entry);
    }

    return matrix;
  }

  private extractRoutes(): Array<{
    path: string;
    method: string;
    permissions: string[];
  }> {
    const routes = [];
    
    // NestJS의 라우터 메타데이터에서 추출
    const httpAdapter = this.app.getHttpAdapter();
    const router = httpAdapter.getInstance()._router;

    for (const layer of router.stack) {
      if (layer.route) {
        const route = layer.route;
        const permissions = this.extractPermissionsFromRoute(route);
        
        for (const method of Object.keys(route.methods)) {
          routes.push({
            path: route.path,
            method: method.toUpperCase(),
            permissions,
          });
        }
      }
    }

    return routes;
  }

  private extractPermissionsFromRoute(route: any): string[] {
    // Route handler에서 @RequirePermissions 데코레이터 정보 추출
    const handler = route.stack[0]?.handle;
    if (handler) {
      return this.reflector.get('permissions', handler) || [];
    }
    return [];
  }

  async generateReport(): Promise<void> {
    const matrix = await this.generateMatrix();
    
    // CSV 리포트 생성
    const csvData = this.generateCsvReport(matrix);
    await fs.writeFile('permission-matrix.csv', csvData);

    // HTML 리포트 생성
    const htmlData = this.generateHtmlReport(matrix);
    await fs.writeFile('permission-matrix.html', htmlData);

    // JSON 리포트 생성
    await fs.writeFile(
      'permission-matrix.json',
      JSON.stringify(matrix, null, 2),
    );

    console.log('Permission matrix reports generated:');
    console.log('- permission-matrix.csv');
    console.log('- permission-matrix.html');
    console.log('- permission-matrix.json');
  }

  private generateHtmlReport(matrix: PermissionMatrixEntry[]): string {
    const roles = Object.keys(matrix[0]?.roles || {});
    
    return `
      <!DOCTYPE html>
      <html>
      <head>
        <title>API Permission Matrix</title>
        <style>
          table { border-collapse: collapse; width: 100%; }
          th, td { border: 1px solid #ddd; padding: 8px; text-align: left; }
          th { background-color: #f2f2f2; }
          .has-access { background-color: #d4edda; }
          .no-access { background-color: #f8d7da; }
          .endpoint { font-family: monospace; }
        </style>
      </head>
      <body>
        <h1>API Permission Matrix</h1>
        <table>
          <thead>
            <tr>
              <th>Endpoint</th>
              <th>Method</th>
              <th>Required Permissions</th>
              ${roles.map(role => `<th>${role}</th>`).join('')}
            </tr>
          </thead>
          <tbody>
            ${matrix.map(entry => `
              <tr>
                <td class="endpoint">${entry.endpoint}</td>
                <td>${entry.method}</td>
                <td>${entry.requiredPermissions.join(', ')}</td>
                ${roles.map(role => `
                  <td class="${entry.roles[role].hasAccess ? 'has-access' : 'no-access'}"
                      title="${entry.roles[role].reason}">
                    ${entry.roles[role].hasAccess ? '✓' : '✗'}
                  </td>
                `).join('')}
              </tr>
            `).join('')}
          </tbody>
        </table>
      </body>
      </html>
    `;
  }
}

// src/test/security/security-vulnerability.test.ts
describe('Security Vulnerability Tests', () => {
  describe('Authorization Bypass Attempts', () => {
    it('should prevent JWT token manipulation', async () => {
      const { tokens } = await testUtils.createTestUser(['viewer']);
      
      // 토큰 페이로드 조작 시도
      const [header, payload, signature] = tokens.accessToken.split('.');
      const decodedPayload = JSON.parse(Buffer.from(payload, 'base64').toString());
      
      // 권한을 super_admin으로 변경 시도
      decodedPayload.roles = ['super_admin'];
      const manipulatedPayload = Buffer.from(JSON.stringify(decodedPayload)).toString('base64');
      const manipulatedToken = `${header}.${manipulatedPayload}.${signature}`;

      // 조작된 토큰으로 Admin API 접근 시도
      await request(app.getHttpServer())
        .get('/admin/users')
        .set('Authorization', `Bearer ${manipulatedToken}`)
        .expect(401); // 토큰 검증 실패
    });

    it('should prevent privilege escalation through parameter injection', async () => {
      const { user, tokens } = await testUtils.createTestUser(['editor']);
      
      // 자신의 역할을 super_admin으로 변경 시도
      await request(app.getHttpServer())
        .put(`/admin/users/${user.id}`)
        .set('Authorization', `Bearer ${tokens.accessToken}`)
        .send({
          roles: ['super_admin'], // 권한 없이 역할 변경 시도
        })
        .expect(403);
    });

    it('should prevent SQL injection in search parameters', async () => {
      const { tokens } = await testUtils.createTestUser(['admin']);
      
      const maliciousSearches = [
        "'; DROP TABLE users; --",
        "' OR '1'='1",
        "'; INSERT INTO users (email) VALUES ('hacker@evil.com'); --",
      ];

      for (const search of maliciousSearches) {
        const response = await request(app.getHttpServer())
          .get('/admin/users')
          .query({ search })
          .set('Authorization', `Bearer ${tokens.accessToken}`)
          .expect(200);

        // 결과가 안전하게 필터링되었는지 확인
        expect(response.body.data).toBeDefined();
        expect(Array.isArray(response.body.data)).toBe(true);
      }
    });
  });

  describe('Information Disclosure Prevention', () => {
    it('should not expose sensitive information in error messages', async () => {
      const { tokens } = await testUtils.createTestUser(['viewer']);
      
      const response = await request(app.getHttpServer())
        .get('/admin/users/non-existent-id')
        .set('Authorization', `Bearer ${tokens.accessToken}`)
        .expect(403);

      // 에러 메시지가 권한 없음을 나타내야 하고, 
      // 리소스 존재 여부를 알려주면 안됨
      expect(response.body.message).not.toContain('not found');
      expect(response.body.message).toContain('permission');
    });

    it('should mask sensitive audit log data for unauthorized users', async () => {
      const { tokens } = await testUtils.createTestUser(['admin']);
      
      // 민감한 정보가 포함된 로그 생성
      await testUtils.createAuditLog({
        action: 'PASSWORD_RESET',
        details: {
          oldPassword: 'secret123',
          newPassword: 'newsecret456',
          resetToken: 'sensitive-token',
        },
      });

      const response = await request(app.getHttpServer())
        .get('/admin/audit/logs')
        .set('Authorization', `Bearer ${tokens.accessToken}`)
        .expect(200);

      // 민감한 정보가 마스킹되었는지 확인
      const logs = response.body.data;
      const passwordResetLog = logs.find(log => log.action === 'PASSWORD_RESET');
      
      if (passwordResetLog) {
        expect(passwordResetLog.details.oldPassword).toBe('[REDACTED]');
        expect(passwordResetLog.details.newPassword).toBe('[REDACTED]');
        expect(passwordResetLog.details.resetToken).toBe('[REDACTED]');
      }
    });
  });
});
