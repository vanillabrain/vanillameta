```typescript
@Entity('users')
export class User {
  @PrimaryGeneratedColumn('uuid')
  id: string;

  @Column({ unique: true })
  email: string;

  @Column()
  name: string;

  @Column({ select: false })
  password: string;

  @Column({
    type: 'enum',
    enum: UserStatus,
    default: UserStatus.PENDING,
  })
  status: UserStatus;

  @Column({ nullable: true })
  avatar: string;

  @Column({ nullable: true })
  phone: string;

  @Column({ nullable: true })
  department: string;

  @Column({ type: 'timestamp', nullable: true })
  lastLoginAt: Date;

  @Column({ type: 'timestamp', nullable: true })
  emailVerifiedAt: Date;

  @ManyToMany(() => Role, (role) => role.users)
  @JoinTable({
    name: 'user_roles',
    joinColumn: { name: 'userId', referencedColumnName: 'id' },
    inverseJoinColumn: { name: 'roleId', referencedColumnName: 'id' },
  })
  roles: Role[];

  @CreateDateColumn()
  createdAt: Date;

  @UpdateDateColumn()
  updatedAt: Date;

  @DeleteDateColumn()
  deletedAt: Date;
}
```

```typescript
@Controller('admin/users')
@UseGuards(JwtAuthGuard, PermissionGuard)
export class AdminUsersController {
  constructor(private readonly usersService: AdminUsersService) {}

  @Get()
  @RequirePermissions('admin.users.view')
  async getUsers(
    @Query() query: GetUsersQueryDto,
  ): Promise<PaginatedResponseDto<UserDto>> {
    return this.usersService.getUsers(query);
  }

  @Get(':id')
  @RequirePermissions('admin.users.view')
  async getUserById(@Param('id') id: string): Promise<UserDetailDto> {
    return this.usersService.getUserById(id);
  }

  @Post()
  @RequirePermissions('admin.users.create')
  async createUser(@Body() createUserDto: CreateUserDto): Promise<UserDto> {
    return this.usersService.createUser(createUserDto);
  }

  @Put(':id')
  @RequirePermissions('admin.users.edit')
  async updateUser(
    @Param('id') id: string,
    @Body() updateUserDto: UpdateUserDto,
  ): Promise<UserDto> {
    return this.usersService.updateUser(id, updateUserDto);
  }

  @Delete(':id')
  @RequirePermissions('admin.users.delete')
  async deleteUser(@Param('id') id: string): Promise<void> {
    return this.usersService.deleteUser(id);
  }

  @Put(':id/status')
  @RequirePermissions('admin.users.edit')
  async updateUserStatus(
    @Param('id') id: string,
    @Body() updateStatusDto: UpdateUserStatusDto,
  ): Promise<UserDto> {
    return this.usersService.updateUserStatus(id, updateStatusDto.status);
  }

  @Post('bulk-action')
  @RequirePermissions('admin.users.edit')
  async bulkAction(@Body() bulkActionDto: BulkActionDto): Promise<void> {
    return this.usersService.bulkAction(bulkActionDto);
  }
}
```

```typescript
// src/modules/admin/users/dto/get-users-query.dto.ts
export class GetUsersQueryDto {
  @IsOptional()
  @IsInt()
  @Transform(({ value }) => parseInt(value))
  page?: number = 1;

  @IsOptional()
  @IsInt()
  @Transform(({ value }) => parseInt(value))
  limit?: number = 20;

  @IsOptional()
  @IsString()
  search?: string;

  @IsOptional()
  @IsEnum(UserStatus)
  status?: UserStatus;

  @IsOptional()
  @IsString()
  role?: string;

  @IsOptional()
  @IsDateString()
  createdAfter?: string;

  @IsOptional()
  @IsDateString()
  createdBefore?: string;

  @IsOptional()
  @IsIn(['name', 'email', 'createdAt', 'lastLoginAt'])
  sortBy?: string = 'createdAt';

  @IsOptional()
  @IsIn(['ASC', 'DESC'])
  sortOrder?: 'ASC' | 'DESC' = 'DESC';
}
```

#### 4. Admin Users Service
```typescript
// src/modules/admin/users/admin-users.service.ts
@Injectable()
export class AdminUsersService {
  constructor(
    @InjectRepository(User)
    private userRepository: Repository<User>,
    @InjectRepository(Role)
    private roleRepository: Repository<Role>,
    private emailService: EmailService,
    private auditLogService: AuditLogService,
  ) {}

  async getUsers(query: GetUsersQueryDto): Promise<PaginatedResponseDto<UserDto>> {
    const qb = this.userRepository
      .createQueryBuilder('user')
      .leftJoinAndSelect('user.roles', 'roles')
      .withDeleted(); // 삭제된 사용자도 포함

    // 검색 조건 적용
    if (query.search) {
      qb.andWhere(
        '(user.name ILIKE :search OR user.email ILIKE :search)',
        { search: `%${query.search}%` },
      );
    }

    // 필터 조건 적용
    if (query.status) {
      qb.andWhere('user.status = :status', { status: query.status });
    }

    if (query.role) {
      qb.andWhere('roles.name = :role', { role: query.role });
    }

    if (query.createdAfter) {
      qb.andWhere('user.createdAt >= :createdAfter', {
        createdAfter: query.createdAfter,
      });
    }

    if (query.createdBefore) {
      qb.andWhere('user.createdAt <= :createdBefore', {
        createdBefore: query.createdBefore,
      });
    }

    // 정렬 적용
    qb.orderBy(`user.${query.sortBy}`, query.sortOrder);

    // 페이지네이션 적용
    const [users, total] = await qb
      .skip((query.page - 1) * query.limit)
      .take(query.limit)
      .getManyAndCount();

    return {
      data: users.map(user => new UserDto(user)),
      meta: {
        total,
        page: query.page,
        limit: query.limit,
        totalPages: Math.ceil(total / query.limit),
      },
    };
  }

  async createUser(createUserDto: CreateUserDto): Promise<UserDto> {
    // 임시 비밀번호 생성
    const tempPassword = this.generateTempPassword();
    
    const user = this.userRepository.create({
      ...createUserDto,
      password: await bcrypt.hash(tempPassword, 10),
      status: UserStatus.PENDING,
    });

    // 역할 할당
    if (createUserDto.roleIds?.length) {
      user.roles = await this.roleRepository.findByIds(createUserDto.roleIds);
    }

    const savedUser = await this.userRepository.save(user);

    // 환영 이메일 발송
    await this.emailService.sendWelcomeEmail(savedUser, tempPassword);

    // 감사 로그 기록
    await this.auditLogService.log({
      action: 'USER_CREATED',
      resourceType: 'User',
      resourceId: savedUser.id,
      details: { email: savedUser.email },
    });

    return new UserDto(savedUser);
  }

  async updateUserStatus(userId: string, status: UserStatus): Promise<UserDto> {
    const user = await this.userRepository.findOne({
      where: { id: userId },
      relations: ['roles'],
    });

    if (!user) {
      throw new NotFoundException('사용자를 찾을 수 없습니다');
    }

    const oldStatus = user.status;
    user.status = status;
    
    const updatedUser = await this.userRepository.save(user);

    // 상태 변경 알림 이메일
    if (status === UserStatus.ACTIVE && oldStatus === UserStatus.PENDING) {
      await this.emailService.sendAccountActivatedEmail(user);
    } else if (status === UserStatus.SUSPENDED) {
      await this.emailService.sendAccountSuspendedEmail(user);
    }

    // 감사 로그 기록
    await this.auditLogService.log({
      action: 'USER_STATUS_CHANGED',
      resourceType: 'User',
      resourceId: userId,
      details: { oldStatus, newStatus: status },
    });

    return new UserDto(updatedUser);
  }

  async bulkAction(bulkActionDto: BulkActionDto): Promise<void> {
    const { action, userIds } = bulkActionDto;

    switch (action) {
      case 'activate':
        await this.userRepository.update(
          { id: In(userIds) },
          { status: UserStatus.ACTIVE },
        );
        break;
      case 'deactivate':
        await this.userRepository.update(
          { id: In(userIds) },
          { status: UserStatus.INACTIVE },
        );
        break;
      case 'delete':
        await this.userRepository.softDelete({ id: In(userIds) });
        break;
    }

    // 감사 로그 기록
    await this.auditLogService.log({
      action: `BULK_USER_${action.toUpperCase()}`,
      resourceType: 'User',
      details: { userIds, count: userIds.length },
    });
  }

  private generateTempPassword(): string {
    return Math.random().toString(36).slice(-12);
  }
}
```

```typescript
// src/components/admin/users/UserManagement.tsx
export const UserManagement: React.FC = () => {
  const [users, setUsers] = useState<PaginatedResponse<User>>();
  const [loading, setLoading] = useState(false);
  const [selectedUsers, setSelectedUsers] = useState<string[]>([]);
  const [filters, setFilters] = useState<UserFilters>({
    page: 1,
    limit: 20,
    search: '',
    status: undefined,
    sortBy: 'createdAt',
    sortOrder: 'DESC',
  });

  const { data, isLoading, error, refetch } = useQuery({
    queryKey: ['admin-users', filters],
    queryFn: () => adminApiClient.getUsers(filters),
  });

  const handleFilterChange = (newFilters: Partial<UserFilters>) => {
    setFilters(prev => ({ ...prev, ...newFilters, page: 1 }));
  };

  const handleBulkAction = async (action: BulkAction) => {
    if (selectedUsers.length === 0) return;
    
    try {
      await adminApiClient.bulkAction({ action, userIds: selectedUsers });
      setSelectedUsers([]);
      refetch();
      toast.success(`${selectedUsers.length}명의 사용자에 대한 작업이 완료되었습니다.`);
    } catch (error) {
      toast.error('작업 중 오류가 발생했습니다.');
    }
  };

  return (
    <div className="user-management">
      <div className="user-management-header">
        <h1>사용자 관리</h1>
        <div className="actions">
          <Button onClick={() => setCreateModalOpen(true)}>
            사용자 추가
          </Button>
        </div>
      </div>

      <UserFilters
        filters={filters}
        onFilterChange={handleFilterChange}
      />

      {selectedUsers.length > 0 && (
        <BulkActionBar
          selectedCount={selectedUsers.length}
          onAction={handleBulkAction}
          onClear={() => setSelectedUsers([])}
        />
      )}

      <UserTable
        users={data?.data || []}
        loading={isLoading}
        selectedUsers={selectedUsers}
        onSelectionChange={setSelectedUsers}
        onUserUpdate={refetch}
      />

      <Pagination
        current={filters.page}
        total={data?.meta.total || 0}
        pageSize={filters.limit}
        onChange={(page) => handleFilterChange({ page })}
      />
    </div>
  );
};
```

```typescript
// src/components/admin/users/UserTable.tsx
interface UserTableProps {
  users: User[];
  loading: boolean;
  selectedUsers: string[];
  onSelectionChange: (userIds: string[]) => void;
  onUserUpdate: () => void;
}

export const UserTable: React.FC<UserTableProps> = ({
  users,
  loading,
  selectedUsers,
  onSelectionChange,
  onUserUpdate,
}) => {
  const [editingUser, setEditingUser] = useState<User | null>(null);

  const columns: TableColumn<User>[] = [
    {
      key: 'selection',
      title: (
        <Checkbox
          checked={selectedUsers.length === users.length && users.length > 0}
          indeterminate={selectedUsers.length > 0 && selectedUsers.length < users.length}
          onChange={(checked) =>
            onSelectionChange(checked ? users.map(u => u.id) : [])
          }
        />
      ),
      render: (user) => (
        <Checkbox
          checked={selectedUsers.includes(user.id)}
          onChange={(checked) =>
            onSelectionChange(
              checked
                ? [...selectedUsers, user.id]
                : selectedUsers.filter(id => id !== user.id)
            )
          }
        />
      ),
    },
    {
      key: 'user',
      title: '사용자',
      render: (user) => (
        <div className="user-cell">
          <Avatar src={user.avatar} name={user.name} />
          <div>
            <div className="user-name">{user.name}</div>
            <div className="user-email">{user.email}</div>
          </div>
        </div>
      ),
    },
    {
      key: 'roles',
      title: '역할',
      render: (user) => (
        <div className="roles">
          {user.roles.map(role => (
            <Tag key={role.id} color={getRoleColor(role.name)}>
              {role.displayName}
            </Tag>
          ))}
        </div>
      ),
    },
    {
      key: 'status',
      title: '상태',
      render: (user) => (
        <UserStatusBadge
          status={user.status}
          onStatusChange={(status) => handleStatusChange(user.id, status)}
        />
      ),
    },
    {
      key: 'lastLoginAt',
      title: '최종 접속',
      render: (user) =>
        user.lastLoginAt ? formatDate(user.lastLoginAt) : '접속 기록 없음',
    },
    {
      key: 'createdAt',
      title: '가입일',
      render: (user) => formatDate(user.createdAt),
    },
    {
      key: 'actions',
      title: '작업',
      render: (user) => (
        <div className="table-actions">
          <Button
            size="small"
            onClick={() => setEditingUser(user)}
          >
            편집
          </Button>
          <Button
            size="small"
            variant="outline"
            onClick={() => handleViewDetails(user.id)}
          >
            상세
          </Button>
        </div>
      ),
    },
  ];

  const handleStatusChange = async (userId: string, status: UserStatus) => {
    try {
      await adminApiClient.updateUserStatus(userId, status);
      onUserUpdate();
      toast.success('사용자 상태가 변경되었습니다.');
    } catch (error) {
      toast.error('상태 변경 중 오류가 발생했습니다.');
    }
  };

  return (
    <>
      <Table
        columns={columns}
        data={users}
        loading={loading}
        className="user-table"
      />

      {editingUser && (
        <EditUserModal
          user={editingUser}
          onClose={() => setEditingUser(null)}
          onUpdate={onUserUpdate}
        />
      )}
    </>
  );
};
```
