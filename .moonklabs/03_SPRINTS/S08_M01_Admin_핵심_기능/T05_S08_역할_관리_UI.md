```typescript
// src/modules/admin/roles/admin-roles.service.ts
@Injectable()
export class AdminRolesService {
  constructor(
    @InjectRepository(Role)
    private roleRepository: Repository<Role>,
    @InjectRepository(Permission)
    private permissionRepository: Repository<Permission>,
    @InjectRepository(User)
    private userRepository: Repository<User>,
    @InjectRepository(UserRole)
    private userRoleRepository: Repository<UserRole>,
    private rbacService: RBACService,
    private auditLogService: AuditLogService,
  ) {}

  async getRolesWithStats(query: GetRolesQueryDto): Promise<RoleWithStatsDto[]> {
    const qb = this.roleRepository
      .createQueryBuilder('role')
      .leftJoinAndSelect('role.permissions', 'permissions')
      .loadRelationCountAndMap('role.userCount', 'role.users');

    if (query.search) {
      qb.andWhere(
        '(role.name ILIKE :search OR role.displayName ILIKE :search)',
        { search: `%${query.search}%` },
      );
    }

    if (query.isActive !== undefined) {
      qb.andWhere('role.isActive = :isActive', { isActive: query.isActive });
    }

    qb.orderBy('role.level', 'DESC');

    const roles = await qb.getMany();

    return roles.map(role => ({
      ...new RoleDto(role),
      userCount: role.userCount || 0,
      permissionCount: role.permissions?.length || 0,
    }));
  }

  async getRoleDetail(id: string): Promise<RoleDetailDto> {
    const role = await this.roleRepository.findOne({
      where: { id },
      relations: ['permissions', 'users'],
    });

    if (!role) {
      throw new NotFoundException('역할을 찾을 수 없습니다');
    }

    return {
      ...new RoleDto(role),
      permissions: role.permissions.map(p => new PermissionDto(p)),
      users: role.users.map(u => new UserDto(u)),
      createdAt: role.createdAt,
      updatedAt: role.updatedAt,
    };
  }

  async createRole(createRoleDto: CreateRoleDto, creator: User): Promise<RoleDto> {
    // 역할명 중복 확인
    const existingRole = await this.roleRepository.findOne({
      where: { name: createRoleDto.name },
    });

    if (existingRole) {
      throw new BadRequestException('이미 존재하는 역할명입니다');
    }

    const role = this.roleRepository.create({
      ...createRoleDto,
      isActive: true,
    });

    const savedRole = await this.roleRepository.save(role);

    // 권한 할당
    if (createRoleDto.permissionIds?.length) {
      const permissions = await this.permissionRepository.findByIds(
        createRoleDto.permissionIds,
      );
      savedRole.permissions = permissions;
      await this.roleRepository.save(savedRole);
    }

    // 감사 로그 기록
    await this.auditLogService.log({
      action: 'ROLE_CREATED',
      resourceType: 'Role',
      resourceId: savedRole.id,
      userId: creator.id,
      details: {
        roleName: savedRole.name,
        permissionCount: createRoleDto.permissionIds?.length || 0,
      },
    });

    return new RoleDto(savedRole);
  }

  async updateRole(
    id: string,
    updateRoleDto: UpdateRoleDto,
    updater: User,
  ): Promise<RoleDto> {
    const role = await this.roleRepository.findOne({
      where: { id },
      relations: ['permissions'],
    });

    if (!role) {
      throw new NotFoundException('역할을 찾을 수 없습니다');
    }

    // 시스템 기본 역할 수정 제한
    if (['super_admin', 'admin'].includes(role.name) && updater.id !== role.id) {
      throw new ForbiddenException('시스템 기본 역할은 수정할 수 없습니다');
    }

    const oldData = { ...role };

    // 기본 정보 업데이트
    Object.assign(role, updateRoleDto);

    const updatedRole = await this.roleRepository.save(role);

    // 감사 로그 기록
    await this.auditLogService.log({
      action: 'ROLE_UPDATED',
      resourceType: 'Role',
      resourceId: id,
      userId: updater.id,
      details: {
        oldData: {
          displayName: oldData.displayName,
          description: oldData.description,
          level: oldData.level,
        },
        newData: {
          displayName: updatedRole.displayName,
          description: updatedRole.description,
          level: updatedRole.level,
        },
      },
    });

    return new RoleDto(updatedRole);
  }

  async updateRolePermissions(
    roleId: string,
    updatePermissionsDto: UpdateRolePermissionsDto,
    updater: User,
  ): Promise<void> {
    const role = await this.roleRepository.findOne({
      where: { id: roleId },
      relations: ['permissions'],
    });

    if (!role) {
      throw new NotFoundException('역할을 찾을 수 없습니다');
    }

    const oldPermissionIds = role.permissions.map(p => p.id);
    const newPermissions = await this.permissionRepository.findByIds(
      updatePermissionsDto.permissionIds,
    );

    role.permissions = newPermissions;
    await this.roleRepository.save(role);

    // 해당 역할을 가진 모든 사용자의 권한 캐시 무효화
    const usersWithRole = await this.userRepository.find({
      where: { roles: { id: roleId } },
    });

    await Promise.all(
      usersWithRole.map(user =>
        this.rbacService.invalidateUserPermissionsCache(user.id),
      ),
    );

    // 감사 로그 기록
    await this.auditLogService.log({
      action: 'ROLE_PERMISSIONS_UPDATED',
      resourceType: 'Role',
      resourceId: roleId,
      userId: updater.id,
      details: {
        oldPermissionIds,
        newPermissionIds: updatePermissionsDto.permissionIds,
        affectedUserCount: usersWithRole.length,
      },
    });
  }

  async getGroupedPermissions(): Promise<GroupedPermissionsDto> {
    const permissions = await this.permissionRepository.find({
      order: { module: 'ASC', resource: 'ASC', action: 'ASC' },
    });

    const grouped = permissions.reduce((acc, permission) => {
      if (!acc[permission.module]) {
        acc[permission.module] = {};
      }
      if (!acc[permission.module][permission.resource]) {
        acc[permission.module][permission.resource] = [];
      }
      acc[permission.module][permission.resource].push(new PermissionDto(permission));
      return acc;
    }, {} as Record<string, Record<string, PermissionDto[]>>);

    return grouped;
  }

  async cloneRole(
    sourceRoleId: string,
    cloneRoleDto: CloneRoleDto,
    creator: User,
  ): Promise<RoleDto> {
    const sourceRole = await this.roleRepository.findOne({
      where: { id: sourceRoleId },
      relations: ['permissions'],
    });

    if (!sourceRole) {
      throw new NotFoundException('복사할 역할을 찾을 수 없습니다');
    }

    const clonedRole = this.roleRepository.create({
      name: cloneRoleDto.name,
      displayName: cloneRoleDto.displayName,
      description: cloneRoleDto.description || `${sourceRole.description} (복사본)`,
      level: cloneRoleDto.level || sourceRole.level - 1,
      isActive: true,
      permissions: sourceRole.permissions,
    });

    const savedRole = await this.roleRepository.save(clonedRole);

    // 감사 로그 기록
    await this.auditLogService.log({
      action: 'ROLE_CLONED',
      resourceType: 'Role',
      resourceId: savedRole.id,
      userId: creator.id,
      details: {
        sourceRoleId,
        sourceRoleName: sourceRole.name,
        newRoleName: savedRole.name,
      },
    });

    return new RoleDto(savedRole);
  }

  async getRoleDeletionImpact(roleId: string): Promise<RoleDeletionImpactDto> {
    const role = await this.roleRepository.findOne({
      where: { id: roleId },
      relations: ['users'],
    });

    if (!role) {
      throw new NotFoundException('역할을 찾을 수 없습니다');
    }

    const affectedUsers = role.users.map(user => ({
      id: user.id,
      name: user.name,
      email: user.email,
    }));

    return {
      roleId,
      roleName: role.name,
      affectedUserCount: affectedUsers.length,
      affectedUsers,
      canDelete: !role.isDefault && affectedUsers.length === 0,
      warnings: [
        ...(role.isDefault ? ['기본 역할은 삭제할 수 없습니다'] : []),
        ...(affectedUsers.length > 0 ? [`${affectedUsers.length}명의 사용자가 영향받습니다`] : []),
      ],
    };
  }
}

export const PermissionTree: React.FC<PermissionTreeProps> = ({

  onPermissionChange,
  loading = false,
}) => {
  const [expandedKeys, setExpandedKeys] = useState<string[]>([]);

  const buildTreeData = useMemo(() => {

              <span className="resource-name">{getResourceDisplayName(resource)}</span>
              <span className="resource-count">({permissions.length}개)</span>
            </div>
          ),
          key: `resource-${module}-${resource}`,
          children: permissions.map(permission => ({
            title: (
              <div className="permission-item">
                <span className="permission-name">{permission.displayName}</span>
                <span className="permission-description">{permission.description}</span>
              </div>
            ),
            key: permission.id,
            isLeaf: true,
          })),
        };

        moduleNode.children.push(resourceNode);
      });

      treeData.push(moduleNode);
    });

    return treeData;
  }, [groupedPermissions]);

  const handleCheck = (checkedKeys: string[]) => {
    // 리프 노드(실제 권한)만 필터링
    const permissionIds = checkedKeys.filter(key => 
      !key.startsWith('module-') && !key.startsWith('resource-')
    );
    onPermissionChange(permissionIds);
  };

  const handleSelectAll = (module: string) => {
    const modulePermissions = Object.values(groupedPermissions[module] || {})
      .flat()
      .map(p => p.id);
    
    const newSelected = [
      ...selectedPermissions.filter(id => 
        !modulePermissions.includes(id)
      ),
      ...modulePermissions,
    ];
    
    onPermissionChange(newSelected);
  };

  const handleDeselectAll = (module: string) => {
    const modulePermissions = Object.values(groupedPermissions[module] || {})
      .flat()
      .map(p => p.id);
    
    const newSelected = selectedPermissions.filter(id => 
      !modulePermissions.includes(id)
    );
    
    onPermissionChange(newSelected);
  };

  return (
    <div className="permission-tree">
      <div className="permission-tree-header">
        <div className="bulk-actions">
          <Button
            size="small"
            onClick={() => setExpandedKeys(buildTreeData.map(node => node.key))}
          >
            모두 펼치기
          </Button>
          <Button
            size="small"
            onClick={() => setExpandedKeys([])}
          >
            모두 접기
          </Button>
        </div>
      </div>

      <Tree
        checkable
        checkedKeys={selectedPermissions}
        expandedKeys={expandedKeys}
        onExpand={setExpandedKeys}
        onCheck={handleCheck}
        treeData={buildTreeData}
        className="permission-tree-content"
        titleRender={(nodeData) => (
          <div className="tree-node-title">
            {nodeData.title}
            {nodeData.key.startsWith('module-') && (
              <div className="module-actions">
                <Button
                  size="small"
                  type="text"
                  onClick={(e) => {
                    e.stopPropagation();
                    handleSelectAll(nodeData.key.replace('module-', ''));
                  }}
                >
                  전체 선택
                </Button>
                <Button
                  size="small"
                  type="text"
                  onClick={(e) => {
                    e.stopPropagation();
                    handleDeselectAll(nodeData.key.replace('module-', ''));
                  }}
                >
                  전체 해제
                </Button>
              </div>
            )}
          </div>
        )}
      />

      {loading && (
        <div className="permission-tree-loading">
          <Spin size="small" />
          <span>권한 업데이트 중...</span>
        </div>
      )}
    </div>
  );
};
```

interface CreateRoleModalProps {

            key="submit"
            type="primary"

            onClick={handleSubmit}
          >
            생성
          </Button>
        ),
      ]}

        current={currentStep}
        items={steps.map(step => ({ title: step.title }))}
        className="create-role-steps"
      />

      <div className="step-content">
        {steps[currentStep].content}
      </div>
    </Modal>
  );
};
