export class PaginationDto {
  @Type(() => Number)
  @IsOptional()
  readonly page: number = 1;

  @Type(() => Number)
  @IsOptional()
  readonly limit: number = 10;

  @IsOptional()
  @IsString()
  readonly sortBy?: string;

  @IsOptional()
  @IsEnum(['ASC', 'DESC'])
  readonly sortOrder?: 'ASC' | 'DESC' = 'ASC';
}

export class PaginatedResponseDto<T> {
  readonly data: T[];
  readonly total: number;
  readonly page: number;
  readonly limit: number;
  readonly totalPages: number;

  constructor(data: T[], total: number, page: number, limit: number) {
    this.data = data;
    this.total = total;
    this.page = page;
    this.limit = limit;
    this.totalPages = Math.ceil(total / limit);
  }
}

### 4. 데이터베이스 쿼리 타입 안전성

#### 4.1 TypeORM 쿼리 빌더 타입 정의
```typescript
export class UserQueryBuilder {
  constructor(private readonly repository: Repository<User>) {}

  createQueryBuilder(): SelectQueryBuilder<User> {
    return this.repository.createQueryBuilder('user');
  }

  findByFilters(filters: UserFiltersDto): Promise<User[]> {
    const qb = this.createQueryBuilder();

    if (filters.name) {
      qb.andWhere('user.name ILIKE :name', { name: `%${filters.name}%` });
    }

    if (filters.status) {
      qb.andWhere('user.status = :status', { status: filters.status });
    }

    if (filters.createdAfter) {
      qb.andWhere('user.createdAt >= :createdAfter', { createdAfter: filters.createdAfter });
    }

    return qb.getMany();
  }
}
```

#### 4.2 Raw Query 타입 정의
```typescript
export interface UserStatsResult {
  totalUsers: number;
  activeUsers: number;
  newUsersThisMonth: number;
}

export class UserStatsRepository {
  constructor(
    @InjectRepository(User)
    private readonly userRepository: Repository<User>
  ) {}

  async getUserStats(): Promise<UserStatsResult> {
    const result = await this.userRepository.query(`
      SELECT 
        COUNT(*) as "totalUsers",
        COUNT(*) FILTER (WHERE status = 'active') as "activeUsers",
        COUNT(*) FILTER (WHERE created_at >= date_trunc('month', CURRENT_DATE)) as "newUsersThisMonth"
      FROM users
    `);

    return result[0] as UserStatsResult;
  }
}
```

#### 6.1 Configuration 타입 정의
```typescript
export interface DatabaseConfig {
  host: string;
  port: number;
  username: string;
  password: string;
  database: string;
}

export interface AppConfig {
  port: number;
  environment: 'development' | 'staging' | 'production';
  jwtSecret: string;
  database: DatabaseConfig;
}

@Injectable()
export class ConfigService {
  private readonly config: AppConfig;

  constructor() {
    this.config = this.validateConfig();
  }

  get<K extends keyof AppConfig>(key: K): AppConfig[K] {
    return this.config[key];
  }

  private validateConfig(): AppConfig {
    // 환경 변수 검증 및 타입 변환
    return {
      port: Number(process.env.PORT) || 3000,
      environment: (process.env.NODE_ENV as AppConfig['environment']) || 'development',
      jwtSecret: process.env.JWT_SECRET || 'default-secret',
      database: {
        host: process.env.DB_HOST || 'localhost',
        port: Number(process.env.DB_PORT) || 5432,
        username: process.env.DB_USERNAME || 'postgres',
        password: process.env.DB_PASSWORD || '',
        database: process.env.DB_DATABASE || 'vanillameta'
      }
    };
  }
}
```
