```javascript
#!/usr/bin/env node
const { exec } = require('child_process');
const path = require('path');

const projects = [
  'backend-api',
  'frontend-web',
  'shared-types'
];

async function runTypeCheck(project) {
  return new Promise((resolve, reject) => {
    const projectPath = path.join(__dirname, '..', project);
    const command = `cd ${projectPath} && npx tsc --noEmit`;
    
    console.log(`🔍 Checking types for ${project}...`);
    
    exec(command, (error, stdout, stderr) => {
      if (error) {
        console.error(`❌ Type check failed for ${project}:`);
        console.error(stderr);
        reject(error);
      } else {
        console.log(`✅ Type check passed for ${project}`);
        resolve(stdout);
      }
    });
  });
}

async function checkAllProjects() {
  let hasErrors = false;
  
  for (const project of projects) {
    try {
      await runTypeCheck(project);
    } catch (error) {
      hasErrors = true;
    }
  }
  
  if (hasErrors) {
    console.error('\n❌ Type check failed for one or more projects');
    process.exit(1);
  } else {
    console.log('\n✅ All type checks passed successfully!');
  }
}

checkAllProjects();
```

#### 1.2 타입 커버리지 측정 도구
**파일**: `/workspace/vanillameta/scripts/type-coverage.js`

```javascript
#!/usr/bin/env node
const { exec } = require('child_process');
const fs = require('fs');
const path = require('path');

class TypeCoverageReporter {
  constructor(project) {
    this.project = project;
    this.projectPath = path.join(__dirname, '..', project);
  }

  async measureCoverage() {
    return new Promise((resolve, reject) => {
      const command = `cd ${this.projectPath} && npx typescript-coverage-report`;
      
      exec(command, (error, stdout, stderr) => {
        if (error) {
          reject(error);
        } else {
          const coverage = this.parseCoverageOutput(stdout);
          resolve(coverage);
        }
      });
    });
  }

  parseCoverageOutput(output) {
    // typescript-coverage-report 출력 파싱
    const lines = output.split('\n');
    const coverageLine = lines.find(line => line.includes('Overall coverage:'));
    
    if (coverageLine) {
      const match = coverageLine.match(/(\d+\.?\d*)%/);
      return match ? parseFloat(match[1]) : 0;
    }
    
    return 0;
  }

  async generateReport() {
    try {
      const coverage = await this.measureCoverage();
      
      const report = {
        project: this.project,
        coverage: coverage,
        timestamp: new Date().toISOString(),
        threshold: 90, // 목표 커버리지
        passed: coverage >= 90
      };
      
      console.log(`📊 ${this.project} Type Coverage: ${coverage}%`);
      
      // 보고서 저장
      const reportPath = path.join(this.projectPath, 'type-coverage-report.json');
      fs.writeFileSync(reportPath, JSON.stringify(report, null, 2));
      
      return report;
    } catch (error) {
      console.error(`Failed to measure type coverage for ${this.project}:`, error);
      return null;
    }
  }
}

async function generateAllReports() {
  const projects = ['backend-api', 'frontend-web', 'shared-types'];
  const reports = [];
  
  for (const project of projects) {
    const reporter = new TypeCoverageReporter(project);
    const report = await reporter.generateReport();
    if (report) reports.push(report);
  }
  
  // 전체 보고서 생성
  const overallReport = {
    timestamp: new Date().toISOString(),
    projects: reports,
    overallPassed: reports.every(r => r.passed)
  };
  
  fs.writeFileSync(
    path.join(__dirname, '..', 'type-coverage-overall.json'),
    JSON.stringify(overallReport, null, 2)
  );
  
  console.log('\n📈 Type coverage report generated');
  
  if (!overallReport.overallPassed) {
    console.error('❌ Some projects failed to meet type coverage threshold');
    process.exit(1);
  }
}

generateAllReports();
```

```yaml
      run: |
        cd ${{ matrix.project }}

    - name: ESLint check (${{ matrix.project }})
      run: |
        cd ${{ matrix.project }}
        npx eslint . --ext .ts,.tsx --max-warnings 0
        
    - name: Type coverage check (${{ matrix.project }})
      run: |
        cd ${{ matrix.project }}
        npx typescript-coverage-report --threshold 90
        
    - name: Upload type coverage report
      uses: actions/upload-artifact@v3
      if: always()
      with:
        name: type-coverage-${{ matrix.project }}
        path: ${{ matrix.project }}/type-coverage-report.json

  type-check-summary:
    needs: type-check
    runs-on: ubuntu-latest
    if: always()
    
    steps:
    - name: Download all coverage reports
      uses: actions/download-artifact@v3
      
    - name: Generate summary report
      run: |
        echo "## TypeScript Type Check Summary" >> $GITHUB_STEP_SUMMARY
        echo "" >> $GITHUB_STEP_SUMMARY
        
        for project in backend-api frontend-web shared-types; do
          if [ -f "type-coverage-${project}/type-coverage-report.json" ]; then
            coverage=$(cat "type-coverage-${project}/type-coverage-report.json" | jq -r '.coverage')
            passed=$(cat "type-coverage-${project}/type-coverage-report.json" | jq -r '.passed')
            
            if [ "$passed" = "true" ]; then
              echo "✅ **${project}**: ${coverage}% coverage (PASSED)" >> $GITHUB_STEP_SUMMARY
            else
              echo "❌ **${project}**: ${coverage}% coverage (FAILED)" >> $GITHUB_STEP_SUMMARY
            fi
          else
            echo "⚠️ **${project}**: Coverage report not found" >> $GITHUB_STEP_SUMMARY
          fi
        done
```

### 5. 타입 품질 메트릭

#### 5.1 타입 품질 리포터
**파일**: `/workspace/vanillameta/scripts/type-quality-report.js`

```javascript
#!/usr/bin/env node
const fs = require('fs');
const path = require('path');
const { exec } = require('child_process');

class TypeQualityReporter {
  constructor() {
    this.metrics = {
      anyTypeUsage: 0,
      totalTypes: 0,
      tsIgnoreUsage: 0,
      typeAssertions: 0,
      missingReturnTypes: 0,
      implicitAnyParams: 0
    };
  }

  async analyzeProject(projectPath) {
    const files = this.getTypeScriptFiles(projectPath);
    
    for (const file of files) {
      await this.analyzeFile(file);
    }
    
    return this.generateReport(projectPath);
  }

  getTypeScriptFiles(dir) {
    let files = [];
    const items = fs.readdirSync(dir);
    
    for (const item of items) {
      const fullPath = path.join(dir, item);
      const stat = fs.statSync(fullPath);
      
      if (stat.isDirectory() && !item.includes('node_modules') && !item.includes('.git')) {
        files = files.concat(this.getTypeScriptFiles(fullPath));
      } else if (item.endsWith('.ts') || item.endsWith('.tsx')) {
        files.push(fullPath);
      }
    }
    
    return files;
  }

  async analyzeFile(filePath) {
    const content = fs.readFileSync(filePath, 'utf8');
    const lines = content.split('\n');
    
    for (const line of lines) {
      // any 타입 사용 검사
      if (line.includes(': any') || line.includes('<any>')) {
        this.metrics.anyTypeUsage++;
      }
      
      // @ts-ignore 사용 검사
      if (line.includes('@ts-ignore')) {
        this.metrics.tsIgnoreUsage++;
      }
      
      // 타입 단언 사용 검사
      if (line.includes(' as ') || line.includes('<')) {
        this.metrics.typeAssertions++;
      }
      
      // 함수/메서드 정의 검사 (간단한 패턴)
      if (line.includes('function ') || line.includes('=> ')) {
        this.metrics.totalTypes++;
        
        // 반환 타입 누락 검사
        if (!line.includes(': ') || !line.includes(') =>')) {
          this.metrics.missingReturnTypes++;
        }
      }
    }
  }

  generateReport(projectPath) {
    const report = {
      project: path.basename(projectPath),
      timestamp: new Date().toISOString(),
      metrics: this.metrics,
      scores: {
        anyTypeScore: Math.max(0, 100 - (this.metrics.anyTypeUsage / this.metrics.totalTypes * 100)),
        tsIgnoreScore: this.metrics.tsIgnoreUsage === 0 ? 100 : 0,
        typeAssertionScore: Math.max(0, 100 - (this.metrics.typeAssertions / this.metrics.totalTypes * 50)),
        returnTypeScore: Math.max(0, 100 - (this.metrics.missingReturnTypes / this.metrics.totalTypes * 100))
      }
    };
    
    report.overallScore = Object.values(report.scores).reduce((sum, score) => sum + score, 0) / Object.keys(report.scores).length;
    
    return report;
  }
}

async function generateQualityReports() {
  const projects = ['backend-api', 'frontend-web', 'shared-types'];
  const reports = [];
  
  for (const project of projects) {
    const projectPath = path.join(__dirname, '..', project);
    if (fs.existsSync(projectPath)) {
      const reporter = new TypeQualityReporter();
      const report = await reporter.analyzeProject(projectPath);
      reports.push(report);
      
      console.log(`📊 Type Quality Report for ${project}:`);
      console.log(`   Overall Score: ${report.overallScore.toFixed(1)}%`);
      console.log(`   Any Type Usage: ${report.metrics.anyTypeUsage}`);
      console.log(`   TS Ignore Usage: ${report.metrics.tsIgnoreUsage}`);
      console.log(`   Type Assertions: ${report.metrics.typeAssertions}`);
      console.log('');
    }
  }
  
  // 전체 보고서 저장
  const overallReport = {
    timestamp: new Date().toISOString(),
    projects: reports,
    summary: {
      averageScore: reports.reduce((sum, r) => sum + r.overallScore, 0) / reports.length,
      totalAnyUsage: reports.reduce((sum, r) => sum + r.metrics.anyTypeUsage, 0),
      totalTsIgnoreUsage: reports.reduce((sum, r) => sum + r.metrics.tsIgnoreUsage, 0)
    }
  };
  
  fs.writeFileSync(
    path.join(__dirname, '..', 'type-quality-report.json'),
    JSON.stringify(overallReport, null, 2)
  );
  
  console.log('📈 Type quality report saved to type-quality-report.json');
  
  if (overallReport.summary.averageScore < 80) {
    console.error('❌ Type quality below threshold (80%)');
    process.exit(1);
  }
}

generateQualityReports();
```

```html
        // 타입 품질 데이터 로드 및 시각화
        async function loadTypeData() {
            try {
                const qualityResponse = await fetch('./type-quality-report.json');
                const qualityData = await qualityResponse.json();
                
                const coverageResponse = await fetch('./type-coverage-overall.json');
                const coverageData = await coverageResponse.json();
                
                renderMetrics(qualityData, coverageData);
                renderCharts(qualityData, coverageData);
            } catch (error) {
                console.error('데이터 로드 실패:', error);
            }
        }
        
        function renderMetrics(qualityData, coverageData) {
            const container = document.getElementById('metrics-container');
            
            // 전체 품질 점수
            const overallScore = qualityData.summary.averageScore;
            const scoreClass = overallScore >= 90 ? 'success' : overallScore >= 70 ? 'warning' : 'error';
            
            container.innerHTML = `
                <div class="metric-card">
                    <div class="metric-value ${scoreClass}">${overallScore.toFixed(1)}%</div>
                    <div class="metric-label">전체 타입 품질 점수</div>
                </div>
                
                <div class="metric-card">
                    <div class="metric-value">${qualityData.summary.totalAnyUsage}</div>
                    <div class="metric-label">any 타입 사용 횟수</div>
                </div>
                
                <div class="metric-card">
                    <div class="metric-value">${qualityData.summary.totalTsIgnoreUsage}</div>
                    <div class="metric-label">@ts-ignore 사용 횟수</div>
                </div>
                
                <div class="metric-card">
                    <div class="metric-value">${coverageData.projects.length}</div>
                    <div class="metric-label">검사된 프로젝트 수</div>
                </div>
            `;
        }
        
        function renderCharts(qualityData, coverageData) {
            // 커버리지 차트
            const coverageCtx = document.getElementById('coverageChart').getContext('2d');
            new Chart(coverageCtx, {
                type: 'bar',
                data: {
                    labels: coverageData.projects.map(p => p.project),
                    datasets: [{
                        label: '타입 커버리지 (%)',
                        data: coverageData.projects.map(p => p.coverage),
                        backgroundColor: 'rgba(54, 162, 235, 0.6)',
                        borderColor: 'rgba(54, 162, 235, 1)',
                        borderWidth: 1
                    }]
                },
                options: {
                    responsive: true,
                    plugins: {
                        title: {
                            display: true,
                            text: '프로젝트별 타입 커버리지'
                        }
                    },
                    scales: {
                        y: {
                            beginAtZero: true,
                            max: 100
                        }
                    }
                }
            });
            
            // 품질 점수 차트
            const qualityCtx = document.getElementById('qualityChart').getContext('2d');
            new Chart(qualityCtx, {
                type: 'radar',
                data: {
                    labels: ['Any 타입 점수', 'TS Ignore 점수', '타입 단언 점수', '반환 타입 점수'],
                    datasets: qualityData.projects.map((project, index) => ({
                        label: project.project,
                        data: [
                            project.scores.anyTypeScore,
                            project.scores.tsIgnoreScore,
                            project.scores.typeAssertionScore,
                            project.scores.returnTypeScore
                        ],
                        borderColor: `hsl(${index * 120}, 70%, 50%)`,
                        backgroundColor: `hsla(${index * 120}, 70%, 50%, 0.1)`
                    }))
                },
                options: {
                    responsive: true,
                    plugins: {
                        title: {
                            display: true,
                            text: '프로젝트별 타입 품질 점수'
                        }
                    },
                    scales: {
                        r: {
                            beginAtZero: true,
                            max: 100
                        }
                    }
                }
            });
        }
        
        // 페이지 로드 시 데이터 로드
        loadTypeData();
        
        // 5분마다 데이터 새로고침
        setInterval(loadTypeData, 5 * 60 * 1000);
```
