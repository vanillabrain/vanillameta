```typescript
function isUser(data: unknown): data is User {

         data !== null && 
         'id' in data && 
         'name' in data;
```

@Controller('users')

  @Post()

  async createUser(
    @Body() createUserDto: CreateUserDto
  ): Promise<ApiResponse<UserResponseDto>> {

## 2. 에러 처리 패턴

### 2.1 Result 패턴
```typescript
type Result<T, E = Error> = 
  | { success: true; data: T }
  | { success: false; error: E };

async function fetchUser(id: number): Promise<Result<User, ApiError>> {
  try {
    const user = await userService.getUser(id);
    return { success: true, data: user };
  } catch (error) {
    return { 
      success: false, 
      error: new ApiError('Failed to fetch user', error) 
    };
  }
}

// 사용법
const result = await fetchUser(1);
if (result.success) {
  console.log(result.data.name); // 타입 안전
} else {
  console.error(result.error.message);
}
```

### 2.2 Custom Error Types
```typescript
abstract class AppError extends Error {
  abstract readonly code: string;
  abstract readonly statusCode: number;
  
  constructor(message: string, public readonly cause?: Error) {
    super(message);
    this.name = this.constructor.name;
  }
}

class ValidationError extends AppError {
  readonly code = 'VALIDATION_ERROR';
  readonly statusCode = 400;
  
  constructor(
    message: string,
    public readonly field: string,
    cause?: Error
  ) {
    super(message, cause);
  }
}

class NotFoundError extends AppError {
  readonly code = 'NOT_FOUND';
  readonly statusCode = 404;
}
```

```typescript
// ✅ Custom Hook 반환 타입 명시
interface UseApiReturn<T> {
  data: T | null;
  loading: boolean;
  error: Error | null;
  refetch: () => Promise<void>;
}

function useApi<T>(url: string): UseApiReturn<T> {
  const [data, setData] = useState<T | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);
  
  const refetch = useCallback(async () => {
    try {
      setLoading(true);
      const response = await fetch(url);
      const result = await response.json();
      setData(result);
    } catch (err) {
      setError(err instanceof Error ? err : new Error('Unknown error'));
    } finally {
      setLoading(false);
    }
  }, [url]);
  
  useEffect(() => {
    refetch();
  }, [refetch]);
  
  return { data, loading, error, refetch };
}
```

```javascript
#!/usr/bin/env node
const fs = require('fs');
const path = require('path');

class TypeQualityMonitor {
  constructor() {
    this.config = {
      reportPath: './type-quality-history.json',
      thresholds: {
        anyUsage: 10,        // any 사용 최대 허용 횟수
        tsIgnore: 0,         // @ts-ignore 사용 허용 안함
        overallScore: 80     // 전체 품질 점수 최소값
      }
    };
  }

  async generateDailyReport() {
    const today = new Date().toISOString().split('T')[0];
    const qualityData = this.loadLatestQualityData();
    
    const report = {
      date: today,
      metrics: qualityData.summary,
      trends: this.calculateTrends(qualityData),
      recommendations: this.generateRecommendations(qualityData)
    };

    this.saveReport(report);
    this.sendSlackNotification(report);
    
    return report;
  }

  calculateTrends(currentData) {
    const history = this.loadReportHistory();
    if (history.length < 2) return null;

    const previous = history[history.length - 2];
    const current = currentData.summary;

    return {
      anyUsageTrend: current.totalAnyUsage - previous.metrics.totalAnyUsage,
      scoreTrend: current.averageScore - previous.metrics.averageScore,
      tsIgnoreTrend: current.totalTsIgnoreUsage - previous.metrics.totalTsIgnoreUsage
    };
  }

  generateRecommendations(qualityData) {
    const recommendations = [];
    
    if (qualityData.summary.totalAnyUsage > this.config.thresholds.anyUsage) {
      recommendations.push({
        type: 'warning',
        message: `any 타입 사용이 ${qualityData.summary.totalAnyUsage}회로 임계값을 초과했습니다.`,
        action: 'any 타입을 구체적인 타입으로 교체하세요.'
      });
    }

    if (qualityData.summary.totalTsIgnoreUsage > 0) {
      recommendations.push({
        type: 'error',
        message: `@ts-ignore가 ${qualityData.summary.totalTsIgnoreUsage}회 사용되었습니다.`,
        action: '타입 문제를 근본적으로 해결하세요.'
      });
    }

    if (qualityData.summary.averageScore < this.config.thresholds.overallScore) {
      recommendations.push({
        type: 'warning',
        message: `전체 타입 품질 점수가 ${qualityData.summary.averageScore}%로 목표치 미달입니다.`,
        action: '타입 정의 개선 작업을 진행하세요.'
      });
    }

    return recommendations;
  }

  async sendSlackNotification(report) {
    // Slack 웹훅을 통한 알림 전송
    if (process.env.SLACK_WEBHOOK_URL) {
      const message = this.formatSlackMessage(report);
      // 실제 Slack API 호출 구현
      console.log('Slack notification sent:', message);
    }
  }

  formatSlackMessage(report) {
    let message = `📊 *TypeScript 품질 일일 리포트* (${report.date})\n\n`;
    
    message += `*전체 점수*: ${report.metrics.averageScore.toFixed(1)}%\n`;
    message += `*any 사용*: ${report.metrics.totalAnyUsage}회\n`;
    message += `*@ts-ignore 사용*: ${report.metrics.totalTsIgnoreUsage}회\n`;

    if (report.trends) {
      message += '\n*전일 대비 변화*:\n';
      message += `점수: ${report.trends.scoreTrend > 0 ? '+' : ''}${report.trends.scoreTrend.toFixed(1)}%\n`;
      message += `any 사용: ${report.trends.anyUsageTrend > 0 ? '+' : ''}${report.trends.anyUsageTrend}회\n`;
    }

    if (report.recommendations.length > 0) {
      message += '\n*개선 권장사항*:\n';
      report.recommendations.forEach(rec => {
        const emoji = rec.type === 'error' ? '🔴' : '⚠️';
        message += `${emoji} ${rec.message}\n`;
      });
    }

    return message;
  }
}
```
