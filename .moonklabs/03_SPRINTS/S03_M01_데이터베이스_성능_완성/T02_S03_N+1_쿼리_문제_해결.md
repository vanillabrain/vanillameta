```typescript
// ❌ N+1 문제 발생 코드
async function getDashboardsWithWidgets(): Promise<Dashboard[]> {
  const dashboards = await dashboardRepository.find();
  
  // 각 대시보드마다 별도 쿼리 실행 (N개의 추가 쿼리)
  for (const dashboard of dashboards) {
    dashboard.widgets = await widgetRepository.find({
      where: { dashboardId: dashboard.id }
    });
  }
  
  return dashboards;
}

// ✅ 해결된 코드 - Eager Loading
async function getDashboardsWithWidgets(): Promise<Dashboard[]> {
  return await dashboardRepository.find({
    relations: ['widgets']
  });
}

// ✅ 해결된 코드 - Query Builder
async function getDashboardsWithWidgets(): Promise<Dashboard[]> {
  return await dashboardRepository
    .createQueryBuilder('dashboard')
    .leftJoinAndSelect('dashboard.widgets', 'widget')
    .getMany();
}
```

### N+1 감지 도구
```typescript
// backend-api/src/database/utils/n-plus-one-detector.ts
export class NPlusOneDetector {
  private queryLog: QueryLog[] = [];
  private threshold = 10; // 의심 기준

  logQuery(query: string, params: any[]): void {
    this.queryLog.push({
      query: this.normalizeQuery(query),
      params,
      timestamp: Date.now()
    });
    
    this.detectNPlusOne();
  }

  private detectNPlusOne(): void {
    const recentQueries = this.getRecentQueries(100);
    const patterns = this.groupByPattern(recentQueries);
    
    for (const [pattern, queries] of patterns) {
      if (queries.length > this.threshold) {
        logger.warn('Potential N+1 detected', {
          pattern,
          count: queries.length,
          sample: queries.slice(0, 3)
        });
      }
    }
  }
}
```

### DataLoader 구현
```typescript
// backend-api/src/common/dataloaders/widget.dataloader.ts
import DataLoader from 'dataloader';

export class WidgetDataLoader {
  private loader: DataLoader<string, Widget[]>;

  constructor(private widgetRepository: Repository<Widget>) {
    this.loader = new DataLoader(this.batchLoadWidgets.bind(this));
  }

  private async batchLoadWidgets(dashboardIds: string[]): Promise<Widget[][]> {
    // 한 번의 쿼리로 모든 위젯 로드
    const widgets = await this.widgetRepository.find({
      where: { dashboardId: In(dashboardIds) }
    });

    // 대시보드 ID별로 그룹화
    const widgetsByDashboard = widgets.reduce((acc, widget) => {
      if (!acc[widget.dashboardId]) {
        acc[widget.dashboardId] = [];
      }
      acc[widget.dashboardId].push(widget);
      return acc;
    }, {} as Record<string, Widget[]>);

    // DataLoader가 기대하는 순서로 반환
    return dashboardIds.map(id => widgetsByDashboard[id] || []);
  }

  async loadByDashboardId(dashboardId: string): Promise<Widget[]> {
    return this.loader.load(dashboardId);
  }
}
```

### 2단계: Relations 전략 구현
```typescript
// backend-api/src/entities/dashboard.entity.ts
@Entity()
export class Dashboard {
  @PrimaryGeneratedColumn('uuid')
  id: string;

  @OneToMany(() => Widget, widget => widget.dashboard)
  widgets: Widget[];

  @ManyToOne(() => User, user => user.dashboards)
  @JoinColumn({ name: 'user_id' })
  user: User;

  @ManyToOne(() => Workspace, workspace => workspace.dashboards)
  @JoinColumn({ name: 'workspace_id' })
  workspace: Workspace;
}

// Repository에서 relations 사용
export class DashboardRepository extends Repository<Dashboard> {
  async findWithRelations(id: string): Promise<Dashboard> {
    return this.findOne({
      where: { id },
      relations: ['widgets', 'user', 'workspace']
    });
  }

  async findAllWithWidgets(): Promise<Dashboard[]> {
    return this.find({
      relations: ['widgets'],
      order: { createdAt: 'DESC' }
    });
  }
}
```

### 3단계: Query Builder 최적화
```typescript
// backend-api/src/modules/dashboard/dashboard.service.ts
export class DashboardService {
  async getComplexDashboardData(workspaceId: string): Promise<DashboardData[]> {
    return await this.dashboardRepository
      .createQueryBuilder('dashboard')
      .leftJoinAndSelect('dashboard.widgets', 'widget')
      .leftJoinAndSelect('widget.dataSource', 'dataSource')
      .leftJoinAndSelect('dashboard.user', 'user')
      .where('dashboard.workspaceId = :workspaceId', { workspaceId })
      .andWhere('dashboard.isActive = :isActive', { isActive: true })
      .orderBy('dashboard.createdAt', 'DESC')
      .getMany();
  }

  // 조건부 조인을 통한 최적화
  async getDashboardsOptimized(options: DashboardQueryOptions): Promise<Dashboard[]> {
    const qb = this.dashboardRepository.createQueryBuilder('dashboard');

    if (options.includeWidgets) {
      qb.leftJoinAndSelect('dashboard.widgets', 'widget');
    }

    if (options.includeUser) {
      qb.leftJoinAndSelect('dashboard.user', 'user');
    }

    if (options.includeWorkspace) {
      qb.leftJoinAndSelect('dashboard.workspace', 'workspace');
    }

    return qb.getMany();
  }
}
```

```typescript
@Injectable()
export class DataLoaderProvider {
  constructor(
    @InjectRepository(Widget)
    private widgetRepository: Repository<Widget>,
    @InjectRepository(User)
    private userRepository: Repository<User>
  ) {}

  createLoaders() {
    return {
      widget: new WidgetDataLoader(this.widgetRepository),
      user: new UserDataLoader(this.userRepository)
    };
  }
}

// GraphQL Resolver에서 사용
@Resolver(() => Dashboard)
export class DashboardResolver {
  @ResolveField(() => [Widget])
  async widgets(
    @Parent() dashboard: Dashboard,
    @Context('loaders') loaders: DataLoaders
  ): Promise<Widget[]> {
    return loaders.widget.loadByDashboardId(dashboard.id);
  }
}
```

### 쿼리 카운트 모니터링
```typescript
// backend-api/src/common/interceptors/query-count.interceptor.ts
@Injectable()
export class QueryCountInterceptor implements NestInterceptor {
  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const queryCountBefore = getQueryCount();
    
    return next.handle().pipe(
      tap(() => {
        const queryCountAfter = getQueryCount();
        const queryCount = queryCountAfter - queryCountBefore;
        
        if (queryCount > 10) {
          logger.warn('High query count detected', {
            endpoint: context.switchToHttp().getRequest().url,
            queryCount
          });
        }
        
        // CloudWatch 메트릭 전송
        sendMetric('QueryCount', queryCount, {
          endpoint: context.switchToHttp().getRequest().url
        });
      })
    );
  }
}
```
