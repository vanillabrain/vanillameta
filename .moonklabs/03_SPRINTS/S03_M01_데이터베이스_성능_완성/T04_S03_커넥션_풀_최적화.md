```typescript
// backend-api/src/config/database/lambda-connection-manager.ts
export class LambdaConnectionManager {
  private static connections: Map<string, Connection> = new Map();
  private static lastActivity: Map<string, number> = new Map();
  private static readonly IDLE_TIMEOUT = 60000; // 1분

  static async getConnection(
    config: ConnectionOptions
  ): Promise<Connection> {
    const key = this.getConnectionKey(config);
    const existingConnection = this.connections.get(key);

    // 기존 연결 재사용
    if (existingConnection && existingConnection.isConnected) {
      this.lastActivity.set(key, Date.now());
      return existingConnection;
    }

    // 새 연결 생성
    const connection = await this.createOptimizedConnection(config);
    this.connections.set(key, connection);
    this.lastActivity.set(key, Date.now());

    // 유휴 연결 정리 스케줄
    this.scheduleCleanup();

    return connection;
  }

  private static async createOptimizedConnection(
    config: ConnectionOptions
  ): Promise<Connection> {
    return createConnection({
      ...config,
      // Lambda 최적화 설정
      extra: {
        ...config.extra,
        // 연결 풀 크기 제한
        max: process.env.IS_LAMBDA ? 1 : 10,
        // 연결 타임아웃
        connectionTimeoutMillis: 5000,
        // 유휴 타임아웃
        idleTimeoutMillis: 30000,
        // 연결 재시도
        retries: 3,
        // SSL 설정 (RDS Proxy 사용 시)
        ssl: process.env.DB_SSL === 'true' ? {
          rejectUnauthorized: false
        } : undefined
      }
    });
  }

  private static scheduleCleanup(): void {
    setTimeout(() => {
      const now = Date.now();
      
      for (const [key, lastActive] of this.lastActivity.entries()) {
        if (now - lastActive > this.IDLE_TIMEOUT) {
          const connection = this.connections.get(key);
          if (connection && connection.isConnected) {
            connection.close();
            this.connections.delete(key);
            this.lastActivity.delete(key);
          }
        }
      }
    }, this.IDLE_TIMEOUT);
  }
}
```

```typescript
export function createTypeOrmConfig(
  dbType: string,
  connectionOptions: Partial<ConnectionOptions>
): ConnectionOptions {
  const poolConfig = databasePoolConfigs[dbType];
  
  return {
    ...connectionOptions,
    type: poolConfig.type as any,
    synchronize: false,
    logging: process.env.NODE_ENV === 'development',

      // 연결 풀 크기

      max: poolConfig.poolSize.max,

      acquireTimeoutMillis: poolConfig.timeouts.acquire,
      idleTimeoutMillis: poolConfig.timeouts.idle,
      evictionRunIntervalMillis: poolConfig.timeouts.eviction,

      validationQuery: getValidationQuery(dbType),
      
      // 추가 최적화
      enableKeepAlive: true,
      keepAliveInitialDelayMillis: 0
    }
  };
}

function getValidationQuery(dbType: string): string {
  switch (dbType) {
    case 'mysql':
    case 'mariadb':
      return 'SELECT 1';
    case 'postgres':
      return 'SELECT 1';
    default:
      return 'SELECT 1';
  }
}
```

### 연결 상태 모니터링
```typescript
// backend-api/src/common/monitoring/connection-pool-monitor.ts
export class ConnectionPoolMonitor {
  private metrics: ConnectionPoolMetrics[] = [];
  private monitoringInterval: NodeJS.Timer;

  constructor(
    private connections: Map<string, Connection>,
    private cloudWatch: CloudWatchService
  ) {}

  startMonitoring(intervalMs: number = 30000): void {
    this.monitoringInterval = setInterval(() => {
      this.collectMetrics();
    }, intervalMs);
  }

  private async collectMetrics(): Promise<void> {
    for (const [name, connection] of this.connections) {
      if (!connection.isConnected) continue;

      const driver = connection.driver as any;
      const pool = driver.pool || driver.master?.pool;

      if (!pool) continue;

      const metrics: ConnectionPoolMetrics = {
        connectionName: name,
        timestamp: new Date(),
        poolSize: pool.size || 0,
        available: pool.available || 0,
        waiting: pool.pending || 0,
        using: (pool.size || 0) - (pool.available || 0),
        maxSize: pool.max || 0
      };

      this.metrics.push(metrics);
      
      // CloudWatch로 메트릭 전송
      await this.sendMetricsToCloudWatch(metrics);
      
      // 경고 조건 확인
      this.checkWarningConditions(metrics);
    }
  }

  private async sendMetricsToCloudWatch(
    metrics: ConnectionPoolMetrics
  ): Promise<void> {
    const namespace = 'VanillaMeta/Database';
    
    await this.cloudWatch.putMetricData({
      Namespace: namespace,
      MetricData: [
        {
          MetricName: 'ConnectionPoolUsage',
          Value: (metrics.using / metrics.maxSize) * 100,
          Unit: 'Percent',
          Dimensions: [
            { Name: 'ConnectionName', Value: metrics.connectionName }
          ]
        },
        {
          MetricName: 'WaitingConnections',
          Value: metrics.waiting,
          Unit: 'Count',
          Dimensions: [
            { Name: 'ConnectionName', Value: metrics.connectionName }
          ]
        }
      ]
    });
  }

  private checkWarningConditions(metrics: ConnectionPoolMetrics): void {
    const usagePercent = (metrics.using / metrics.maxSize) * 100;
    
    // 높은 사용률 경고
    if (usagePercent > 80) {
      logger.warn('High connection pool usage', {
        connectionName: metrics.connectionName,
        usage: `${usagePercent.toFixed(2)}%`,
        using: metrics.using,
        max: metrics.maxSize
      });
    }
    
    // 대기 중인 연결 경고
    if (metrics.waiting > 5) {
      logger.warn('Connections waiting in queue', {
        connectionName: metrics.connectionName,
        waiting: metrics.waiting
      });
    }
  }

  getMetricsReport(): ConnectionPoolReport {
    const grouped = this.groupMetricsByConnection();
    
    return {
      connections: Object.keys(grouped).map(name => ({
        name,
        avgUsage: this.calculateAverage(grouped[name], 'using'),
        maxUsage: Math.max(...grouped[name].map(m => m.using)),
        avgWaiting: this.calculateAverage(grouped[name], 'waiting'),
        maxWaiting: Math.max(...grouped[name].map(m => m.waiting))
      })),
      recommendations: this.generateRecommendations(grouped)
    };
  }
}
```

### RDS Proxy 통합
```typescript
// backend-api/src/config/database/rds-proxy-config.ts
export class RDSProxyConfig {
  static getProxyEndpoint(): string {
    if (process.env.USE_RDS_PROXY === 'true') {
      return process.env.RDS_PROXY_ENDPOINT || '';
    }
    return process.env.DATABASE_HOST || '';
  }

  static getOptimizedConfig(): Partial<ConnectionOptions> {
    if (process.env.USE_RDS_PROXY !== 'true') {
      return {};
    }

    return {
      host: this.getProxyEndpoint(),
      port: 3306,
      extra: {
        // RDS Proxy는 자체 연결 풀을 관리하므로 클라이언트 풀 최소화
        max: 1,
        min: 0,
        // 빠른 연결 해제로 RDS Proxy 연결 슬롯 확보
        idleTimeoutMillis: 1000,
        // SSL 필수
        ssl: {
          rejectUnauthorized: true
        },
        // 연결 재사용 비활성화 (RDS Proxy가 관리)
        pool: false
      }
    };
  }
}
```

```typescript
// backend-api/src/app.module.ts
@Module({
  imports: [
    TypeOrmModule.forRootAsync({
      useFactory: async (configService: ConfigService) => {
        const dbType = configService.get('DATABASE_TYPE');
        const isLambda = configService.get('IS_LAMBDA') === 'true';
        const useRDSProxy = configService.get('USE_RDS_PROXY') === 'true';

        let config = createTypeOrmConfig(dbType, {
          host: configService.get('DATABASE_HOST'),
          port: configService.get('DATABASE_PORT'),
          username: configService.get('DATABASE_USERNAME'),
          password: configService.get('DATABASE_PASSWORD'),
          database: configService.get('DATABASE_NAME')
        });

        // Lambda 환경 특별 처리
        if (isLambda) {
          config = {
            ...config,
            keepConnectionAlive: true,
            ...LambdaConnectionManager.getLambdaConfig()
          };
        }

        // RDS Proxy 사용 시
        if (useRDSProxy) {
          config = {
            ...config,
            ...RDSProxyConfig.getOptimizedConfig()
          };
        }

        return config;
      },
      inject: [ConfigService]
    })
  ]
})
export class AppModule {}
```

```typescript
// backend-api/src/common/providers/connection-monitor.provider.ts
@Injectable()
export class ConnectionMonitorProvider implements OnModuleInit, OnModuleDestroy {
  private monitor: ConnectionPoolMonitor;

  constructor(
    private connection: Connection,
    private cloudWatchService: CloudWatchService
  ) {}

  onModuleInit() {
    this.monitor = new ConnectionPoolMonitor(
      new Map([['main', this.connection]]),
      this.cloudWatchService
    );
    
    this.monitor.startMonitoring(30000); // 30초마다 모니터링
  }

  onModuleDestroy() {
    this.monitor.stopMonitoring();
  }

  getPoolStatus(): ConnectionPoolStatus {
    return this.monitor.getCurrentStatus();
  }
}
```

```typescript
// 항상 finally 블록에서 연결 해제
async function safeQuery<T>(
  queryFn: (connection: Connection) => Promise<T>
): Promise<T> {
  const connection = await getConnection();
  try {
    return await queryFn(connection);
  } finally {
    // Lambda 환경에서는 연결 유지
    if (!process.env.IS_LAMBDA) {
      await connection.close();
    }
  }
}
```
