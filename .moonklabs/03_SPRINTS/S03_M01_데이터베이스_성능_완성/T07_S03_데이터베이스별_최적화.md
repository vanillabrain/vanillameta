# T07_S03: 데이터베이스별 최적화

## 태스크 개요
- **ID**: T07_S03

## 태스크 목표
VanillaMeta가 지원하는 각 데이터베이스(MySQL, PostgreSQL, MariaDB)의 특성을 활용한 최적화 전략을 구현하여 데이터베이스별 최상의 성능을 달성합니다.

## 구현 범위

### 1. 데이터베이스별 특화 기능
- MySQL: 파티셔닝, 스토리지 엔진 최적화
- PostgreSQL: BRIN 인덱스, 파티셔닝, VACUUM 전략
- MariaDB: 컬럼스토어, 시퀀스 엔진

### 2. 쿼리 최적화 전략
- 데이터베이스별 쿼리 힌트
- 옵티마이저 설정 조정
- 특화 함수 활용

### 3. 성능 튜닝
- 버퍼 풀 최적화
- 캐시 설정
- I/O 최적화

// backend-api/src/database/optimization/database-optimizer.ts
export abstract class DatabaseOptimizer {
  abstract optimizeTableStructure(tableName: string): Promise<void>;
  abstract optimizeQuery(query: string): string;
  abstract getOptimalSettings(): Record<string, any>;
  abstract performMaintenance(): Promise<void>;
}

// MySQL 최적화 구현
export class MySQLOptimizer extends DatabaseOptimizer {
  constructor(private connection: Connection) {
    super();
  }

  async optimizeTableStructure(tableName: string): Promise<void> {
    // 테이블 분석
    await this.connection.query(`ANALYZE TABLE ${tableName}`);
    
    // 파티셔닝 검토
    const shouldPartition = await this.shouldUsePartitioning(tableName);
    if (shouldPartition) {
      await this.implementPartitioning(tableName);
    }
    
    // 스토리지 엔진 최적화
    await this.optimizeStorageEngine(tableName);
  }

  private async shouldUsePartitioning(tableName: string): Promise<boolean> {
    const result = await this.connection.query(`
      SELECT 
        table_rows,
        avg_row_length,
        data_length
      FROM information_schema.tables
      WHERE table_schema = DATABASE()
        AND table_name = ?
    `, [tableName]);
    
    const tableInfo = result[0];
    // 1천만 행 이상이거나 10GB 이상인 경우 파티셔닝 권장
    return tableInfo.table_rows > 10000000 || 
           tableInfo.data_length > 10 * 1024 * 1024 * 1024;
  }

  private async implementPartitioning(tableName: string): Promise<void> {
    // 시간 기반 파티셔닝 (created_at 컬럼 기준)
    const partitionSQL = `
      ALTER TABLE ${tableName}
      PARTITION BY RANGE (YEAR(created_at) * 100 + MONTH(created_at)) (
        PARTITION p_202301 VALUES LESS THAN (202302),
        PARTITION p_202302 VALUES LESS THAN (202303),
        PARTITION p_202303 VALUES LESS THAN (202304),
        PARTITION p_202304 VALUES LESS THAN (202305),
        PARTITION p_202305 VALUES LESS THAN (202306),
        PARTITION p_202306 VALUES LESS THAN (202307),
        PARTITION p_future VALUES LESS THAN MAXVALUE
      )
    `;
    
    try {
      await this.connection.query(partitionSQL);
      logger.info(`Partitioning implemented for table ${tableName}`);
    } catch (error) {
      logger.error(`Failed to partition table ${tableName}`, error);
    }
  }

  optimizeQuery(query: string): string {
    let optimized = query;
    
    // USE INDEX 힌트 추가
    if (query.includes('WHERE') && !query.includes('USE INDEX')) {
      optimized = this.addIndexHints(optimized);
    }
    
    // STRAIGHT_JOIN 힌트 (조인 순서 강제)
    if (this.shouldUseStraightJoin(query)) {
      optimized = optimized.replace(/SELECT/i, 'SELECT STRAIGHT_JOIN');
    }
    
    // SQL_CALC_FOUND_ROWS 제거 (성능 저하 원인)
    optimized = optimized.replace(/SQL_CALC_FOUND_ROWS/i, '');
    
    return optimized;
  }

  getOptimalSettings(): Record<string, any> {
    return {
      // InnoDB 버퍼 풀 (전체 메모리의 70-80%)
      innodb_buffer_pool_size: '4G',
      innodb_buffer_pool_instances: 4,
      
      // 쿼리 캐시 (MySQL 8.0에서는 제거됨)
      query_cache_type: 0,
      
      // 조인 버퍼
      join_buffer_size: '256K',
      
      // 임시 테이블
      tmp_table_size: '64M',
      max_heap_table_size: '64M',
      
      // 로그 설정
      innodb_flush_log_at_trx_commit: 2,
      sync_binlog: 0,
      
      // 연결 설정
      max_connections: 200,
      thread_cache_size: 8,
      
      // 인덱스 설정
      innodb_ft_min_token_size: 2,
      ft_min_word_len: 2
    };
  }

  async performMaintenance(): Promise<void> {
    // 테이블 통계 업데이트
    const tables = await this.getTables();
    
    for (const table of tables) {
      await this.connection.query(`ANALYZE TABLE ${table}`);
      
      // 프래그멘테이션 확인 및 최적화
      const fragmentation = await this.checkFragmentation(table);
      if (fragmentation > 20) {
        await this.connection.query(`OPTIMIZE TABLE ${table}`);
      }
    }
  }
}

// PostgreSQL 최적화 구현
export class PostgreSQLOptimizer extends DatabaseOptimizer {
  async optimizeTableStructure(tableName: string): Promise<void> {
    // VACUUM 및 ANALYZE 실행
    await this.connection.query(`VACUUM ANALYZE ${tableName}`);
    
    // 테이블 크기 확인
    const tableSize = await this.getTableSize(tableName);
    
    // 대용량 테이블에 대한 BRIN 인덱스 검토
    if (tableSize.rows > 1000000) {
      await this.considerBRINIndex(tableName);
    }
    
    // 파티셔닝 검토
    if (tableSize.size_gb > 10) {
      await this.implementDeclarativePartitioning(tableName);
    }
  }

  private async considerBRINIndex(tableName: string): Promise<void> {
    // 시계열 데이터에 BRIN 인덱스 적용
    const timeColumns = ['created_at', 'updated_at', 'timestamp'];
    
    for (const column of timeColumns) {
      const hasColumn = await this.checkColumnExists(tableName, column);
      if (hasColumn) {
        const indexName = `idx_brin_${tableName}_${column}`;
        await this.connection.query(`
          CREATE INDEX IF NOT EXISTS ${indexName}
          ON ${tableName} USING BRIN (${column})
        `);
        logger.info(`BRIN index created: ${indexName}`);
      }
    }
  }

  private async implementDeclarativePartitioning(tableName: string): Promise<void> {
    // 선언적 파티셔닝 (PostgreSQL 10+)
    const partitionedTable = `${tableName}_partitioned`;
    
    // 파티션 테이블 생성
    await this.connection.query(`
      CREATE TABLE ${partitionedTable} (LIKE ${tableName} INCLUDING ALL)
      PARTITION BY RANGE (created_at)
    `);
    
    // 월별 파티션 생성
    const startDate = new Date();
    startDate.setMonth(startDate.getMonth() - 12); // 12개월 전부터
    
    for (let i = 0; i < 13; i++) {
      const partitionStart = new Date(startDate);
      partitionStart.setMonth(startDate.getMonth() + i);
      
      const partitionEnd = new Date(partitionStart);
      partitionEnd.setMonth(partitionStart.getMonth() + 1);
      
      const partitionName = `${tableName}_y${partitionStart.getFullYear()}_m${partitionStart.getMonth() + 1}`;
      
      await this.connection.query(`
        CREATE TABLE ${partitionName} 
        PARTITION OF ${partitionedTable}
        FOR VALUES FROM ('${partitionStart.toISOString()}') 
        TO ('${partitionEnd.toISOString()}')
      `);
    }
  }

  optimizeQuery(query: string): string {
    let optimized = query;
    
    // CTE (Common Table Expression) 최적화
    if (query.includes('WITH')) {
      optimized = this.optimizeCTE(optimized);
    }
    
    // 병렬 쿼리 힌트
    if (this.shouldUseParallelQuery(query)) {
      optimized = `/*+ parallel(4) */ ${optimized}`;
    }
    
    // EXISTS vs IN 최적화
    optimized = this.optimizeExistsVsIn(optimized);
    
    return optimized;
  }

  private optimizeCTE(query: string): string {
    // MATERIALIZED 힌트 추가 (PostgreSQL 12+)
    return query.replace(
      /WITH\s+(\w+)\s+AS\s*\(/gi,
      'WITH $1 AS MATERIALIZED ('
    );
  }

  getOptimalSettings(): Record<string, any> {
    return {
      // 메모리 설정
      shared_buffers: '4GB',
      effective_cache_size: '12GB',
      work_mem: '16MB',
      maintenance_work_mem: '256MB',
      
      // WAL 설정
      wal_buffers: '16MB',
      checkpoint_segments: 32,
      checkpoint_completion_target: 0.9,
      
      // 쿼리 플래너
      random_page_cost: 1.1, // SSD 환경
      effective_io_concurrency: 200,
      
      // 병렬 처리
      max_parallel_workers_per_gather: 4,
      max_parallel_workers: 8,
      
      // 통계
      default_statistics_target: 100,
      
      // 연결 관리
      max_connections: 200,
      
      // 자동 VACUUM
      autovacuum: 'on',
      autovacuum_max_workers: 4,
      autovacuum_naptime: '30s'
    };
  }

  async performMaintenance(): Promise<void> {
    // VACUUM 실행
    await this.connection.query('VACUUM ANALYZE');
    
    // 블로트 확인 및 정리
    const bloatedTables = await this.findBloatedTables();
    for (const table of bloatedTables) {
      await this.connection.query(`VACUUM FULL ${table.tableName}`);
    }
    
    // 인덱스 재구축
    const fragmentedIndexes = await this.findFragmentedIndexes();
    for (const index of fragmentedIndexes) {
      await this.connection.query(`REINDEX INDEX ${index.indexName}`);
    }
  }
}

// MariaDB 최적화 구현
export class MariaDBOptimizer extends DatabaseOptimizer {
  async optimizeTableStructure(tableName: string): Promise<void> {
    // 테이블 분석
    await this.connection.query(`ANALYZE TABLE ${tableName} PERSISTENT FOR ALL`);
    
    // 컬럼스토어 인덱스 검토 (MariaDB 10.5+)
    const shouldUseColumnStore = await this.shouldUseColumnStore(tableName);
    if (shouldUseColumnStore) {
      await this.convertToColumnStore(tableName);
    }
    
    // 시퀀스 엔진 활용
    await this.optimizeSequences(tableName);
  }

  private async shouldUseColumnStore(tableName: string): Promise<boolean> {
    // 분석 쿼리가 많고 대용량 데이터인 경우
    const tableStats = await this.getTableStatistics(tableName);
    return tableStats.rows > 5000000 && 
           tableStats.readWriteRatio > 10;
  }

  private async convertToColumnStore(tableName: string): Promise<void> {
    // ColumnStore 테이블로 변환
    const columnStoreTable = `${tableName}_cs`;
    
    await this.connection.query(`
      CREATE TABLE ${columnStoreTable} (
        SELECT * FROM ${tableName}
      ) ENGINE=ColumnStore
    `);
    
    // 인덱스 재생성
    await this.recreateIndexes(tableName, columnStoreTable);
  }

  optimizeQuery(query: string): string {
    let optimized = query;
    
    // MariaDB 특화 힌트
    if (query.includes('JOIN')) {
      optimized = this.addJoinOrderHints(optimized);
    }
    
    // 윈도우 함수 최적화
    if (query.includes('OVER')) {
      optimized = this.optimizeWindowFunctions(optimized);
    }
    
    return optimized;
  }

  getOptimalSettings(): Record<string, any> {
    return {
      // InnoDB 설정 (MySQL과 유사)
      innodb_buffer_pool_size: '4G',
      innodb_buffer_pool_instances: 4,
      
      // MariaDB 특화 설정
      aria_pagecache_buffer_size: '1G',
      
      // 쿼리 캐시
      query_cache_type: 'ON',
      query_cache_size: '256M',
      
      // 스레드 풀 (MariaDB 특화)
      thread_handling: 'pool-of-threads',
      thread_pool_size: 16,
      
      // 옵티마이저
      optimizer_switch: 'rowid_filter=on,index_merge_sort_intersection=on',
      
      // Galera 클러스터 (사용 시)
      wsrep_provider_options: 'gcache.size=1G'
    };
  }

  async performMaintenance(): Promise<void> {
    // 테이블 최적화
    const tables = await this.getTables();
    
    for (const table of tables) {
      // 통계 업데이트
      await this.connection.query(`ANALYZE TABLE ${table} PERSISTENT FOR ALL`);
      
      // 히스토그램 생성 (MariaDB 10.4+)
      await this.createHistograms(table);
      
      // 테이블 정리
      await this.connection.query(`OPTIMIZE TABLE ${table}`);
    }
  }

  private async createHistograms(tableName: string): Promise<void> {
    // 선택도가 낮은 컬럼에 히스토그램 생성
    const columns = await this.getLowSelectivityColumns(tableName);
    
    for (const column of columns) {
      await this.connection.query(`
        ANALYZE TABLE ${tableName} 
        UPDATE HISTOGRAM ON ${column}
      `);
    }
  }
}

// backend-api/src/database/optimization/optimization-service.ts
@Injectable()
export class DatabaseOptimizationService implements OnModuleInit {
  private optimizer: DatabaseOptimizer;
  private optimizationSchedule: NodeJS.Timer;

  constructor(
    private connection: Connection,
    private configService: ConfigService
  ) {}

  async onModuleInit() {
    // 데이터베이스 타입 감지
    this.optimizer = this.createOptimizer();
    
    // 초기 최적화 실행
    await this.applyInitialOptimizations();
    
    // 정기 최적화 스케줄 설정
    this.scheduleOptimizations();
  }

  private createOptimizer(): DatabaseOptimizer {
    switch (this.connection.options.type) {
      case 'mysql':
        return new MySQLOptimizer(this.connection);
      case 'postgres':
        return new PostgreSQLOptimizer(this.connection);
      case 'mariadb':
        return new MariaDBOptimizer(this.connection);
      default:
        throw new Error(`Unsupported database type: ${this.connection.options.type}`);
    }
  }

  private async applyInitialOptimizations(): Promise<void> {
    try {
      // 최적 설정 적용
      const settings = this.optimizer.getOptimalSettings();
      await this.applyDatabaseSettings(settings);
      
      // 주요 테이블 구조 최적화
      const mainTables = ['dashboard', 'widget', 'query', 'connection'];
      for (const table of mainTables) {
        await this.optimizer.optimizeTableStructure(table);
      }
      
      logger.info('Initial database optimizations applied successfully');
    } catch (error) {
      logger.error('Failed to apply initial optimizations', error);
    }
  }

  private scheduleOptimizations(): void {
    // 매일 새벽 2시에 유지보수 작업 실행
    const schedule = '0 2 * * *';
    this.optimizationSchedule = cron.schedule(schedule, async () => {
      await this.performScheduledMaintenance();
    });
  }

  private async performScheduledMaintenance(): Promise<void> {
    logger.info('Starting scheduled database maintenance');
    
    try {
      await this.optimizer.performMaintenance();
      logger.info('Database maintenance completed successfully');
    } catch (error) {
      logger.error('Database maintenance failed', error);
    }
  }

  // 쿼리 인터셉터에서 호출
  optimizeQuery(query: string): string {
    return this.optimizer.optimizeQuery(query);
  }

  // 성능 벤치마크
  async runPerformanceBenchmark(): Promise<BenchmarkResult> {
    const benchmarks = [
      this.benchmarkSimpleSelect(),
      this.benchmarkComplexJoin(),
      this.benchmarkAggregation(),
      this.benchmarkInsertPerformance()
    ];
    
    const results = await Promise.all(benchmarks);
    
    return {
      databaseType: this.connection.options.type as string,
      timestamp: new Date(),
      results,
      overallScore: this.calculateOverallScore(results)
    };
  }

  private async benchmarkSimpleSelect(): Promise<BenchmarkMetric> {
    const iterations = 1000;
    const start = Date.now();
    
    for (let i = 0; i < iterations; i++) {
      await this.connection.query('SELECT 1');
    }
    
    const duration = Date.now() - start;
    
    return {
      name: 'Simple SELECT',
      iterations,
      totalTime: duration,
      avgTime: duration / iterations
    };
  }
}

// backend-api/src/database/patterns/database-patterns.ts
export class DatabaseQueryPatterns {
  static getOptimizedPattern(
    dbType: string,
    pattern: string,
    params: any
  ): string {
    const patterns = this.getPatterns(dbType);
    return patterns[pattern] ? patterns[pattern](params) : '';
  }

  private static getPatterns(dbType: string): Record<string, Function> {
    switch (dbType) {
      case 'mysql':
        return this.getMySQLPatterns();
      case 'postgres':
        return this.getPostgreSQLPatterns();
      case 'mariadb':
        return this.getMariaDBPatterns();
      default:
        return {};
    }
  }

  private static getMySQLPatterns() {
    return {
      upsert: (params: any) => `
        INSERT INTO ${params.table} (${params.columns.join(', ')})
        VALUES (${params.values.map(() => '?').join(', ')})
        ON DUPLICATE KEY UPDATE
        ${params.updateColumns.map(col => `${col} = VALUES(${col})`).join(', ')}
      `,
      
      rankingQuery: (params: any) => `
        SELECT 
          ${params.columns.join(', ')},
          @rank := @rank + 1 AS rank
        FROM ${params.table}, (SELECT @rank := 0) r
        WHERE ${params.conditions}
        ORDER BY ${params.orderBy}
        LIMIT ${params.limit}
      `,
      
      recursiveCTE: (params: any) => `
        WITH RECURSIVE ${params.cteName} AS (
          ${params.initialQuery}
          UNION ALL
          ${params.recursiveQuery}
        )
        SELECT * FROM ${params.cteName}
      `
    };
  }

  private static getPostgreSQLPatterns() {
    return {
      upsert: (params: any) => `
        INSERT INTO ${params.table} (${params.columns.join(', ')})
        VALUES (${params.values.map((_, i) => `$${i + 1}`).join(', ')})
        ON CONFLICT (${params.conflictColumns.join(', ')})
        DO UPDATE SET
        ${params.updateColumns.map(col => `${col} = EXCLUDED.${col}`).join(', ')}
      `,
      
      rankingQuery: (params: any) => `
        SELECT 
          ${params.columns.join(', ')},
          RANK() OVER (ORDER BY ${params.orderBy}) AS rank
        FROM ${params.table}
        WHERE ${params.conditions}
        LIMIT ${params.limit}
      `,
      
      jsonAggregation: (params: any) => `
        SELECT 
          ${params.groupBy},
          JSON_AGG(
            JSON_BUILD_OBJECT(
              ${params.jsonFields.map(f => `'${f.key}', ${f.value}`).join(', ')}
            )
          ) AS ${params.alias}
        FROM ${params.table}
        GROUP BY ${params.groupBy}
      `
    };
  }
}
