```typescript
// backend-api/src/database/analyzer/explain-analyzer.ts
export abstract class ExplainAnalyzer {
  abstract analyze(query: string, params?: any[]): Promise<ExplainResult>;
  abstract getOptimizationSuggestions(result: ExplainResult): Suggestion[];
}

// MySQL/MariaDB EXPLAIN 분석기
export class MySQLExplainAnalyzer extends ExplainAnalyzer {
  constructor(private connection: Connection) {
    super();
  }

  async analyze(query: string, params?: any[]): Promise<ExplainResult> {
    // EXPLAIN 실행
    const explainQuery = `EXPLAIN FORMAT=JSON ${query}`;
    const rawResult = await this.connection.query(explainQuery, params);
    
    // JSON 형식 파싱
    const explainData = JSON.parse(rawResult[0].EXPLAIN);
    
    return this.parseExplainResult(explainData);
  }

  private parseExplainResult(data: any): ExplainResult {
    const queryBlock = data.query_block;
    
    return {
      queryType: queryBlock.select_id,
      tables: this.parseTables(queryBlock),
      cost: queryBlock.cost_info,
      optimizations: this.extractOptimizations(queryBlock),
      warnings: this.checkForWarnings(queryBlock)
    };
  }

  private parseTables(queryBlock: any): TableAccess[] {
    const tables: TableAccess[] = [];
    
    const parseTable = (table: any): TableAccess => ({
      tableName: table.table_name,
      accessType: table.access_type,
      possibleKeys: table.possible_keys || [],
      key: table.key,
      keyLength: table.key_length,
      ref: table.ref,
      rows: table.rows_examined_per_scan,
      filtered: table.filtered,
      extra: table.extra || []
    });

    // 단일 테이블
    if (queryBlock.table) {
      tables.push(parseTable(queryBlock.table));
    }
    
    // 조인된 테이블들
    if (queryBlock.nested_loop) {
      queryBlock.nested_loop.forEach((loop: any) => {
        if (loop.table) {
          tables.push(parseTable(loop.table));
        }
      });
    }
    
    return tables;
  }

  getOptimizationSuggestions(result: ExplainResult): Suggestion[] {
    const suggestions: Suggestion[] = [];
    
    // 테이블 스캔 확인
    result.tables.forEach(table => {
      if (table.accessType === 'ALL') {
        suggestions.push({
          severity: 'high',
          type: 'full_table_scan',
          message: `Table '${table.tableName}' is using full table scan`,
          recommendation: `Consider adding an index on the WHERE clause columns`
        });
      }
      
      // 인덱스 미사용
      if (table.possibleKeys.length > 0 && !table.key) {
        suggestions.push({
          severity: 'medium',
          type: 'index_not_used',
          message: `Table '${table.tableName}' has possible indexes but none are used`,
          recommendation: `Review query conditions and index selectivity`
        });
      }
      
      // Using filesort
      if (table.extra.includes('Using filesort')) {
        suggestions.push({
          severity: 'medium',
          type: 'filesort',
          message: `Query requires filesort for table '${table.tableName}'`,
          recommendation: `Add an index that covers the ORDER BY columns`
        });
      }
    });
    
    return suggestions;
  }
}

// PostgreSQL EXPLAIN 분석기
export class PostgreSQLExplainAnalyzer extends ExplainAnalyzer {
  async analyze(query: string, params?: any[]): Promise<ExplainResult> {
    // EXPLAIN ANALYZE for actual execution stats
    const explainQuery = `EXPLAIN (ANALYZE, BUFFERS, FORMAT JSON) ${query}`;
    const result = await this.connection.query(explainQuery, params);
    
    const plan = result[0]['QUERY PLAN'][0].Plan;
    
    return this.parsePlan(plan);
  }

  private parsePlan(plan: any): ExplainResult {
    return {
      queryType: plan['Node Type'],
      executionTime: plan['Actual Total Time'],
      planningTime: plan['Planning Time'],
      cost: {
        startup: plan['Startup Cost'],
        total: plan['Total Cost']
      },
      nodes: this.parseNodes(plan),
      buffers: this.parseBuffers(plan)
    };
  }

  private parseNodes(node: any): PlanNode[] {
    const nodes: PlanNode[] = [];
    
    const parseNode = (n: any): PlanNode => ({
      nodeType: n['Node Type'],
      relationName: n['Relation Name'],
      scanType: n['Scan Direction'],
      actualRows: n['Actual Rows'],
      actualTime: n['Actual Total Time'],
      loops: n['Actual Loops'],
      filter: n['Filter'],
      indexCond: n['Index Cond'],
      children: n.Plans ? n.Plans.map(parseNode) : []
    });
    
    nodes.push(parseNode(node));
    
    return nodes;
  }

  getOptimizationSuggestions(result: ExplainResult): Suggestion[] {
    const suggestions: Suggestion[] = [];
    
    // Sequential scan on large tables
    result.nodes.forEach(node => {
      if (node.nodeType === 'Seq Scan' && node.actualRows > 10000) {
        suggestions.push({
          severity: 'high',
          type: 'sequential_scan',
          message: `Sequential scan on '${node.relationName}' with ${node.actualRows} rows`,
          recommendation: 'Consider adding an index'
        });
      }
      
      // Nested loops with high iterations
      if (node.nodeType === 'Nested Loop' && node.loops > 1000) {
        suggestions.push({
          severity: 'high',
          type: 'nested_loop',
          message: `Nested loop with ${node.loops} iterations`,
          recommendation: 'Consider using hash join or merge join'
        });
      }
    });
    
    return suggestions;
  }
}
```

### 실행 계획 시각화
```typescript
// backend-api/src/database/analyzer/explain-visualizer.ts
export class ExplainVisualizer {
  static generateVisualization(result: ExplainResult): VisualizationData {
    return {
      type: 'tree',
      nodes: this.buildNodeTree(result),
      metrics: this.extractMetrics(result),
      suggestions: result.suggestions
    };
  }

  private static buildNodeTree(result: ExplainResult): TreeNode[] {
    const nodes: TreeNode[] = [];
    
    if (result.nodes) {
      // PostgreSQL style
      return result.nodes.map(node => this.convertToTreeNode(node));
    } else if (result.tables) {
      // MySQL style
      return result.tables.map((table, index) => ({
        id: `table-${index}`,
        label: table.tableName,
        type: table.accessType,
        metrics: {
          rows: table.rows,
          cost: result.cost?.total_cost || 0
        },
        children: []
      }));
    }
    
    return nodes;
  }

  private static convertToTreeNode(node: PlanNode): TreeNode {
    return {
      id: generateNodeId(),
      label: `${node.nodeType}${node.relationName ? ` on ${node.relationName}` : ''}`,
      type: node.nodeType.toLowerCase().replace(' ', '-'),
      metrics: {
        rows: node.actualRows,
        time: node.actualTime,
        loops: node.loops
      },
      children: node.children?.map(child => this.convertToTreeNode(child)) || []
    };
  }

  static generateReport(result: ExplainResult): string {
    const report = [];
    
    report.push('# Query Execution Plan Analysis\n');
    
    // 실행 시간
    if (result.executionTime) {
      report.push(`**Execution Time**: ${result.executionTime.toFixed(2)}ms`);
    }
    
    // 비용 정보
    if (result.cost) {
      report.push(`**Estimated Cost**: ${result.cost.total || result.cost.total_cost}`);
    }
    
    // 테이블 접근 정보
    if (result.tables) {
      report.push('\n## Table Access:');
      result.tables.forEach(table => {
        report.push(`- **${table.tableName}**`);
        report.push(`  - Access Type: ${table.accessType}`);
        report.push(`  - Key Used: ${table.key || 'None'}`);
        report.push(`  - Rows Examined: ${table.rows}`);
      });
    }
    
    // 최적화 제안
    if (result.suggestions && result.suggestions.length > 0) {
      report.push('\n## Optimization Suggestions:');
      result.suggestions.forEach(suggestion => {
        report.push(`- **${suggestion.severity.toUpperCase()}**: ${suggestion.message}`);
        report.push(`  - ${suggestion.recommendation}`);
      });
    }
    
    return report.join('\n');
  }
}
```

```typescript
// backend-api/src/database/analyzer/auto-analyzer.ts
@Injectable()
export class QueryAutoAnalyzer {
  private analyzer: ExplainAnalyzer;
  private threshold = {
    executionTime: 1000,    // 1초
    rowsExamined: 10000,    // 1만 행
    costThreshold: 1000
  };

  constructor(
    private connection: Connection,
    private metricsService: MetricsService
  ) {
    this.analyzer = this.createAnalyzer();
  }

  private createAnalyzer(): ExplainAnalyzer {
    switch (this.connection.options.type) {
      case 'mysql':
      case 'mariadb':
        return new MySQLExplainAnalyzer(this.connection);
      case 'postgres':
        return new PostgreSQLExplainAnalyzer(this.connection);
      default:
        throw new Error(`Unsupported database type: ${this.connection.options.type}`);
    }
  }

  async analyzeQuery(
    query: string,
    params?: any[],
    context?: QueryContext
  ): Promise<AnalysisResult> {
    const startTime = Date.now();
    
    try {
      // 실행 계획 분석
      const explainResult = await this.analyzer.analyze(query, params);
      
      // 최적화 제안 생성
      const suggestions = this.analyzer.getOptimizationSuggestions(explainResult);
      
      // 성능 문제 감지
      const issues = this.detectPerformanceIssues(explainResult);
      
      // 분석 결과 저장
      const result: AnalysisResult = {
        query,
        context,
        explainResult,
        suggestions,
        issues,
        analysisTime: Date.now() - startTime
      };
      
      // 메트릭 전송
      await this.sendMetrics(result);
      
      // 임계값 초과 시 알림
      if (issues.length > 0) {
        await this.notifyPerformanceIssues(result);
      }
      
      return result;
    } catch (error) {
      logger.error('Query analysis failed', { query, error });
      throw error;
    }
  }

  private detectPerformanceIssues(result: ExplainResult): PerformanceIssue[] {
    const issues: PerformanceIssue[] = [];
    
    // 실행 시간 확인
    if (result.executionTime && result.executionTime > this.threshold.executionTime) {
      issues.push({
        type: 'slow_execution',
        severity: 'high',
        details: `Query execution time ${result.executionTime}ms exceeds threshold`
      });
    }
    
    // 검색된 행 수 확인
    const totalRows = result.tables?.reduce((sum, t) => sum + t.rows, 0) || 0;
    if (totalRows > this.threshold.rowsExamined) {
      issues.push({
        type: 'excessive_rows',
        severity: 'medium',
        details: `Query examines ${totalRows} rows`
      });
    }
    
    return issues;
  }

  async batchAnalyze(queries: QueryInfo[]): Promise<BatchAnalysisResult> {
    const results = await Promise.all(
      queries.map(q => this.analyzeQuery(q.query, q.params, q.context))
    );
    
    return {
      total: queries.length,
      analyzed: results.length,
      issues: results.filter(r => r.issues.length > 0).length,
      suggestions: results.flatMap(r => r.suggestions),
      topIssues: this.groupIssuesByType(results)
    };
  }
}
```

```typescript
// backend-api/src/database/analyzer/performance-baseline.ts
export class PerformanceBaseline {
  private baselines: Map<string, QueryBaseline> = new Map();

  async establishBaseline(
    queryId: string,
    query: string,
    explainResult: ExplainResult
  ): Promise<void> {
    const baseline: QueryBaseline = {
      queryId,
      query,
      establishedAt: new Date(),
      metrics: {
        avgExecutionTime: explainResult.executionTime || 0,
        avgRowsExamined: this.calculateRowsExamined(explainResult),
        avgCost: this.calculateCost(explainResult),
        planHash: this.generatePlanHash(explainResult)
      },
      samples: [explainResult]
    };
    
    this.baselines.set(queryId, baseline);
  }

  async compareWithBaseline(
    queryId: string,
    currentResult: ExplainResult
  ): Promise<BaselineComparison> {
    const baseline = this.baselines.get(queryId);
    if (!baseline) {
      throw new Error(`No baseline found for query ${queryId}`);
    }
    
    const comparison: BaselineComparison = {
      queryId,
      baseline: baseline.metrics,
      current: {
        executionTime: currentResult.executionTime || 0,
        rowsExamined: this.calculateRowsExamined(currentResult),
        cost: this.calculateCost(currentResult),
        planHash: this.generatePlanHash(currentResult)
      },
      deviations: {}
    };
    
    // 계산 편차
    comparison.deviations.executionTime = 
      ((comparison.current.executionTime - baseline.metrics.avgExecutionTime) / 
       baseline.metrics.avgExecutionTime) * 100;
    
    // 실행 계획 변경 감지
    comparison.planChanged = 
      comparison.current.planHash !== baseline.metrics.planHash;
    
    return comparison;
  }

  private generatePlanHash(result: ExplainResult): string {
    // 실행 계획의 고유 해시 생성
    const planString = JSON.stringify({
      tables: result.tables?.map(t => ({
        name: t.tableName,
        accessType: t.accessType,
        key: t.key
      })),
      nodes: result.nodes?.map(n => ({
        type: n.nodeType,
        relation: n.relationName
      }))
    });
    
    return crypto.createHash('md5').update(planString).digest('hex');
  }
}
```

```typescript
// backend-api/src/common/interceptors/query-analysis.interceptor.ts
@Injectable()
export class QueryAnalysisInterceptor implements NestInterceptor {
  constructor(private autoAnalyzer: QueryAutoAnalyzer) {}

  async intercept(
    context: ExecutionContext,
    next: CallHandler
  ): Observable<any> {
    const request = context.switchToHttp().getRequest();
    const queryContext: QueryContext = {
      endpoint: request.url,
      method: request.method,
      user: request.user?.id
    };
    
    // 쿼리 후킹을 위한 이벤트 리스너 설정
    const queryListener = (query: string, params: any[]) => {
      // 비동기로 분석 수행 (응답 지연 방지)
      setImmediate(() => {
        this.autoAnalyzer.analyzeQuery(query, params, queryContext)
          .catch(err => logger.error('Query analysis failed', err));
      });
    };
    
    // TypeORM 쿼리 이벤트에 리스너 등록
    this.connection.driver.afterQuery = queryListener;
    
    return next.handle();
  }
}
```
