```typescript
// backend-api/src/common/monitoring/slow-query-detector.ts
@Injectable()
export class SlowQueryDetector {
  private slowQueryThreshold: number;
  private queryMetrics: Map<string, QueryMetric[]> = new Map();

  constructor(
    private configService: ConfigService,
    private cloudWatchService: CloudWatchService,
    private logger: LoggerService
  ) {
    this.slowQueryThreshold = this.configService.get('SLOW_QUERY_THRESHOLD', 1000); // 1초
  }

  async trackQuery(
    query: string,
    params: any[],
    executionTime: number,
    context?: QueryContext
  ): Promise<void> {
    const queryInfo: QueryInfo = {
      query: this.normalizeQuery(query),
      params,
      executionTime,
      timestamp: new Date(),
      context
    };

    // 슬로우 쿼리 판단
    if (executionTime >= this.slowQueryThreshold) {
      await this.handleSlowQuery(queryInfo);
    }

    // 메트릭 저장
    this.storeMetric(queryInfo);

    // CloudWatch 메트릭 전송
    await this.sendMetrics(queryInfo);
  }

  private async handleSlowQuery(queryInfo: QueryInfo): Promise<void> {
    // 로그 기록
    this.logger.warn('Slow query detected', {
      query: queryInfo.query,
      executionTime: queryInfo.executionTime,
      threshold: this.slowQueryThreshold,
      context: queryInfo.context
    });

    // CloudWatch 알람 트리거
    await this.cloudWatchService.putMetricData({
      Namespace: 'VanillaMeta/Database',
      MetricData: [{
        MetricName: 'SlowQueryCount',
        Value: 1,
        Unit: 'Count',
        Dimensions: [
          { Name: 'Environment', Value: process.env.NODE_ENV },
          { Name: 'QueryType', Value: this.getQueryType(queryInfo.query) }
        ]
      }]
    });

    // 상세 정보 저장
    await this.storeSlowQueryDetails(queryInfo);
  }

  private normalizeQuery(query: string): string {
    // SQL 정규화: 파라미터 제거, 공백 정리 등
    return query
      .replace(/\s+/g, ' ')
      .replace(/\$\d+/g, '?')  // PostgreSQL 파라미터
      .replace(/\?/g, '?')     // MySQL 파라미터
      .trim();
  }

  private getQueryType(query: string): string {
    const normalized = query.toUpperCase().trim();
    if (normalized.startsWith('SELECT')) return 'SELECT';
    if (normalized.startsWith('INSERT')) return 'INSERT';
    if (normalized.startsWith('UPDATE')) return 'UPDATE';
    if (normalized.startsWith('DELETE')) return 'DELETE';
    return 'OTHER';
  }

  async getSlowQueryReport(
    startDate: Date,
    endDate: Date
  ): Promise<SlowQueryReport> {
    const slowQueries = await this.getSlowQueriesInRange(startDate, endDate);
    
    return {
      period: { start: startDate, end: endDate },
      totalSlowQueries: slowQueries.length,
      averageExecutionTime: this.calculateAverage(slowQueries.map(q => q.executionTime)),
      topSlowQueries: this.getTopSlowQueries(slowQueries, 10),
      queryTypeDistribution: this.getQueryTypeDistribution(slowQueries),
      timeDistribution: this.getTimeDistribution(slowQueries),
      recommendations: await this.generateRecommendations(slowQueries)
    };
  }

  private async generateRecommendations(
    slowQueries: QueryInfo[]
  ): Promise<Recommendation[]> {
    const recommendations: Recommendation[] = [];
    const queryPatterns = this.groupByPattern(slowQueries);

    for (const [pattern, queries] of queryPatterns) {
      if (queries.length > 5) {
        recommendations.push({
          severity: 'high',
          pattern,
          occurrences: queries.length,
          avgExecutionTime: this.calculateAverage(queries.map(q => q.executionTime)),
          suggestion: 'This query pattern appears frequently. Consider optimization.'
        });
      }
    }

    return recommendations;
  }
}
```

### TypeORM 쿼리 로거

```typescript
// backend-api/src/common/logging/typeorm-slow-query-logger.ts
export class TypeORMSlowQueryLogger implements Logger {
  constructor(
    private slowQueryDetector: SlowQueryDetector,
    private baseLogger: Logger
  ) {}

  logQuery(query: string, parameters?: any[], queryRunner?: QueryRunner) {
    const startTime = Date.now();
    
    // 원래 로거 호출
    if (this.baseLogger.logQuery) {
      this.baseLogger.logQuery(query, parameters, queryRunner);
    }

    // 쿼리 실행 시간 측정을 위한 후킹
    if (queryRunner) {
      const originalQuery = queryRunner.query.bind(queryRunner);
      
      queryRunner.query = async (query: string, parameters?: any[]) => {
        const queryStartTime = Date.now();
        
        try {
          const result = await originalQuery(query, parameters);
          const executionTime = Date.now() - queryStartTime;
          
          // 슬로우 쿼리 감지
          await this.slowQueryDetector.trackQuery(
            query,
            parameters || [],
            executionTime,
            {
              connection: queryRunner.connection.name,
              transactionId: queryRunner.isTransactionActive ? 
                queryRunner.data.transactionId : undefined
            }
          );
          
          return result;
        } catch (error) {
          // 에러도 로깅
          const executionTime = Date.now() - queryStartTime;
          await this.slowQueryDetector.trackQuery(
            query,
            parameters || [],
            executionTime,
            {
              connection: queryRunner.connection.name,
              error: error.message
            }
          );
          
          throw error;
        }
      };
    }
  }

  // 다른 Logger 메소드들 구현
  logQueryError(error: string, query: string, parameters?: any[], queryRunner?: QueryRunner) {
    if (this.baseLogger.logQueryError) {
      this.baseLogger.logQueryError(error, query, parameters, queryRunner);
    }
  }

  logQuerySlow(time: number, query: string, parameters?: any[], queryRunner?: QueryRunner) {
    if (this.baseLogger.logQuerySlow) {
      this.baseLogger.logQuerySlow(time, query, parameters, queryRunner);
    }
    
    // 추가로 슬로우 쿼리 감지기에도 전송
    this.slowQueryDetector.trackQuery(query, parameters || [], time);
  }

  logSchemaBuild(message: string, queryRunner?: QueryRunner) {
    if (this.baseLogger.logSchemaBuild) {
      this.baseLogger.logSchemaBuild(message, queryRunner);
    }
  }

  logMigration(message: string, queryRunner?: QueryRunner) {
    if (this.baseLogger.logMigration) {
      this.baseLogger.logMigration(message, queryRunner);
    }
  }

  log(level: "log" | "info" | "warn", message: any, queryRunner?: QueryRunner) {
    if (this.baseLogger.log) {
      this.baseLogger.log(level, message, queryRunner);
    }
  }
}
```

```typescript
// backend-api/src/database/config/slow-query-config.ts
export class DatabaseSlowQueryConfig {
  static getMySQLConfig(): string[] {
    return [
      'SET GLOBAL slow_query_log = ON',
      'SET GLOBAL long_query_time = 1',
      'SET GLOBAL log_queries_not_using_indexes = ON',
      `SET GLOBAL slow_query_log_file = '/var/log/mysql/slow.log'`
    ];
  }

  static getPostgreSQLConfig(): Record<string, string> {
    return {
      'log_min_duration_statement': '1000', // 1초
      'log_statement': 'all',
      'log_duration': 'on',
      'log_lock_waits': 'on',
      'log_temp_files': '0'
    };
  }

  static async applyConfiguration(
    connection: Connection
  ): Promise<void> {
    const dbType = connection.options.type;
    
    switch (dbType) {
      case 'mysql':
      case 'mariadb':
        const mysqlCommands = this.getMySQLConfig();
        for (const command of mysqlCommands) {
          try {
            await connection.query(command);
          } catch (error) {
            logger.warn(`Failed to set MySQL config: ${command}`, error);
          }
        }
        break;
        
      case 'postgres':
        const pgConfig = this.getPostgreSQLConfig();
        for (const [key, value] of Object.entries(pgConfig)) {
          try {
            await connection.query(`ALTER SYSTEM SET ${key} = '${value}'`);
          } catch (error) {
            logger.warn(`Failed to set PostgreSQL config: ${key}`, error);
          }
        }
        await connection.query('SELECT pg_reload_conf()');
        break;
    }
  }
}
```

```typescript
@Injectable()
export class SlowQueryReportService {
  constructor(
    private slowQueryDetector: SlowQueryDetector,
    private emailService: EmailService,
    private s3Service: S3Service
  ) {}

  @Cron('0 0 * * MON') // 매주 월요일 자정
  async generateWeeklyReport(): Promise<void> {
    const endDate = new Date();
    const startDate = new Date();
    startDate.setDate(startDate.getDate() - 7);

    const report = await this.slowQueryDetector.getSlowQueryReport(
      startDate,
      endDate
    );

    // HTML 리포트 생성
    const htmlReport = await this.generateHTMLReport(report);
    
    // S3에 저장
    const reportKey = `reports/slow-queries/weekly-${format(endDate, 'yyyy-MM-dd')}.html`;
    await this.s3Service.uploadFile(reportKey, htmlReport, 'text/html');

    // 이메일 발송
    await this.emailService.sendReport({
      to: process.env.ADMIN_EMAILS?.split(',') || [],
      subject: `Weekly Slow Query Report - ${format(endDate, 'yyyy-MM-dd')}`,
      html: htmlReport,
      attachments: [{
        filename: 'slow-queries.csv',
```
